* Data Structures

#+BEGIN_SRC elisp
(defvar code-agent-org--scheduled-blocks nil
  "Alist of scheduled AI blocks.
Keys are CUSTOM-ID strings.
Values are plists: (:file FILE :scheduled-time TIME)
where TIME is Emacs time from `org-get-scheduled-time'.")

(defvar code-agent-org--scheduled-timer nil
  "Timer that checks for due scheduled AI blocks every 60 seconds.")

(defcustom code-agent-org-scheduled-files nil
  "List of files to scan for scheduled AI blocks.
If nil, uses `org-agenda-files'."
  :type '(repeat file)
  :group 'code-agent-org)
#+END_SRC

* Alist Management

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--update-entry (custom-id file scheduled-time)
  "Update or add entry for CUSTOM-ID in scheduled blocks alist.
FILE is the org file path.
SCHEDULED-TIME is Emacs time from `org-get-scheduled-time'."
  (setf (alist-get custom-id code-agent-org--scheduled-blocks nil nil #'string=)
        (list :file file :scheduled-time scheduled-time)))

(defun code-agent-org-scheduled--remove-entry (custom-id)
  "Remove entry for CUSTOM-ID from scheduled blocks alist."
  (setq code-agent-org--scheduled-blocks
        (assoc-delete-all custom-id code-agent-org--scheduled-blocks #'string=)))
#+END_SRC

* Execution Decision Logic

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--should-execute-p (scheduled-time last-executed-str)
  "Return t if AI block should execute based on times.
SCHEDULED-TIME is Emacs time from `org-get-scheduled-time'.
LAST-EXECUTED-STR is string from LAST_AI_EXECUTED property (or nil).

Execution happens when:
1. Scheduled time has arrived (scheduled-time <= now)
2. AND either never executed OR scheduled after last execution"
  (let ((now (current-time))
        (last-executed (when (and last-executed-str
                                  (not (string-empty-p last-executed-str)))
                         (encode-time (parse-time-string last-executed-str)))))
    (and
     ;; Scheduled time has arrived
     (not (time-less-p now scheduled-time))
     ;; Either never executed, or scheduled after last execution
     (or (null last-executed)
         (time-less-p last-executed scheduled-time)))))
#+END_SRC

* AI Block Detection

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--has-ai-block-p ()
  "Return t if heading at point contains an AI block in its subtree.
Only searches within the current heading's subtree."
  (save-excursion
    (let ((subtree-end (save-excursion (org-end-of-subtree t) (point))))
      (re-search-forward "#\\+begin_src ai" subtree-end t))))
#+END_SRC

* File Scanning

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--collect-from-file (file)
  "Collect scheduled AI blocks from FILE.
Returns alist of (CUSTOM-ID . (:file FILE :scheduled-time TIME))."
  (let (results)
    (when (file-exists-p file)
      (with-current-buffer (find-file-noselect file)
        (org-with-wide-buffer
         (goto-char (point-min))
         (while (re-search-forward "^\\*+ " nil t)
           (let* ((custom-id (org-entry-get nil "CUSTOM_ID"))
                  (scheduled-time (org-get-scheduled-time (point))))
             (when (and custom-id scheduled-time
                        (code-agent-org-scheduled--has-ai-block-p))
               (push (cons custom-id
                           (list :file file
                                 :scheduled-time scheduled-time))
                     results)))))))
    results))

(defun code-agent-org-scheduled-scan-all ()
  "Scan org files and build scheduled blocks alist.
Uses `code-agent-org-scheduled-files' or `org-agenda-files' if nil."
  (interactive)
  (setq code-agent-org--scheduled-blocks nil)
  (let ((files (or code-agent-org-scheduled-files
                   (and (fboundp 'org-agenda-files)
                        (org-agenda-files 'unrestricted)))))
    (dolist (file files)
      (dolist (entry (code-agent-org-scheduled--collect-from-file file))
        (push entry code-agent-org--scheduled-blocks)))
    ;; Update mode-line indicator
    (when (fboundp 'code-agent-org-scheduled--update-mode-line)
      (code-agent-org-scheduled--update-mode-line))
    (message "Found %d scheduled AI block(s)"
             (length code-agent-org--scheduled-blocks))))
#+END_SRC

* Timer System

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--check-and-execute ()
  "Check all scheduled AI blocks and execute those that are due.
Called by the timer every 60 seconds."
  (dolist (entry code-agent-org--scheduled-blocks)
    (let* ((custom-id (car entry))
           (plist (cdr entry))
           (file (plist-get plist :file))
           (scheduled-time (plist-get plist :scheduled-time)))
      (code-agent-org-scheduled--maybe-execute file custom-id scheduled-time))))

(defun code-agent-org-scheduled--maybe-execute (file custom-id scheduled-time)
  "Execute AI block if due and not already executed.
FILE is the org file path.
CUSTOM-ID identifies the heading.
SCHEDULED-TIME is when the block should execute.
Returns t if execution started, nil if skipped (busy, not due, etc)."
  (when (file-exists-p file)
    (with-current-buffer (find-file-noselect file)
      (org-with-wide-buffer
       (goto-char (point-min))
       (when (re-search-forward
              (format ":CUSTOM_ID: %s" (regexp-quote custom-id)) nil t)
         (org-back-to-heading t)
         (let ((last-exec-str (org-entry-get nil "LAST_AI_EXECUTED"))
               (heading-pos (point)))
           (when (code-agent-org-scheduled--should-execute-p
                  scheduled-time last-exec-str)
             ;; Find AI block within subtree
             (let ((subtree-end (save-excursion (org-end-of-subtree t) (point))))
               (when (re-search-forward "#\\+begin_src ai" subtree-end t)
                 (forward-line 1)  ; Move INTO the block, not on begin_src line
                 ;; Check if session is busy before executing
                 (let ((session-key (code-agent-org--current-session-key)))
                   (if (code-agent-org--session-get session-key :busy)
                       (message "Skipping scheduled block %s - session busy" custom-id)
                     ;; Session not busy, proceed with execution
                     (message "Executing scheduled AI block: %s" custom-id)
                     (code-agent-org-execute)
                     ;; After execution, advance repeater if present
                     (goto-char heading-pos)
                     (let ((new-time (code-agent-org-scheduled--advance-repeater)))
                       (when new-time
                         ;; Update the in-memory alist with new scheduled time
                         (code-agent-org-scheduled--update-entry custom-id file new-time)
                         (message "Advanced repeater for %s to next occurrence"
                                  custom-id)
                         ;; Save the buffer to persist the change
                         (save-buffer))))))))))))))

(defun code-agent-org-scheduled--advance-repeater ()
  "Advance the SCHEDULED timestamp if it has a repeater.
Must be called with point on the heading.
Returns the new scheduled time if advanced, nil otherwise."
  (let ((repeat (org-get-repeat)))
    (when repeat
      (save-excursion
        ;; Find the SCHEDULED line within the heading
        (let ((heading-end (save-excursion (outline-next-heading) (point))))
          (when (re-search-forward "SCHEDULED:" heading-end t)
            ;; Find the timestamp on this line
            (when (re-search-forward "<[^>]+>" (line-end-position) t)
              (goto-char (match-beginning 0))
              ;; Parse the repeater: +1d, +1w, +1m, .+1d, ++1d, etc.
              (let* ((clean-repeat (if (string-match "^[.+]+" repeat)
                                       (replace-match "" nil nil repeat)
                                     repeat))
                     (n (string-to-number (substring clean-repeat 0 -1)))
                     (unit-char (substring clean-repeat -1))
                     ;; Convert to days - org-timestamp-change works best with 'day
                     (days (pcase unit-char
                             ("d" n)
                             ("w" (* n 7))
                             ("m" (* n 30))  ; Approximate
                             ("y" (* n 365)))))  ; Approximate
                (when (and days (> days 0))
                  (org-timestamp-change days 'day)
                  ;; Return the new scheduled time
                  (org-get-scheduled-time (point)))))))))))

* Mode-Line Indicator

Display scheduled AI blocks count in mode-line with clickable popup.

#+BEGIN_SRC elisp
(defvar code-agent-org-scheduled-string ""
  "Mode-line string showing scheduled AI blocks count.
Clickable to show list of scheduled blocks.")

(defun code-agent-org-scheduled--update-mode-line ()
  "Update the scheduled blocks mode-line indicator."
  (let ((count (length code-agent-org--scheduled-blocks)))
    (if (and code-agent-org--scheduled-timer (> count 0))
        (setq code-agent-org-scheduled-string
              (propertize (format " [S:%d]" count)
                          'face 'font-lock-constant-face
                          'help-echo (format "%d scheduled AI block(s)\nmouse-1: Show schedule list"
                                             count)
                          'mouse-face 'mode-line-highlight
                          'local-map (let ((map (make-sparse-keymap)))
                                       (define-key map [mode-line mouse-1]
                                                   #'code-agent-org-scheduled-list)
                                       map)))
      (setq code-agent-org-scheduled-string "")))
  (force-mode-line-update t))
#+END_SRC

* Timer Start/Stop

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled-start ()
  "Start the scheduled AI block checker.
Scans files and starts a timer that checks every 60 seconds.
Call this from your init.el to enable scheduled execution."
  (interactive)
  (code-agent-org-scheduled-stop)
  (code-agent-org-scheduled-scan-all)
  (setq code-agent-org--scheduled-timer
        (run-at-time 0 60 #'code-agent-org-scheduled--check-and-execute))
  ;; Enable mode-line indicator
  (code-agent-org-scheduled--update-mode-line)
  (unless (member '(:eval code-agent-org-scheduled-string) mode-line-misc-info)
    (push '(:eval code-agent-org-scheduled-string) mode-line-misc-info))
  (force-mode-line-update t)
  (message "Scheduled AI execution started (checking every 60s)"))

(defun code-agent-org-scheduled-stop ()
  "Stop the scheduled AI block checker."
  (interactive)
  (when (timerp code-agent-org--scheduled-timer)
    (cancel-timer code-agent-org--scheduled-timer)
    (setq code-agent-org--scheduled-timer nil))
  ;; Remove mode-line indicator
  (setq code-agent-org-scheduled-string "")
  (setq mode-line-misc-info
        (delete '(:eval code-agent-org-scheduled-string) mode-line-misc-info))
  (force-mode-line-update t)
  (message "Scheduled AI execution stopped"))
#+END_SRC

* List Buffer Mode

A tabulated-list buffer for viewing and navigating to scheduled AI blocks.

#+BEGIN_SRC elisp
(defvar code-agent-org-scheduled-list-mode-map
  (let ((map (make-sparse-keymap)))
    (set-keymap-parent map tabulated-list-mode-map)
    (define-key map (kbd "g") #'code-agent-org-scheduled-list-refresh)
    (define-key map (kbd "RET") #'code-agent-org-scheduled-list-goto)
    map)
  "Keymap for `code-agent-org-scheduled-list-mode'.")

(define-derived-mode code-agent-org-scheduled-list-mode tabulated-list-mode
  "Scheduled AI"
  "Major mode for listing scheduled AI blocks.
\\{code-agent-org-scheduled-list-mode-map}"
  (setq tabulated-list-format
        [("ID" 25 t)
         ("File" 25 t)
         ("Last Run" 18 t)
         ("Next Run" 18 t)])
  (setq tabulated-list-padding 2)
  (setq tabulated-list-sort-key (cons "Next Run" nil))
  (add-hook 'tabulated-list-revert-hook #'code-agent-org-scheduled--revert nil t)
  (tabulated-list-init-header))

(defun code-agent-org-scheduled--revert ()
  "Revert function for scheduled list buffer.
Called by `revert-buffer' via `tabulated-list-revert-hook'."
  (code-agent-org-scheduled-scan-all)
  (setq tabulated-list-entries (code-agent-org-scheduled--build-list-entries)))

(defun code-agent-org-scheduled--get-last-executed (file custom-id)
  "Get LAST_AI_EXECUTED property for CUSTOM-ID in FILE."
  (when (file-exists-p file)
    (with-current-buffer (find-file-noselect file)
      (org-with-wide-buffer
       (goto-char (point-min))
       (when (re-search-forward
              (format ":CUSTOM_ID: %s" (regexp-quote custom-id)) nil t)
         (org-back-to-heading t)
         (org-entry-get nil "LAST_AI_EXECUTED"))))))

(defun code-agent-org-scheduled--format-time (time)
  "Format TIME as human-readable string, or \"Never\" if nil."
  (if time
      (format-time-string "%Y-%m-%d %H:%M" time)
    "Never"))

(defun code-agent-org-scheduled--build-list-entries ()
  "Build tabulated-list entries from scheduled blocks."
  (mapcar
   (lambda (entry)
     (let* ((custom-id (car entry))
            (plist (cdr entry))
            (file (plist-get plist :file))
            (scheduled-time (plist-get plist :scheduled-time))
            (last-exec-str (code-agent-org-scheduled--get-last-executed file custom-id))
            (last-time (when (and last-exec-str (not (string-empty-p last-exec-str)))
                         (encode-time (parse-time-string last-exec-str))))
            (file-basename (file-name-nondirectory file))
            (overdue-p (and scheduled-time
                            (time-less-p scheduled-time (current-time)))))
       (list custom-id
             (vector
              ;; ID column - clickable
              (propertize custom-id
                          'face 'link
                          'action #'code-agent-org-scheduled-list-goto
                          'follow-link t
                          'help-echo "Click to jump to this AI block")
              ;; File column
              file-basename
              ;; Last Run column
              (code-agent-org-scheduled--format-time last-time)
              ;; Next Run column - highlight if overdue
              (propertize (code-agent-org-scheduled--format-time scheduled-time)
                          'face (if overdue-p 'warning 'default))))))
   code-agent-org--scheduled-blocks))

(defun code-agent-org-scheduled-list ()
  "Display a list of all scheduled AI blocks."
  (interactive)
  (let ((buf (get-buffer-create "*Claude Scheduled Blocks*")))
    (with-current-buffer buf
      (code-agent-org-scheduled-list-mode)
      (setq tabulated-list-entries (code-agent-org-scheduled--build-list-entries))
      (tabulated-list-print t))
    (pop-to-buffer buf)))

(defun code-agent-org-scheduled-list-refresh ()
  "Refresh the scheduled blocks list buffer."
  (interactive)
  (code-agent-org-scheduled-scan-all)
  (setq tabulated-list-entries (code-agent-org-scheduled--build-list-entries))
  (tabulated-list-print t)
  (message "Scheduled blocks list refreshed"))

(defun code-agent-org-scheduled-list-goto (&optional _button)
  "Jump to the AI block at point.
BUTTON argument is ignored; we always use `tabulated-list-get-id'."
  (interactive)
  (let ((custom-id (tabulated-list-get-id)))
    (when custom-id
      (code-agent-org-scheduled--goto-block custom-id))))

(defun code-agent-org-scheduled--goto-block (custom-id)
  "Navigate to the AI block with CUSTOM-ID."
  (let ((entry (assoc custom-id code-agent-org--scheduled-blocks #'string=)))
    (when entry
      (let ((file (plist-get (cdr entry) :file)))
        (when (file-exists-p file)
          (find-file file)
          (widen)
          (goto-char (point-min))
          (when (re-search-forward
                 (format ":CUSTOM_ID: %s" (regexp-quote custom-id)) nil t)
            (org-back-to-heading t)
            (org-reveal)
            (recenter)))))))

* org-schedule Advice

#+BEGIN_SRC elisp
(defun code-agent-org-scheduled--after-schedule (&rest _)
  "After advice for `org-schedule'.
Updates the in-memory scheduled blocks list when user schedules a heading.
Only active when scheduler is running."
  (when code-agent-org--scheduled-timer  ; Only if scheduler is running
    (let* ((custom-id (org-entry-get nil "CUSTOM_ID"))
           (scheduled-time (org-get-scheduled-time (point)))
           (file (buffer-file-name)))
      (cond
       ;; Has all required: update entry
       ((and custom-id scheduled-time file
             (code-agent-org-scheduled--has-ai-block-p))
        (code-agent-org-scheduled--update-entry custom-id file scheduled-time)
        (message "Updated schedule for AI block: %s" custom-id))
       ;; Had custom-id but schedule removed: remove entry
       ((and custom-id (not scheduled-time))
        (code-agent-org-scheduled--remove-entry custom-id))))))

(advice-add 'org-schedule :after #'code-agent-org-scheduled--after-schedule)
