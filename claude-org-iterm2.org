iTerm2 backend and Terminal SDD Bridge for claude-org.
Sends prompts to Claude Code running in iTerm2; responses come back via SDD bridge hooks.
Part of claude-org; loaded after claude-org.org.

* Terminal SDD Bridge

Functions for bridging between Claude Code CLI (running in a terminal) and
org-mode SDD sections. Called non-interactively via MCP ~evalElisp~ from
hook scripts in ~scripts/sdd-bridge.sh~.

** Buffer and Navigation

#+BEGIN_SRC elisp
(defun claude-org-sdd-bridge--ensure-buffer (org-file)
  "Ensure ORG-FILE is loaded in Emacs and return its buffer.
Opens the file via `find-file-noselect' if not already visiting."
  (or (find-buffer-visiting org-file)
      (find-file-noselect org-file)))

(defun claude-org-sdd-bridge-list-sessions (org-file)
  "List all SDD sessions in ORG-FILE.
Returns a newline-separated string of \"SESSION_ID\\tHEADING\\tPARENT\" lines,
one per SDD section found. PARENT is the outline path above the SDD heading."
  (with-current-buffer (claude-org-sdd-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (goto-char (point-min))
        (let (results)
          (while (re-search-forward
                  "^[ \t]*:CLAUDE_SESSION_ID:[ \t]+\\(.+?\\)[ \t]*$" nil t)
            (let ((sid (match-string-no-properties 1)))
              (save-excursion
                (org-back-to-heading t)
                (let* ((heading (substring-no-properties (org-get-heading t t t t)))
                       (parent (mapconcat
                                (lambda (s) (substring-no-properties s))
                                (org-get-outline-path)
                                " / ")))
                  (push (format "%s\t%s\t%s" sid heading parent) results)))))
          (mapconcat #'identity (nreverse results) "\n"))))))

(defun claude-org-sdd-bridge--goto-session (session-id)
  "Move point to the SDD section matching SESSION-ID in current buffer.
Searches for :CLAUDE_SESSION_ID: property matching SESSION-ID.
Signals error if not found."
  (goto-char (point-min))
  (let ((found nil))
    (while (and (not found)
                (re-search-forward
                 (format "^[ \t]*:CLAUDE_SESSION_ID:[ \t]+%s[ \t]*$"
                         (regexp-quote session-id))
                 nil t))
      (setq found (point)))
    (unless found
      (error "SDD session %s not found in buffer" session-id))
    ;; Move to the heading that owns this property
    (org-back-to-heading t)))

(defun claude-org-sdd-bridge--find-workflow (session-id)
  "Find the Workflow subsection under the SDD heading for SESSION-ID.
Returns the position of the Workflow heading, or signals error."
  (claude-org-sdd-bridge--goto-session session-id)
  (let ((sdd-end (save-excursion (org-end-of-subtree t t) (point)))
        (found nil))
    (save-excursion
      (while (and (not found)
                  (re-search-forward "^\\*+ +Workflow\\b.*:sdd:" sdd-end t))
        (setq found (line-beginning-position))))
    (unless found
      (error "Workflow section not found under session %s" session-id))
    (goto-char found)))
#+END_SRC

** Prompt Insertion

#+BEGIN_SRC elisp
(defun claude-org-sdd-bridge-insert-prompt (org-file session-id prompt)
  "Insert PROMPT as a new AI block in the SDD Workflow for SESSION-ID in ORG-FILE.
Creates a new Instruction N subsection under the Workflow heading.
Returns the CUSTOM_ID string for correlating with the Stop hook."
  (with-current-buffer (claude-org-sdd-bridge--ensure-buffer org-file)
    (let ((custom-id nil))
      (save-excursion
        (save-restriction
          (widen)
          (claude-org-sdd-bridge--find-workflow session-id)
          (let* ((workflow-level (org-current-level))
                 (instr-level (1+ workflow-level))
                 (stars (make-string instr-level ?*))
                 ;; Use regex to find workflow boundary (safe against embedded headings)
                 (workflow-end (claude-org-sdd-bridge--find-next-sibling-heading
                                workflow-level (point-max)))
                 (instr-count 0))
            ;; Count existing instructions by tag (headings may be renamed by auto-title)
            (save-excursion
              (while (re-search-forward
                      (format "^\\*\\{%d\\} +.+:%s:" instr-level claude-org-heading-tag)
                      workflow-end t)
                (setq instr-count (1+ instr-count))))
            (let* ((instr-num (1+ instr-count))
                   (file-base (file-name-sans-extension (file-name-nondirectory org-file))))
              (setq custom-id (claude-org--generate-instruction-custom-id file-base instr-num session-id))
              ;; Go to end of workflow subtree to insert new instruction
              (goto-char workflow-end)
              ;; Ensure blank line before heading
              (unless (bolp) (insert "\n"))
              (unless (save-excursion (forward-line -1) (looking-at "^[ \t]*$"))
                (insert "\n"))
              (insert (format "%s Instruction %d :%s:\n" stars instr-num claude-org-heading-tag))
              (let ((heading-marker (copy-marker (line-beginning-position 0))))
                (insert ":PROPERTIES:\n")
                (insert (format ":CUSTOM_ID: %s\n" custom-id))
                (insert ":END:\n\n")
                (insert "#+begin_src ai\n")
                (insert prompt "\n")
                (insert "#+end_src\n")
                ;; Register as active query for mode-line and query manager
                (goto-char heading-marker)
                (let ((iterm2-id (gethash session-id claude-org-iterm2--sdd-to-iterm2)))
                  (claude-org-iterm2--register-query session-id iterm2-id))
                ;; Auto-generate a meaningful title for the heading
                (when claude-org-auto-generate-title
                  (claude-org--maybe-auto-generate-title prompt heading-marker)))))))
      (save-buffer)
      custom-id)))
#+END_SRC

** Response Insertion

#+BEGIN_SRC elisp
(defun claude-org-sdd-bridge--shift-org-headings (text parent-level)
  "Shift org heading levels in TEXT to be subsections of PARENT-LEVEL.
Lines starting with *+ get PARENT-LEVEL additional stars prepended,
making them children of the parent heading."
  (let ((prefix (make-string parent-level ?*)))
    (replace-regexp-in-string
     "^\\(\\*+ \\)"
     (concat prefix "\\1")
     text)))

(defun claude-org-sdd-bridge--find-next-sibling-heading (level limit)
  "Find the next heading at LEVEL or shallower before LIMIT.
Searches by regex to avoid `org-end-of-subtree' confusion from
embedded org headings in response content.
Returns the position, or LIMIT if none found."
  (save-excursion
    ;; Skip current heading line
    (forward-line 1)
    (let ((pattern (format "^\\*\\{1,%d\\} " level)))
      (if (re-search-forward pattern limit t)
          (line-beginning-position)
        limit))))

(defun claude-org-sdd-bridge--find-heading-by-custom-id (custom-id limit)
  "Find heading with CUSTOM_ID before LIMIT.  Returns position or nil."
  (save-excursion
    (let ((pattern (format ":CUSTOM_ID: %s" (regexp-quote custom-id))))
      (when (re-search-forward pattern limit t)
        (org-back-to-heading t)
        (point)))))

(defun claude-org-sdd-bridge-insert-response (org-file session-id response instr-custom-id)
  "Insert RESPONSE after the Instruction with INSTR-CUSTOM-ID in ORG-FILE.
Org headings in response content are shifted to become subsections."
  (with-current-buffer (claude-org-sdd-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-sdd-bridge--find-workflow session-id)
        (let* ((workflow-level (org-current-level))
               (instr-level (1+ workflow-level))
               (stars (make-string instr-level ?*))
               (workflow-end (claude-org-sdd-bridge--find-next-sibling-heading
                              workflow-level (point-max)))
               (instr-pos (claude-org-sdd-bridge--find-heading-by-custom-id
                           instr-custom-id workflow-end))
               (response-num 0))
          (unless instr-pos
            (error "Instruction %s not found under Workflow for session %s"
                   instr-custom-id session-id))
          ;; Count which instruction this is among siblings
          (save-excursion
            (let ((found nil))
              (while (and (not found)
                          (re-search-forward
                           (format "^\\*\\{%d\\} +.+:%s:" instr-level claude-org-heading-tag)
                           workflow-end t))
                (setq response-num (1+ response-num))
                (when (= (line-beginning-position) instr-pos)
                  (setq found t)))))
          ;; Insert after the instruction's subtree
          (goto-char instr-pos)
          (let ((insert-pos (claude-org-sdd-bridge--find-next-sibling-heading
                             instr-level workflow-end))
                (timestamp (format-time-string "%Y-%m-%d %H:%M")))
            (goto-char insert-pos)
            (unless (bolp) (insert "\n"))
            (unless (save-excursion (forward-line -1) (looking-at "^[ \t]*$"))
              (insert "\n"))
            (insert (format "%s Response %d (%s) :ai_output:\n"
                            stars response-num timestamp))
            (insert ":PROPERTIES:\n")
            (insert (format ":QUERY_TYPE: terminal\n"))
            (insert ":END:\n\n")
            (insert (claude-org-sdd-bridge--shift-org-headings response instr-level) "\n")))))
    (save-buffer)
    (format "Inserted Response (%d chars)" (length response))))

#+BEGIN_SRC elisp
(defun claude-org-sdd-bridge-update-todos (org-file session-id todos instr-custom-id)
  "Update checkbox list under the Instruction with INSTR-CUSTOM-ID.
TODOS is a list of plists with :content, :status, and :priority keys.
Replaces or creates the checkbox list between the src block and child headings."
  (with-current-buffer (claude-org-sdd-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-sdd-bridge--find-workflow session-id)
        (let* ((workflow-level (org-current-level))
               (instr-level (1+ workflow-level))
               (workflow-end (claude-org-sdd-bridge--find-next-sibling-heading
                              workflow-level (point-max)))
               (last-instr-pos (claude-org-sdd-bridge--find-heading-by-custom-id
                                instr-custom-id workflow-end)))
          (unless last-instr-pos
            (error "Instruction %s not found under Workflow for session %s"
                   instr-custom-id session-id))
          (goto-char last-instr-pos)
          ;; Find the end of the src block (#+end_src)
          (let* ((instr-end (claude-org-sdd-bridge--find-next-sibling-heading
                             instr-level workflow-end))
                 (src-end nil)
                 (todo-start nil)
                 (todo-end nil))
            ;; Find #+end_src within this instruction
            (save-excursion
              (when (re-search-forward "^#\\+end_src" instr-end t)
                (forward-line 1)
                (setq src-end (point))))
            (when src-end
              ;; Find existing todo region: consecutive lines starting with "- ["
              (goto-char src-end)
              ;; Skip blank lines after src block
              (while (and (< (point) instr-end) (looking-at "^[ \t]*$"))
                (forward-line 1))
              (when (looking-at "^- \\[")
                (setq todo-start (point))
                (while (and (< (point) instr-end) (looking-at "^- \\["))
                  (forward-line 1))
                (setq todo-end (point)))
              ;; Build new checkbox text
              (let ((checkbox-text
                     (mapconcat
                      (lambda (todo)
                        (let ((content (plist-get todo :content))
                              (status (plist-get todo :status)))
                          (format "- [%s] %s"
                                  (if (string= status "completed") "X" " ")
                                  content)))
                      todos "\n")))
                (if todo-start
                    ;; Replace existing todo region
                    (progn
                      (delete-region todo-start todo-end)
                      (goto-char todo-start)
                      (insert checkbox-text "\n"))
                  ;; Insert after src block
                  (goto-char src-end)
                  (unless (looking-at "^[ \t]*$")
                    (insert "\n"))
                  (insert checkbox-text "\n"))
                ;; Update heading cookie
                (goto-char last-instr-pos)
                (let* ((done (cl-count-if
                              (lambda (todo) (string= (plist-get todo :status) "completed"))
                              todos))
                       (total (length todos))
                       (cookie (format " [%d/%d]" done total)))
                  ;; Remove existing cookie if any
                  (when (re-search-forward " \\[[0-9]+/[0-9]+\\]" (line-end-position) t)
                    (replace-match ""))
                  ;; Insert cookie before tags
                  (goto-char last-instr-pos)
                  (when (re-search-forward "\\( +:\\)" (line-end-position) t)
                    (goto-char (match-beginning 1))
                    (insert cookie)))))))
        (save-buffer)
        (format "Updated %d todos" (length todos))))))
#+END_SRC

(defcustom claude-org-iterm2-script
  (expand-file-name "python/claude_agent/iterm2_ctl.py"
                    (file-name-directory
                     (or load-file-name buffer-file-name "")))
  "Path to the iterm2_ctl.py module.
Invoked via `python3' call-process."
  :type 'file
  :group 'claude-org)

(defcustom claude-org-iterm2-launch-command 'claude-sdd
  "Launch command for Claude Code in iTerm2.
When set to `claude-sdd', uses the SDD bridge launcher which sets up
hooks, system prompt, MCP config, and plugin directory automatically.
When set to `claude', uses bare Claude CLI with --system-prompt-file.
Can also be a string for a fully custom command."
  :type '(choice (const :tag "claude-sdd (SDD bridge)" claude-sdd)
                 (const :tag "claude (bare CLI)" claude)
                 (string :tag "Custom command"))
  :group 'claude-org)

(defcustom claude-org-iterm2-sdd-script
  (let* ((pkg-dir (file-name-directory
                   (or load-file-name buffer-file-name "")))
         (uv-cmd (format "uv run --project %s claude-sdd"
                         (expand-file-name "python" pkg-dir))))
    uv-cmd)
  "Command to launch the claude-sdd launcher.
Used when `claude-org-iterm2-launch-command' is `claude-sdd'.
Defaults to `uv run --project <pkg>/python claude-sdd'."
  :type 'string
  :group 'claude-org)

(defcustom claude-org-iterm2-extra-args nil
  "Extra command-line arguments to pass to the Claude launch command.
Each element should be a string. These are appended after the base
command and its standard arguments.
Example: (\"--model\" \"opus\" \"--verbose\")"
  :type '(repeat string)
  :group 'claude-org)

** Internal Helpers

#+BEGIN_SRC elisp
(defun claude-org-iterm2--goto-session-heading ()
  "Move point to the nearest ancestor heading with CLAUDE_SESSION_ID.
Starts at current heading, walks up. Returns non-nil if found.
Caller should wrap in `save-excursion' / `save-restriction' as needed."
  (widen)
  (condition-case nil (org-back-to-heading t) (error nil))
  (or (org-entry-get nil "CLAUDE_SESSION_ID")
      (let ((found nil))
        (while (and (not found) (org-up-heading-safe))
          (setq found (org-entry-get nil "CLAUDE_SESSION_ID")))
        found)))

(defun claude-org-iterm2--find-session-property (prop)
  "Find PROP by checking current heading first, then walking up ancestors.
Returns the property value or nil. Uses `org-back-to-heading' to check
the current heading before `org-up-heading-safe' to avoid skipping it."
  (save-excursion
    (save-restriction
      (widen)
      (condition-case nil (org-back-to-heading t) (error nil))
      (or (org-entry-get nil prop)
          (let ((val nil))
            (while (and (not val) (org-up-heading-safe))
              (setq val (org-entry-get nil prop)))
            val)))))

(defun claude-org-iterm2--call (subcommand &rest args)
  "Call iterm2-ctl.py SUBCOMMAND with ARGS. Return stdout string.
If ARGS contains :stdin KEY, that value is written to a temp file
and used as stdin for the process (avoids ARG_MAX for large prompts)."
  (let* ((stdin-text nil)
         (cli-args nil))
    ;; Separate :stdin from positional args
    (while args
      (if (eq (car args) :stdin)
          (progn
            (setq stdin-text (cadr args))
            (setq args (cddr args)))
        (push (car args) cli-args)
        (setq args (cdr args))))
    (setq cli-args (nreverse cli-args))
    (let* ((flat-args (cons subcommand (flatten-tree cli-args)))
           (stderr-file (make-temp-file "iterm2-stderr-"))
           (buf (generate-new-buffer " *iterm2-ctl*"))
           (stdin-file (when stdin-text
                         (let ((f (make-temp-file "iterm2-stdin-")))
                           (with-temp-file f (insert stdin-text))
                           f)))
           ;; Separate stderr to temp file so it doesn't pollute stdout
           (exit-code (apply #'call-process
                             "python3" stdin-file
                             (list buf stderr-file) nil
                             claude-org-iterm2-script flat-args)))
      (unwind-protect
          (with-current-buffer buf
            (if (= exit-code 0)
                (string-trim (buffer-string))
              ;; Include stderr in error message for diagnostics
              (let ((stderr (string-trim
                             (with-temp-buffer
                               (insert-file-contents stderr-file)
                               (buffer-string)))))
                (error "iterm2-ctl %s failed (exit %d): %s"
                       subcommand exit-code stderr))))
        (kill-buffer buf)
        (when stdin-file (delete-file stdin-file t))
        (delete-file stderr-file t)))))

(defun claude-org-iterm2--tab-title ()
  "Generate iTerm2 tab title for current SDD session.
Format: HEADING [SDD-SESSION-ID] — unique across org files."
  (save-excursion
    (save-restriction
      (let ((sdd-id (claude-org-iterm2--goto-session-heading)))
        (format "%s [%s]"
                (truncate-string-to-width
                 (or (org-get-heading t t t t) "untitled") 40)
                (or sdd-id "no-session"))))))

(defun claude-org-iterm2--build-launch-command (org-file sdd-session-id project-root)
  "Build the shell command to launch Claude Code.
Uses `claude-org-iterm2-launch-command', `claude-org-iterm2-extra-args',
and the org property CLAUDE_EXTRA_ARGS (space-separated CLI flags).
ORG-FILE is the org notebook path, SDD-SESSION-ID the session,
PROJECT-ROOT the working directory."
  (ignore project-root)
  (let* ((prop-args (condition-case nil
                        (claude-org-iterm2--find-session-property
                         "CLAUDE_EXTRA_ARGS")
                      (error nil)))
         (all-extra (append (or claude-org-iterm2-extra-args nil)
                            (when prop-args (split-string prop-args))))
         (extra (mapconcat #'shell-quote-argument all-extra " ")))
    (cond
     ;; claude-sdd: passes org-file and session-id, handles everything
     ;; Also passes --resume from CLAUDE_CLI_SESSION if available, so
     ;; claude-sdd can resume even when MCP is unreachable.
     ((eq claude-org-iterm2-launch-command 'claude-sdd)
      (let* ((sdd-script claude-org-iterm2-sdd-script)
             (cli-session
              (condition-case nil
                  (claude-org-iterm2--find-session-property
                   "CLAUDE_CLI_SESSION")
                (error nil)))
             (parts (list sdd-script
                          (shell-quote-argument (or org-file ""))
                          (shell-quote-argument (or sdd-session-id "")))))
        ;; Pass --resume before extra args so claude-sdd can use it
        (when cli-session
          (setq parts (append parts
                              (list "--"
                                    "--resume"
                                    (shell-quote-argument cli-session)))))
        (when (and extra (not (string-empty-p extra)))
          (unless cli-session
            (setq parts (append parts (list "--"))))
          (setq parts (append parts (list extra))))
        (mapconcat #'identity parts " ")))
     ;; bare claude: build --system-prompt-file and --resume manually
     ((eq claude-org-iterm2-launch-command 'claude)
      (let* ((system-prompt
              (condition-case nil
                  (claude-org-sdd-bridge-system-prompt org-file sdd-session-id)
                (error "")))
             (sp-file (make-temp-file "claude-sp-"))
             (cli-session
              (condition-case nil
                  (claude-org-sdd-bridge-get-cli-session org-file sdd-session-id)
                (error nil)))
             (parts (list "claude")))
        (with-temp-file sp-file (insert (or system-prompt "")))
        (push (format "--system-prompt \"$(cat %s)\"" (shell-quote-argument sp-file))
              parts)
        (when cli-session
          (push (format "--resume %s" (shell-quote-argument cli-session)) parts))
        (when (and extra (not (string-empty-p extra)))
          (push extra parts))
        (mapconcat #'identity (nreverse parts) " ")))
     ;; Custom string command
     ((stringp claude-org-iterm2-launch-command)
      (if (and extra (not (string-empty-p extra)))
          (concat claude-org-iterm2-launch-command " " extra)
        claude-org-iterm2-launch-command))
     (t (error "Invalid claude-org-iterm2-launch-command: %S"
               claude-org-iterm2-launch-command)))))

(defun claude-org-iterm2--ensure-session ()
  "Ensure an iTerm2 session exists for the current SDD story.
Returns the iTerm2 session UUID. Launches if needed, reuses by title.
Auto-starts the Emacs MCP server if not running (claude-sdd needs it)."
  ;; Best-effort MCP server auto-start (claude-sdd degrades gracefully without it)
  (when (and (fboundp 'emacs-mcp-server-running-p)
             (not (emacs-mcp-server-running-p))
             (fboundp 'emacs-mcp-server-start))
    (condition-case err
        (progn
          (emacs-mcp-server-start)
          (sleep-for 1)
          (if (emacs-mcp-server-running-p)
              (message "Started Emacs MCP server for SDD bridge")
            (message "WARNING: MCP server didn't start — SDD bridge will be limited")))
      (error (message "WARNING: MCP server start failed: %s" err))))
  (let* ((saved-id (claude-org-iterm2--find-session-property "ITERM2_SESSION_ID"))
         (title (claude-org-iterm2--tab-title))
         (sdd-session-id
          (claude-org-iterm2--find-session-property "CLAUDE_SESSION_ID"))
         (project-root (or (claude-org--get-project-root)
                           default-directory)))
    ;; Ensure IDE server is running for ALL session paths (new or existing)
    (when sdd-session-id
      (claude-org-iterm2--ensure-ide-server project-root sdd-session-id))
    ;; Check if saved session is alive
    (if (and saved-id
             (condition-case nil
                 (let* ((json-str (claude-org-iterm2--call
                                   "status" "--session" saved-id))
                        (data (json-read-from-string json-str)))
                   (not (string= (cdr (assq 'state data)) "dead")))
               (error nil)))
        saved-id
      ;; Launch: finds existing tab by title or creates new
      (let* ((_ (unless sdd-session-id
                  (user-error
                   "No CLAUDE_SESSION_ID found. Create an SDD section first")))
             (org-file (buffer-file-name))
             ;; Build the launch command string based on defcustom
             (launch-cmd (claude-org-iterm2--build-launch-command
                          org-file sdd-session-id project-root)))
        (let ((result (apply #'claude-org-iterm2--call
                       "launch"
                       "--title" title
                       "--cwd" project-root
                       "--launch-cmd" launch-cmd
                       nil)))
              ;; Save ITERM2_SESSION_ID on the heading that has CLAUDE_SESSION_ID
              (save-excursion
                (save-restriction
                  (claude-org-iterm2--goto-session-heading)
                  (org-set-property "ITERM2_SESSION_ID" result)))
              ;; Store mapping for polling and permission-click lookup
              (when sdd-session-id
                (claude-org--session-put
                 (claude-org--current-session-key)
                 :iterm2-session result)
                (puthash sdd-session-id result
                         claude-org-iterm2--sdd-to-iterm2)
                (puthash sdd-session-id (claude-org--current-session-key)
                         claude-org-iterm2--sdd-to-session-key))
              (save-buffer)
              (message "Claude Code in iTerm2 (session: %s)" result)
              result)))))

(defun claude-org-iterm2--read-hook-status (sdd-session-id)
  "Read Claude Code status from hook-written file. Fast, no API call."
  (let ((file (expand-file-name (or sdd-session-id "unknown")
                                "/tmp/claude-agent-status")))
    (if (file-exists-p file)
        (string-trim (with-temp-buffer
                       (insert-file-contents file)
                       (buffer-string)))
      "unknown")))

(defun claude-org-iterm2--get-status (sdd-session-id iterm2-session-id)
  "Get Claude Code status.
SDD-SESSION-ID keys the hook file; ITERM2-SESSION-ID keys the API."
  (let ((hook-status (claude-org-iterm2--read-hook-status sdd-session-id)))
    (if (member hook-status '("ready" "busy"))
        hook-status
      ;; Fallback to iTerm2 API
      (condition-case nil
          (let* ((json-str (claude-org-iterm2--call "status"
                            "--session" iterm2-session-id))
                 (data (json-read-from-string json-str)))
            (cdr (assq 'state data)))
        (error "unknown")))))

** Query Registration

Register iTerm2 queries into =claude-agent--active-queries= so they appear
in the global mode-line spinner and =*Claude Queries*= buffer alongside
native JSON stream queries.

#+BEGIN_SRC elisp
(defun claude-org-iterm2--save-request-id (sdd-session-id request-id)
  "Save REQUEST-ID to status file for SDD-SESSION-ID.
The Python response hook reads this to call query-completed."
  (let ((dir "/tmp/claude-agent-status"))
    (make-directory dir t)
    (with-temp-file (expand-file-name
                     (concat sdd-session-id ".request-id") dir)
      (insert request-id))))

(defun claude-org-iterm2--read-request-id (sdd-session-id)
  "Read the request-id for SDD-SESSION-ID, or nil if not set."
  (let ((path (expand-file-name
               (concat sdd-session-id ".request-id")
               "/tmp/claude-agent-status")))
    (when (file-exists-p path)
      (let ((val (string-trim (with-temp-buffer
                                (insert-file-contents path)
                                (buffer-string)))))
        (unless (string-empty-p val) val)))))

(defun claude-org-iterm2--register-query (sdd-session-id iterm2-id)
  "Register the current AI block as an active query.
Creates a process-state, registers in `claude-agent--active-queries',
and saves the request-id to a status file for the Python hook.
SDD-SESSION-ID is the SDD session, ITERM2-ID is the iTerm2 pane UUID."
  (let* ((custom-id (org-entry-get nil "CUSTOM_ID"))
         (heading (org-get-heading t t t t))
         (instr-num (when (and heading
                               (string-match "\\bInstruction \\([0-9]+\\)" heading))
                      (string-to-number (match-string 1 heading))))
         (req-id (claude-agent--generate-request-id))
         (ctx (claude-agent-make-query-context
               :instruction-num instr-num
               :custom-id custom-id))
         (state (claude-agent--make-process-state
                 :request-id req-id
                 :start-time (float-time)
                 :source-buffer (current-buffer)
                 :query-context ctx
                 :source-location-fn (when custom-id #'claude-org-goto-custom-id)
                 :source-location-args (when custom-id (list custom-id)))))
    (claude-agent--register-query req-id state)
    (claude-org-iterm2--save-request-id sdd-session-id req-id)))

(defun claude-org-iterm2--query-completed (sdd-session-id)
  "Mark the iTerm2 query for SDD-SESSION-ID as completed.
Unregisters from `claude-agent--active-queries' so the mode-line
and *Claude Queries* buffer update automatically.
When loop state is active, schedules the next iteration."
  (let ((req-id (claude-org-iterm2--read-request-id sdd-session-id)))
    (when req-id
      (let ((state (claude-agent--get-active-query req-id)))
        (when state
          (setf (claude-agent--process-state-closed state) t)
          (claude-agent--unregister-query req-id)))))
  ;; Check loop continuation
  (let ((session-key (gethash sdd-session-id claude-org-iterm2--sdd-to-session-key)))
    (when session-key
      (let ((loop-current (or (claude-org--session-get session-key :loop-current) 1))
            (loop-max (or (claude-org--session-get session-key :loop-max) 1)))
        (if (< loop-current loop-max)
            ;; More iterations — schedule next
            (let ((interval (or (claude-org--session-get session-key :loop-interval) 0))
                  (prompt (claude-org--session-get session-key :original-prompt))
                  (iterm2-id (gethash sdd-session-id claude-org-iterm2--sdd-to-iterm2)))
              (claude-org--session-put session-key :loop-current (1+ loop-current))
              (if (> interval 0)
                  (run-at-time interval nil
                               #'claude-org-iterm2--loop-send
                               sdd-session-id iterm2-id prompt session-key)
                (claude-org-iterm2--loop-send sdd-session-id iterm2-id prompt session-key)))
          ;; Loop complete — cleanup
          (claude-org--session-put session-key :loop-max nil)
          (claude-org--session-put session-key :loop-current nil))))))


(defun claude-org-iterm2--loop-send (sdd-session-id iterm2-id prompt session-key)
  "Send next loop iteration PROMPT to iTerm2.
SDD-SESSION-ID and ITERM2-ID identify the session.
SESSION-KEY is the org session for loop state.
Navigates to saved org buffer/marker for org-context-dependent calls."
  (let ((org-buf (claude-org--session-get session-key :loop-org-buffer))
        (marker (claude-org--session-get session-key :loop-block-marker)))
    ;; write-from-emacs-flag and register-query need org buffer context
    (when (and org-buf (buffer-live-p org-buf) marker (marker-buffer marker))
      (with-current-buffer org-buf
        (save-excursion
          (goto-char marker)
          (when sdd-session-id
            (claude-org-iterm2--write-from-emacs-flag sdd-session-id))
          (save-excursion
            (org-back-to-heading t)
            (when sdd-session-id
              (claude-org-iterm2--register-query sdd-session-id iterm2-id)))))))
  (claude-org-iterm2--call "send" "--session" iterm2-id :stdin prompt)
  (let ((current (claude-org--session-get session-key :loop-current))
        (max (claude-org--session-get session-key :loop-max)))
    (message "Sent loop iteration %d/%d to iTerm2 (%d chars)"
             current max (length prompt))))
#+END_SRC

** Execution

#+BEGIN_SRC elisp
(defun claude-org-iterm2--write-from-emacs-flag (sdd-session-id)
  "Write flag file indicating the next prompt was sent from Emacs.
sdd-bridge.sh checks this to skip duplicate instruction insertion.
Also writes the current instruction's CUSTOM_ID for response correlation."
  (let ((dir "/tmp/claude-agent-status"))
    (make-directory dir t)
    (with-temp-file (expand-file-name
                     (concat sdd-session-id ".from-emacs") dir)
      (insert "1"))
    ;; Save CUSTOM_ID of the current instruction for response hook
    (let ((custom-id (org-entry-get nil "CUSTOM_ID")))
      (when custom-id
        (with-temp-file (expand-file-name
                         (concat sdd-session-id ".custom-id") dir)
          (insert custom-id))))))

(defun claude-org-iterm2--execute-ai-block ()
  "Execute current AI block via iTerm2 backend.
Ensures session exists, checks readiness, sends prompt via stdin.
Writes a from-emacs flag so the prompt hook skips duplicate insertion.
Registers in `claude-agent--active-queries' for mode-line and query manager."
  (let* ((element (org-element-at-point))
         (prompt (when (and (eq (org-element-type element) 'src-block)
                            (string= (org-element-property :language element) "ai"))
                   (string-trim (org-element-property :value element)))))
    (unless prompt
      (user-error "Point is not inside a #+begin_src ai block"))
    (let* ((iterm2-id (claude-org-iterm2--ensure-session))
           (sdd-session-id
            (claude-org-iterm2--find-session-property "CLAUDE_SESSION_ID"))
           (status (claude-org-iterm2--get-status sdd-session-id iterm2-id)))
      (when (string= status "busy")
        (user-error "Claude Code busy (status: %s). Cancel first?" status))
      ;; Ensure sdd→session-key mapping for loop dispatch from query-completed
      (when sdd-session-id
        (puthash sdd-session-id (claude-org--current-session-key)
                 claude-org-iterm2--sdd-to-session-key))
      ;; Save org buffer + marker for loop-send (timer callbacks need context)
      (claude-org--session-put (claude-org--current-session-key)
                               :loop-org-buffer (current-buffer))
      (claude-org--session-put (claude-org--current-session-key)
                               :loop-block-marker (copy-marker (point)))
      ;; Write flag BEFORE sending — hook checks and deletes it
      (when sdd-session-id
        (claude-org-iterm2--write-from-emacs-flag sdd-session-id))
      ;; Register in active-queries for mode-line and *Claude Queries* display
      (save-excursion
        (org-back-to-heading t)
        (when sdd-session-id
          (claude-org-iterm2--register-query sdd-session-id iterm2-id)))
      (claude-org-iterm2--call "send" "--session" iterm2-id :stdin prompt)
      (message "Sent to iTerm2 (%d chars)" (length prompt)))))

(defun claude-org-iterm2-open-tab ()
  "Open (or focus) the iTerm2 tab for this SDD session.
Launches Claude Code if no matching tab exists."
  (interactive)
  (let ((session-id (claude-org-iterm2--ensure-session)))
    (claude-org-iterm2--call "focus" "--session" session-id)
    (message "iTerm2 tab focused (session: %s)" session-id)))

(defun claude-org-iterm2--ensure-ide-server (project-root sdd-session-id)
  "Ensure IDE WebSocket server is running for this SDD session.
Starts the server if not already alive.
The lockfile must exist before Claude Code launches."
  (when (fboundp 'claude-ide--session-alive-p)
    (unless (claude-ide--session-alive-p sdd-session-id)
      (let ((port (claude-ide--session-port
                   (claude-ide-start-server-in-directory sdd-session-id project-root))))
        (message "IDE server started on port %d for %s"
                 port (file-name-nondirectory project-root))))))

(defun claude-org-iterm2-cancel ()
  "Cancel the current Claude Code instruction in iTerm2.
Sends Escape to the TUI. The Stop hook will fire and set-ready."
  (interactive)
  (let ((iterm2-id (claude-org-iterm2--find-session-property "ITERM2_SESSION_ID")))
    (unless iterm2-id
      (user-error "No iTerm2 session for this story"))
    (claude-org-iterm2--call "cancel" "--session" iterm2-id)
    (message "Cancel sent to Claude Code (waiting for hook...)")))

(defun claude-org-iterm2--permission-needed (sdd-session-id tool-name)
  "Handle notification that SDD-SESSION-ID needs permission for TOOL-NAME.
Called by the PreToolUse hook via MCP. Registers the pending permission,
focuses the iTerm2 tab, brings both iTerm2 and Emacs to foreground
\(works across multiple monitors), bounces Emacs dock icon, and sends
macOS notification."
  ;; Register pending permission
  (puthash sdd-session-id tool-name claude-agent--pending-permissions)

  ;; 1. Focus the correct iTerm2 tab (select the tab, no app activation yet)
  (claude-org-iterm2--focus-by-sdd-session sdd-session-id)

  ;; 2. Activate iTerm2 app (brings its window to foreground on its monitor)
  (claude-org-iterm2--activate-app "iTerm2")

  ;; 3. Activate Emacs (brings its window to foreground on its monitor)
  (claude-org-iterm2--activate-emacs)

  ;; 4. macOS notification
  (claude-org--notify
   "Claude: Permission Needed"
   (format "%s needs approval for %s"
           (claude-agent--permission-label sdd-session-id)
           tool-name)
   claude-org-notify-sound)

  ;; 5. Force immediate mode-line update
  (claude-agent--update-activity-string)
  "ok")

(defun claude-org-iterm2--permission-resolved (sdd-session-id)
  "Clear pending permission for SDD-SESSION-ID.
Called by the PostToolUse hook via MCP, or by mode-line click. Idempotent."
  (remhash sdd-session-id claude-agent--pending-permissions)
  (claude-agent--update-activity-string)
  "ok")

(defun claude-org-iterm2--activate-app (app-name)
  "Bring APP-NAME to the foreground on macOS via AppleScript."
  (when (eq system-type 'darwin)
    (start-process (format "activate-%s" (downcase app-name)) nil
                   "osascript" "-e"
                   (format "tell application \"%s\" to activate" app-name))))

(defun claude-org-iterm2--activate-emacs ()
  "Bring Emacs to the foreground on macOS and bounce dock icon."
  (when (eq system-type 'darwin)
    (claude-org-iterm2--activate-app "Emacs"))
  (raise-frame)
  ;; Bounce dock icon if Emacs is not focused (NSInformationalRequest)
  (when (and (display-graphic-p)
             (not (frame-focus-state (selected-frame))))
    (x-focus-frame nil)))

(defun claude-org-iterm2--focus-by-sdd-session (sdd-session-id)
  "Focus the iTerm2 tab associated with SDD-SESSION-ID."
  (let ((iterm2-uuid (claude-org-iterm2--lookup-iterm2-session sdd-session-id)))
    (when iterm2-uuid
      (claude-org-iterm2--call "focus" "--session" iterm2-uuid))))

(defvar claude-org-iterm2--sdd-to-iterm2 (make-hash-table :test 'equal)
  "Map of SDD session-id → iTerm2 UUID for quick lookup from MCP calls.")

(defvar claude-org-iterm2--sdd-to-session-key (make-hash-table :test 'equal)
  "Map of SDD session-id → org session-key for loop dispatch from query-completed.")

(defun claude-org-iterm2--lookup-iterm2-session (sdd-session-id)
  "Look up the iTerm2 UUID for SDD-SESSION-ID.
Checks the fast in-memory map first, then falls back to scanning
org buffers for the ITERM2_SESSION_ID property."
  (or (gethash sdd-session-id claude-org-iterm2--sdd-to-iterm2)
      ;; Fallback: scan org buffers for property match
      (let ((uuid (claude-org-iterm2--find-iterm2-uuid-in-buffers sdd-session-id)))
        (when uuid
          ;; Cache for next time
          (puthash sdd-session-id uuid claude-org-iterm2--sdd-to-iterm2))
        uuid)))

(defun claude-org-iterm2--find-iterm2-uuid-in-buffers (sdd-session-id)
  "Search org buffers for ITERM2_SESSION_ID on the heading with SDD-SESSION-ID."
  (let ((session-pattern (concat ":CLAUDE_SESSION_ID: "
                                 (regexp-quote sdd-session-id)))
        (result nil))
    (cl-dolist (buf (buffer-list))
      (when (and (buffer-file-name buf)
                 (string-suffix-p ".org" (buffer-file-name buf)))
        (with-current-buffer buf
          (save-excursion
            (save-restriction
              (widen)
              (goto-char (point-min))
              (when (re-search-forward session-pattern nil t)
                (org-back-to-heading t)
                (let ((uuid (org-entry-get nil "ITERM2_SESSION_ID")))
                  (when (and uuid (not (string-prefix-p "Warning" uuid)))
                    (setq result uuid)
                    (cl-return result)))))))))
    result))
