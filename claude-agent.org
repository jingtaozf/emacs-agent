* Introduction

This is an Emacs client library for the [[https://claude.ai/code][Claude Code CLI]]. It provides a
literate programming implementation of the Claude Agent SDK, allowing
Emacs users to interact with Claude programmatically.

The library mirrors the Python Claude Agent SDK API, providing:
- ~claude-agent-query~ for one-shot queries
- ~claude-agent-client~ for bidirectional interactive conversations
- Streaming response support via callbacks
- Basic hook support (PreToolUse, PostToolUse, Stop)
- Session management (continue, resume)
- Activity mode-line indicator with spinning animation
- Usage mode-line showing API utilization percentages

** Usage Example

#+BEGIN_SRC elisp :load no
;; One-shot query
(claude-agent-query
 "What is 2+2?"
 :on-message (lambda (msg)
               (when (claude-agent-assistant-message-p msg)
                 (message "Response: %s"
                          (claude-agent-extract-text msg))))
 :on-complete (lambda (_) (message "Done!")))

;; Interactive client
(let ((client (claude-agent-client-create
               :options (claude-agent-options
                         :model "claude-sonnet-4-5"
                         :permission-mode 'accept-edits))))
  (claude-agent-client-connect client)
  (claude-agent-client-send
   client "Hello, Claude!"
   :on-response (lambda (msg) (message "Got: %s" msg))))
#+END_SRC

* Activity Mode-Line

This section provides a mode-line indicator showing when Claude queries are active.
Displays a spinning indicator when queries are in progress. Auto-enabled on load.

** State Variables

#+BEGIN_SRC elisp
(defvar claude-agent-activity-string ""
  "Current activity string for mode-line display.
Shows spinner when queries are active, empty when idle.")

(defvar claude-agent--activity-timer nil
  "Timer for updating the activity spinner.")

(defconst claude-agent--spinner-frames '("◐" "◓" "◑" "◒")
  "Spinner animation frames for activity indicator.")

(defvar claude-agent--spinner-index 0
  "Current index in spinner-frames for animation.")

(defvar claude-agent-pending-alerts nil
  "Alist of (ID . PLIST) for sessions needing attention.
Each PLIST has :label, :tooltip, :face, :action (a function called on click).
Populated by `claude-agent-add-alert', cleared by `claude-agent-remove-alert'.")

(defun claude-agent-add-alert (id &rest props)
  "Add an alert with ID and PROPS plist (:label :tooltip :face :action).
If ID already exists, it is replaced."
  (setq claude-agent-pending-alerts
        (cons (cons id props)
              (assq-delete-all id claude-agent-pending-alerts)))
  (claude-agent--update-activity-string))

(defun claude-agent-remove-alert (id)
  "Remove the alert with ID."
  (setq claude-agent-pending-alerts
        (assq-delete-all id claude-agent-pending-alerts))
  (claude-agent--update-activity-string))
#+END_SRC

** Activity Count

#+BEGIN_SRC elisp
(defun claude-agent-active-query-count ()
  "Return count of currently active queries.
Uses `claude-agent--query-active-p' to check both closed state and process liveness."
  (let ((count 0))
    (maphash (lambda (_id state)
               (when (claude-agent--query-active-p state)
                 (cl-incf count)))
             claude-agent--active-queries)
    count))

(defun claude-agent-active-p ()
  "Return non-nil if any queries are currently active."
  (> (claude-agent-active-query-count) 0))
#+END_SRC

** Activity Display

#+BEGIN_SRC elisp
(defun claude-agent--format-elapsed (start-time)
  "Format elapsed time since START-TIME as human-readable string.
Uses compact format for mode-line display."
  (claude-agent-format-elapsed-time start-time 'compact))

(defun claude-agent--format-query-identity (state &optional long-form)
  "Format human-readable identity for query STATE.
If LONG-FORM is non-nil, use full buffer name instead of abbreviated.
Returns a string like 'dev.org#5' or 'dev.org#5(1/3)'."
  (let ((buf (claude-agent--process-state-source-buffer state))
        (ctx (claude-agent--process-state-query-context state))
        (req-id (claude-agent--process-state-request-id state)))
    (if (and buf (buffer-live-p buf))
        (let* ((buf-name (buffer-name buf))
               (short-name (if long-form
                               buf-name
                             (file-name-base buf-name)))
               (id-suffix (claude-agent-query-context-format-id ctx)))
          (if id-suffix
              (format "%s#%s" short-name id-suffix)
            short-name))
      ;; Fallback to request-id based format
      (if (and req-id (string-match "req-\\([0-9]+\\)" req-id))
          (format "#%s" (match-string 1 req-id))
        (if req-id
            (substring req-id 0 (min 12 (length req-id)))
          "#?")))))

(defun claude-agent--get-single-active-state ()
  "Return the single active query state, or nil if count != 1."
  (let ((result nil)
        (count 0))
    (maphash
     (lambda (_id state)
       (when (and state
                  (claude-agent--process-state-p state)
                  (not (claude-agent--process-state-closed state)))
         (setq result state)
         (cl-incf count)))
     claude-agent--active-queries)
    (when (= count 1) result)))

(defun claude-agent--any-docker-mode-p ()
  "Return non-nil if any active query is running in Docker mode."
  (let ((found nil))
    (maphash (lambda (_id state)
               (when (and state
                          (claude-agent--process-state-p state)
                          (not (claude-agent--process-state-closed state))
                          (claude-agent--process-state-docker-mode state))
                 (setq found t)))
             claude-agent--active-queries)
    found))

(defun claude-agent--format-activity-tooltip ()
  "Format tooltip text for the activity mode-line indicator.
Shows details about all active queries."
  (let ((lines '("Active Claude Queries (click to list)"))
        (count 0))
    (maphash
     (lambda (_id state)
       (when (and state
                  (claude-agent--process-state-p state)
                  (not (claude-agent--process-state-closed state)))
         (cl-incf count)
         (let* ((start-time (claude-agent--process-state-start-time state))
                (elapsed (claude-agent--format-elapsed start-time))
                (identity (claude-agent--format-query-identity state t))
                (docker-p (claude-agent--process-state-docker-mode state)))
           (push (format "  • %s%s [%s]"
                         identity
                         (if docker-p " [Docker]" "")
                         elapsed)
                 lines))))
     claude-agent--active-queries)
    (when (= count 0)
      (push "  (no active queries)" lines))
    (mapconcat #'identity (nreverse lines) "\n")))

(defun claude-agent--activity-alert-string ()
  "Build mode-line string for pending alerts.
Each alert entry becomes a clickable, propertized label."
  (let ((segments '()))
    (dolist (entry claude-agent-pending-alerts)
      (let* ((id (car entry))
             (props (cdr entry))
             (label (or (plist-get props :label)
                        (claude-agent--permission-label (format "%s" id))))
             (tooltip (or (plist-get props :tooltip) "Alert"))
             (face (or (plist-get props :face)
                       '(:foreground "orange" :weight bold)))
             (action (plist-get props :action)))
        (push (propertize
               label
               'face face
               'help-echo tooltip
               'mouse-face 'mode-line-highlight
               'local-map (when action
                            (let ((map (make-sparse-keymap)))
                              (define-key map [mode-line mouse-1]
                                (lexical-let ((fn action))
                                  (lambda (e)
                                    (interactive "e")
                                    (funcall fn))))
                              map)))
              segments)))
    (concat " [C:⚠ " (string-join (nreverse segments) " ") "]")))

(defun claude-agent--activity-queries-string (count)
  "Build mode-line string for COUNT active queries.
Shows a spinner with query identity (single) or count (multiple)."
  (let* ((spinner (nth claude-agent--spinner-index claude-agent--spinner-frames))
         (docker-p (claude-agent--any-docker-mode-p))
         (docker-prefix (if docker-p "D:" "C:"))
         (display (if (= count 1)
                      (let* ((state (claude-agent--get-single-active-state))
                             (identity (when state
                                         (claude-agent--format-query-identity state))))
                        (if identity
                            (format " [%s%s %s]" docker-prefix spinner identity)
                          (format " [%s%s]" docker-prefix spinner)))
                    (format " [%s%s×%d]" docker-prefix spinner count))))
    (setq claude-agent--spinner-index
          (mod (1+ claude-agent--spinner-index)
               (length claude-agent--spinner-frames)))
    (propertize display
                'face '(:weight bold)
                'help-echo (claude-agent--format-activity-tooltip)
                'mouse-face 'mode-line-highlight
                'local-map (let ((map (make-sparse-keymap)))
                             (define-key map [mode-line mouse-1]
                                         #'claude-agent-list-queries)
                             map))))

(defun claude-agent--update-activity-string ()
  "Update the activity mode-line string based on active queries and pending alerts."
  (let ((count (claude-agent-active-query-count))
        (alert-count (length claude-agent-pending-alerts)))
    (cond
     ((> alert-count 0)
      (setq claude-agent-activity-string (claude-agent--activity-alert-string))
      (setq claude-agent--spinner-index 0))
     ((> count 0)
      (setq claude-agent-activity-string (claude-agent--activity-queries-string count)))
     (t
      (setq claude-agent-activity-string "")
      (setq claude-agent--spinner-index 0))))
  (force-mode-line-update t)
  (claude-agent--maybe-refresh-queries-buffer))

(defun claude-agent--permission-label (workspace-session-id)
  "Return a short label for WORKSPACE-SESSION-ID for mode-line display.
Extracts the timestamp portion from session IDs like sdd-20260303-132101."
  (let ((parts (split-string workspace-session-id "-")))
    ;; sdd-YYYYMMDD-HHMMSS → show HHMMSS
    (if (>= (length parts) 3)
        (car (last parts))
      workspace-session-id)))

(defun claude-agent--maybe-refresh-queries-buffer ()
  "Refresh *Claude Queries* buffer if it exists and is visible."
  (when-let* ((buf (get-buffer "*Claude Queries*")))
    (when (get-buffer-window buf t)  ; visible in any frame
      (with-current-buffer buf
        (let ((inhibit-read-only t)
              (pos (point)))
          (erase-buffer)
          (insert "Active Queries\n")
          (insert "══════════════════════════════════════════════════\n\n")
          (let ((has-queries nil))
            (maphash
             (lambda (id state)
               (when (claude-agent--query-active-p state)
                 (setq has-queries t)
                 (let* ((start-time (claude-agent--process-state-start-time state))
                        (elapsed (claude-agent--format-elapsed start-time))
                        (identity (claude-agent--format-query-identity state t))
                        (ctx (claude-agent--process-state-query-context state))
                        (docker-p (claude-agent--process-state-docker-mode state))
                        (loc-fn (claude-agent--process-state-source-location-fn state))
                        (navigable (not (null loc-fn)))
                        (src-buf (claude-agent--process-state-source-buffer state))
                        (session-key (when (and src-buf (buffer-live-p src-buf))
                                       (with-current-buffer src-buf
                                         (condition-case nil
                                             (code-agent-org--current-session-key)
                                           (error nil))))))
                   (insert (propertize (format "● %s" identity)
                                       'face (if navigable 'link 'font-lock-function-name-face)
                                       'query-id id
                                       'session-key session-key
                                       'navigable navigable))
                   (when docker-p
                     (insert (propertize " [Docker]" 'face '(:weight bold))))
                   (when-let* ((label (claude-agent-query-context-format-label ctx)))
                     (insert (propertize (format " %s" label)
                                         'face 'font-lock-keyword-face)))
                   (insert (format "  [%s]\n" elapsed)))))
             claude-agent--active-queries)
            (unless has-queries
              (insert "No active queries.\n")))
          (insert "\n")
          (insert "Keys: RET=goto source  v=verbose  k=cancel  K=cancel all  g=refresh  q=quit\n")
          (goto-char (min pos (point-max))))))))

** Session Manager

The =*Claude Queries*= buffer (opened by =claude-agent-list-queries=,
also =M-x claude-agent-list-queries= or =,m= from the agent menu) is
the middle layer of a three-layer monitoring model. The model was
inspired by Stern (Kubernetes log streamer), prodigy.el, tmux's
=monitor-activity=, and =docker-compose logs=:

| Layer | Surface | Role |
|-------+---------+------|
| 1 | Mode-line =[C:◐×3]= | Always-visible activity indicator (count + spinner) |
| 2 | =*Claude Queries*= dashboard (this section) | One row per active query — session-key, elapsed, status, =[+N lines]= unread badge |
| 3 | =*cmux: <story>*= verbose buffer (per session) | Full live terminal mirror; reached via =RET= or =v= on a dashboard row |

The dashboard's per-row =[+N lines]= badge tracks unread output by
comparing the verbose buffer's current line count against a
=last-viewed-line-count= stored when the user last visited that
buffer. =RET= opens the verbose buffer in the same window; =v= opens
in a side window. =k= cancels a single query, =K= cancels all, =g=
refreshes.

A *unified all-output view* (one scrolling buffer aggregating output
from every active session, like =docker-compose logs=) is intended
future work. The shape would be: dispatcher in the verbose tick that
also inserts into a =*Claude All Output*= buffer with a colour-coded
session-name prefix (using =SDD_COLOR=); same scroll-preservation /
follow-mode (=G=) / transient menu (=,= sends keys to the session
under cursor) as the per-session buffer. Not built today because the
per-session buffer + dashboard navigation has been sufficient.

(defvar claude-agent-queries-mode-map
  (make-sparse-keymap)
  "Keymap for `claude-agent-queries-mode'.")

;; Define keys outside defvar so they are applied on reload
;; (defvar does not re-evaluate when the variable already has a value).
(define-key claude-agent-queries-mode-map (kbd "RET") #'claude-agent-queries-goto-source)
(define-key claude-agent-queries-mode-map (kbd "v") #'claude-agent-queries-show-verbose)
(define-key claude-agent-queries-mode-map (kbd "k") #'claude-agent-queries-cancel-at-point)
(define-key claude-agent-queries-mode-map (kbd "K") #'claude-agent-cancel-all-queries)
(define-key claude-agent-queries-mode-map (kbd "g") #'claude-agent-list-queries)
(define-key claude-agent-queries-mode-map (kbd "q") #'quit-window)

(define-derived-mode claude-agent-queries-mode special-mode "Claude-Queries"
  "Major mode for viewing and managing Claude Agent queries."
  (setq buffer-read-only t)
  (setq truncate-lines t))

(defun claude-agent-list-queries ()
  "Display buffer showing all active queries."
  (interactive)
  (let ((buf (get-buffer-create "*Claude Queries*")))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert "Active Queries\n")
        (insert "══════════════════════════════════════════════════\n\n")
        (let ((has-queries nil)
              (docker-count 0)
              (local-count 0))
          (maphash
           (lambda (id state)
             (when (claude-agent--query-active-p state)
               (setq has-queries t)
               (let* ((start-time (claude-agent--process-state-start-time state))
                      (elapsed (claude-agent--format-elapsed start-time))
                      ;; Use identity with long-form buffer name
                      (identity (claude-agent--format-query-identity state t))
                      (ctx (claude-agent--process-state-query-context state))
                      (docker-p (claude-agent--process-state-docker-mode state))
                      ;; Check if navigable (has source-location-fn)
                      (loc-fn (claude-agent--process-state-source-location-fn state))
                      (navigable (not (null loc-fn))))
                 ;; Track Docker vs local counts
                 (if docker-p (cl-incf docker-count) (cl-incf local-count))
                 ;; Check verbose buffer for activity badge
                 (let* ((src-buf (claude-agent--process-state-source-buffer state))
                        (session-key (when (and src-buf (buffer-live-p src-buf))
                                      (with-current-buffer src-buf
                                        (condition-case nil
                                            (code-agent-org--current-session-key)
                                          (error nil)))))
                        (verbose-buf (when session-key
                                       (gethash session-key claude-agent--session-verbose-buffers)))
                        (verbose-lines (when (and verbose-buf (buffer-live-p verbose-buf))
                                         (with-current-buffer verbose-buf
                                           (count-lines (point-min) (point-max)))))
                        (has-verbose (and verbose-buf (buffer-live-p verbose-buf))))
                   ;; Format: ▶ ● buffer-name Instruction N  [elapsed]  [+N lines]
                   (if has-verbose
                       (insert (propertize "▶ " 'face 'font-lock-warning-face))
                     (insert "  "))
                   (insert (propertize (format "● %s" identity)
                                       'face (if navigable 'link 'font-lock-function-name-face)
                                       'query-id id
                                       'session-key session-key
                                       'navigable navigable))
                   (when docker-p
                     (insert (propertize " [Docker]"
                                         'face '(:weight bold))))
                   (when-let* ((label (claude-agent-query-context-format-label ctx)))
                     (insert (propertize (format " %s" label)
                                         'face 'font-lock-keyword-face)))
                   (insert (format "  [%s]" elapsed))
                   (when verbose-lines
                     (insert (propertize (format "  [%d lines]" verbose-lines)
                                         'face 'font-lock-comment-face)))
                   (insert "\n")))))
           claude-agent--active-queries)
          (unless has-queries
            (insert "No active queries.\n"))
          ;; Show summary if there are Docker queries
          (when (> docker-count 0)
            (insert "\n")
            (insert (propertize (format "Docker: %d" docker-count)
                                'face '(:weight bold)))
            (when (> local-count 0)
              (insert (format "  Local: %d" local-count)))
            (insert "\n")))
        (insert "\n")
        (insert "Keys: RET=goto source  v=verbose output  k=cancel  K=cancel all  g=refresh  q=quit\n"))
      (claude-agent-queries-mode)
      (goto-char (point-min)))
    (display-buffer buf)))

(defun claude-agent-queries-cancel-at-point ()
  "Cancel the query on the current line."
  (interactive)
  (let ((query-id (get-text-property (point) 'query-id)))
    (if query-id
        (progn
          (claude-agent-cancel-query query-id)
          (message "Cancelled query: %s" query-id)
          (claude-agent-list-queries))
      (message "No query on this line."))))

(defun claude-agent-queries-goto-source ()
  "Jump to the source location for query at point.
For navigable queries (shown as links), jumps to the source heading.
For non-navigable queries, jumps to the source buffer only."
  (interactive)
  (let* ((query-id (get-text-property (point) 'query-id))
         (navigable (get-text-property (point) 'navigable)))
    (if (null query-id)
        (message "No query on this line")
      (let ((state (gethash query-id claude-agent--active-queries)))
        (if (null state)
            (message "Query not found (may have completed)")
          (let ((buf (claude-agent--process-state-source-buffer state))
                (loc-fn (claude-agent--process-state-source-location-fn state))
                (loc-args (claude-agent--process-state-source-location-args state)))
            (cond
             ((null buf)
              (message "No source buffer for this query"))
             ((not (buffer-live-p buf))
              (message "Source buffer no longer exists"))
             ((null loc-fn)
              ;; No location function - just jump to buffer
              (pop-to-buffer buf)
              (message "Jumped to buffer (no specific location)"))
             (t
              ;; Has location function - switch buffer and call it
              (pop-to-buffer buf)
              (if (apply loc-fn loc-args)
                  (message "Jumped to source")
                (message "Location function failed to find target"))))))))))

#+BEGIN_SRC elisp
(defun claude-agent-queries-show-verbose ()
  "Show verbose output buffer for the query at point."
  (interactive)
  (let ((session-key (get-text-property (point) 'session-key)))
    (if (null session-key)
        (message "No session on this line")
      (let ((verbose-buf (gethash session-key claude-agent--session-verbose-buffers)))
        (if (and verbose-buf (buffer-live-p verbose-buf))
            (display-buffer verbose-buf

* Rate-Limit Mode-Line
:PROPERTIES:
:CUSTOM_ID: usage-mode-line
:END:

Shows Claude Code's 5-hour and 7-day API rate-limit utilisation in the
Emacs mode-line.  Source of truth: =~/.claude/logs/rate-limits.json=,
written by =~/.claude/statusline-command.sh= every time Claude Code
calls the statusline (a few seconds while a session is up).  The
statusline receives the =rate_limits= block on stdin from Claude Code
itself — a sanctioned channel — so this module never scrapes
claude.ai endpoints or pokes at internal session state.  When no
Claude Code session is running the file simply stops updating; the
mode-line keeps the last known values until you explicitly clear them.

A previous incarnation of this feature (commit f8890f1, Dec 2025)
fetched limits directly from the claude.ai HTTP API using
=CLAUDE_SESSION_KEY=.  That path was undocumented and was removed.
The replacement here re-uses an officially-supplied data flow.

(defun claude-agent-rate-limit--face-for (pct)
  "Return the face symbol that matches usage percent PCT."
  (cond
   ((>= pct claude-agent-rate-limit-crit-threshold)
    'claude-agent-rate-limit-crit-face)
   ((>= pct claude-agent-rate-limit-warn-threshold)
    'claude-agent-rate-limit-warn-face)
   (t 'claude-agent-rate-limit-ok-face)))

(defun claude-agent-rate-limit--format-reset (secs)
  "Format SECS as `Xh YYm', or empty string for nil/non-positive."
  (cond
   ((or (null secs) (not (numberp secs)) (<= secs 0)) "")
   (t (let ((h (/ secs 3600))
            (m (/ (mod secs 3600) 60)))
        (format " %dh%02dm" h m)))))

(defun claude-agent-rate-limit--read-file ()
  "Re-read `claude-agent-rate-limit-file' if mtime advanced.
Updates `claude-agent-rate-limit--data' as a side effect.  Returns t
when data changed, nil otherwise."
  (when (file-readable-p claude-agent-rate-limit-file)
    (let ((mt (float-time
               (file-attribute-modification-time
                (file-attributes claude-agent-rate-limit-file)))))
      (when (> mt claude-agent-rate-limit--last-mtime)
        (setq claude-agent-rate-limit--last-mtime mt)
        (condition-case _
            (with-temp-buffer
              (insert-file-contents claude-agent-rate-limit-file)
              (goto-char (point-min))
              (setq claude-agent-rate-limit--data
                    (json-parse-buffer :object-type 'alist :null-object nil))
              t)
          (error nil))))))

** Render

Mode-line format mirrors the original 2025-12 implementation
(commit f8890f1, before the API-scraping path was removed):

#+begin_example
[C:XX|YY|ZZ]
#+end_example

- =XX= = 5-hour bucket utilisation
- =YY= = 7-day bucket utilisation
- =ZZ= = 7-day Sonnet bucket utilisation (omitted if not provided in
  the JSON; format degrades to =[C:XX|YY]=)

Whole bracket is coloured by the *worst* of the three buckets so a
single glance tells you whether you are safe (green), getting close
(yellow), or about to hit a wall (red).  Tooltip lists each bucket
with its own percentage and reset time, matching the original layout.

Click on the indicator to force a refresh.

#+BEGIN_SRC elisp
(defun claude-agent-rate-limit--pct (bucket)
  "Return integer percentage for BUCKET alist, or nil if absent."
  (let ((p (and bucket (alist-get 'used_percentage bucket))))
    (and (numberp p) (truncate p))))

(defun claude-agent-rate-limit--resets-in-secs (bucket)
  "Return seconds until reset for BUCKET, or nil."
  (let ((at (and bucket (alist-get 'resets_at bucket))))
    (when (numberp at)
      (max 0 (truncate (- at (float-time)))))))

(defun claude-agent-rate-limit--data-stale-p ()
  "Return non-nil if the last update is older than the stale threshold."
  (when claude-agent-rate-limit--data
    (let ((updated (alist-get 'updated_at claude-agent-rate-limit--data)))
      (and updated
           (> (- (float-time) updated)
              claude-agent-rate-limit-stale-after)))))

(defun claude-agent-rate-limit--mouse-map ()
  "Return a keymap that triggers a refresh on mouse-1."
  (let ((map (make-sparse-keymap)))
    (define-key map [mode-line mouse-1] #'claude-agent-rate-limit-refresh)
    map))

(defun claude-agent-rate-limit--tooltip ()
  "Build a multi-line tooltip in the original 2025-12 layout.
Order: header, divider, per-bucket rows, blank, Updated, blank,
\"Click to refresh\"."
  (when claude-agent-rate-limit--data
    (let* ((updated-iso (alist-get 'updated_iso claude-agent-rate-limit--data))
           (five   (alist-get 'five_hour        claude-agent-rate-limit--data))
           (seven  (alist-get 'seven_day        claude-agent-rate-limit--data))
           (sonnet (alist-get 'seven_day_sonnet claude-agent-rate-limit--data))
           (rows nil))
      (dolist (row (list (list "5-hour:      " five)
                         (list "7-day:       " seven)
                         (list "7-day Sonnet:" sonnet)))
        (let* ((label (car row))
               (b (cadr row))
               (pct (claude-agent-rate-limit--pct b))
               (secs (claude-agent-rate-limit--resets-in-secs b)))
          (when b
            (push (format "%s %3d%% (resets in %s)"
                          label
                          (or pct 0)
                          (string-trim (or (claude-agent-rate-limit--format-reset secs) "")))
                  rows))))
      (mapconcat #'identity
                 (append (list "Claude Usage"
                               "─────────────────────────")
                         (nreverse rows)
                         (list ""
                               (format "Updated: %s" (or updated-iso "—"))
                               ""
                               "Click to refresh"))
                 "\n"))))

(defun claude-agent-rate-limit--update-string ()
  "Rebuild `claude-agent-rate-limit-string' from cached data.
Output format: ` [C:XX|YY|ZZ]` where XX/YY/ZZ are integer percentages
for 5-hour, 7-day, and (when present) 7-day Sonnet buckets.  The
whole bracket is coloured by the worst bucket."
  (claude-agent-rate-limit--read-file)
  (setq claude-agent-rate-limit-string
        (cond
         ;; No file yet — show neutral placeholder so users know the
         ;; module is wired but no statusline payload has arrived.
         ((null claude-agent-rate-limit--data)
          (propertize " [C:--]"
                      'face 'claude-agent-rate-limit-stale-face
                      'help-echo
                      (format "No rate-limit data at %s yet.\nStart Claude Code to populate."
                              claude-agent-rate-limit-file)
                      'mouse-face 'mode-line-highlight
                      'local-map (claude-agent-rate-limit--mouse-map)))
         (t
          (let* ((five  (alist-get 'five_hour claude-agent-rate-limit--data))
                 (seven (alist-get 'seven_day claude-agent-rate-limit--data))
                 (sonnet (alist-get 'seven_day_sonnet claude-agent-rate-limit--data))
                 (five-pct  (claude-agent-rate-limit--pct five))
                 (seven-pct (claude-agent-rate-limit--pct seven))
                 (sonnet-pct (claude-agent-rate-limit--pct sonnet))
                 (worst (apply #'max
                               (delq nil
                                     (list five-pct seven-pct sonnet-pct
                                           ;; Default to 0 if every bucket
                                           ;; missing so `max' has an arg.
                                           0))))
                 (face (if (claude-agent-rate-limit--data-stale-p)
                           'claude-agent-rate-limit-stale-face
                         (claude-agent-rate-limit--face-for worst)))
                 (parts (delq nil
                              (list (and five-pct  (format "%d" five-pct))
                                    (and seven-pct (format "%d" seven-pct))
                                    (and sonnet-pct (format "%d" sonnet-pct)))))
                 (display (if parts
                              (format " [C:%s]" (mapconcat #'identity parts "|"))
                            " [C:--]")))
            (propertize display
                        'face face
                        'help-echo (claude-agent-rate-limit--tooltip)
                        'mouse-face 'mode-line-highlight
                        'local-map (claude-agent-rate-limit--mouse-map)))))))
#+END_SRC

(defun claude-agent-chat--handle-message (msg)
  "Handle incoming MSG from Claude in current buffer."
  (let ((type (claude-agent-message-type msg)))
    (pcase type
      ('system
       (when claude-agent-chat-show-system-messages
         (let ((subtype (claude-agent-system-message-subtype msg)))
           (claude-agent-chat--insert
            (format "\n[System: %s]\n" subtype)
            'claude-agent-chat-info-face))))

      ('assistant
       (let ((text (claude-agent-extract-text msg)))
         (when text
           (setq claude-agent-chat--current-response text)
           (claude-agent-chat--update-response text)))
       ;; Check for tool use
       (let ((content (claude-agent-assistant-message-content msg)))
         (dolist (block content)
           (when (claude-agent-tool-use-block-p block)
             (claude-agent-chat--insert
              (format "\n[Tool: %s]\n"
                      (claude-agent-tool-use-block-name block))
              'claude-agent-chat-tool-face)))))

      ('result
       (setq claude-agent-chat--session-id
             (claude-agent-result-message-session-id msg))
       (let ((cost (claude-agent-result-message-total-cost-usd msg)))
         (when cost
           (setq claude-agent-chat--total-cost
                 (+ claude-agent-chat--total-cost cost))
           (when claude-agent-chat-show-cost
             (claude-agent-chat--insert
              (format "\n[Cost: $%.4f | Session total: $%.4f]\n"
                      cost claude-agent-chat--total-cost)
              'claude-agent-chat-info-face))))))))

(defun claude-agent-chat--send-input (input)
  "Send INPUT to Claude via the backend."
  (when (and input (not (string-empty-p (string-trim input))))
    (if claude-agent-chat--waiting
        (message "Still waiting for Claude's response...")
      (setq claude-agent-chat--waiting t)
      (setq claude-agent-chat--current-response "")
      (claude-agent-chat--insert-response-start)
      (claude-agent-chat--insert "[Waiting...]" 'claude-agent-chat-thinking-face)
      ;; Ensure backend exists
      (unless claude-agent-chat--backend
        (setq claude-agent-chat--backend (claude-agent-chat--make-backend)))
      ;; Build options with session continuation if we have a session
      (let* ((base-opts (or claude-agent-chat--options (claude-agent-options)))
             (opts (if claude-agent-chat--session-id
                       (plist-put (plist-put (copy-sequence base-opts)
                                             :continue-conversation t)
                                  :resume claude-agent-chat--session-id)
                     base-opts))
             (opts (if claude-agent-chat--system-prompt
                       (plist-put (copy-sequence opts)
                                  :system-prompt claude-agent-chat--system-prompt)
                     opts))
             (opts (if claude-agent-chat--mcp-config
                       (plist-put (copy-sequence opts)
                                  :mcp-config claude-agent-chat--mcp-config)
                     opts))
             (system-reminder (when claude-agent-chat-include-ide-context
                                (claude-agent-get-system-reminder)))
             (actual-prompt (if (and system-reminder
                                     (not (string-empty-p system-reminder)))
                                (concat system-reminder "\n" input)
                              input)))
        ;; Build callbacks plist for backend-query
        (lexical-let ((chat-buf (current-buffer)))
          (let ((callbacks
                 (list
                  :on-token (lambda (token)
                              (when (buffer-live-p chat-buf)
                                (with-current-buffer chat-buf
                                  (claude-agent-chat--handle-token token))))
                  :on-message (lambda (msg)
                                (when (buffer-live-p chat-buf)
                                  (with-current-buffer chat-buf
                                    (claude-agent-chat--handle-message msg))))
                  :on-error (lambda (err)
                              (when (buffer-live-p chat-buf)
                                (with-current-buffer chat-buf
                                  (claude-agent-chat--handle-error err))))
                  :on-complete (lambda (result)
                                 (when (buffer-live-p chat-buf)
                                   (with-current-buffer chat-buf
                                     (claude-agent-chat--handle-complete result)))))))
            (setq claude-agent-chat--query-handle
                  (claude-agent-backend-query
                   claude-agent-chat--backend
                   actual-prompt
                   callbacks
                   :options opts))))))))

(defun claude-agent-chat-interrupt ()
  "Interrupt the current Claude request via the backend."
  (interactive)
  (when claude-agent-chat--backend
    (claude-agent-backend-cancel claude-agent-chat--backend
                                 claude-agent-chat--query-handle))
  (setq claude-agent-chat--query-handle nil)
  (setq claude-agent-chat--waiting nil)
  (claude-agent-chat--insert "\n[Interrupted]\n" 'claude-agent-chat-error-face)
  (claude-agent-chat--show-prompt))

(defun claude-agent-chat-clear ()
  "Clear the chat buffer and reset the session."
  (interactive)
  (when (yes-or-no-p "Clear chat history and start new session? ")
    ;; Clean up backend
    (when claude-agent-chat--backend
      (claude-agent-backend-cleanup claude-agent-chat--backend)
      (setq claude-agent-chat--backend nil)
      (setq claude-agent-chat--query-handle nil))
    (let ((inhibit-read-only t))
      (erase-buffer)
      (setq claude-agent-chat--session-id nil)
      (setq claude-agent-chat--total-cost 0.0)
      (setq claude-agent-chat--waiting nil)
      (claude-agent-chat--initialize-buffer))))

(defun claude-agent-chat-new-session ()
  "Start a new conversation session without clearing history.
Resets session-id for a fresh conversation."
  (interactive)
  (setq claude-agent-chat--session-id nil)
  (claude-agent-chat--insert "\n--- New Session ---\n" 'claude-agent-chat-info-face)

#+BEGIN_SRC elisp
(defcustom claude-agent-translate-model "haiku"
  "Model to use for translation. Haiku is fastest and cheapest."
  :type 'string
  :group 'claude-agent)

(defvar claude-agent-translate--active-state nil
  "Currently active translation process state.")

(defvar claude-agent-translate--last-buffer nil
  "Most recently created translation output buffer, or nil.
Each call to `claude-agent-translate' updates this to point at the
fresh per-run buffer. Tests and jump-to-latest commands can rely on
this without having to scan buffer lists.")

(defun claude-agent-translate--make-buffer-name (language)
  "Build a unique buffer name for a translation into LANGUAGE.
Uses `claude-agent-translate-buffer-name' as a template, injecting
LANGUAGE before the trailing `*'. `generate-new-buffer-name' then
handles collisions by appending `<2>', `<3>', etc."
  (let* ((template claude-agent-translate-buffer-name)
         (trimmed (if (and (string-prefix-p "*" template)
                           (string-suffix-p "*" template)
                           (> (length template) 2))
                      (substring template 1 -1)
                    template))
         (base (format "*%s: %s*" trimmed language)))
    (generate-new-buffer-name base)))

(defun claude-agent-translate--setup-buffer (language text-preview)
  "Create and return a fresh translation buffer for LANGUAGE.
TEXT-PREVIEW is shown as a source snippet in the header. A unique
buffer name is allocated per call so previous translations remain
intact; the new buffer is also stashed in
`claude-agent-translate--last-buffer'."
  (let ((buf (get-buffer-create
              (claude-agent-translate--make-buffer-name language))))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (format "Translating to %s...\n" language))
        (insert "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n")
        (insert (format "Source: %s...\n\n"
                        (truncate-string-to-width text-preview 60 nil nil "...")))
        (insert "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n")
        (goto-char (point-max))))
    (setq claude-agent-translate--last-buffer buf)
    (display-buffer buf)
    buf))

(defun claude-agent-translate--insert-token (buf text)
  "Insert TEXT token into translation BUF."
  (when (buffer-live-p buf)
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (goto-char (point-max))
        (insert text)))))

(defun claude-agent-translate--complete (buf result)
  "Handle translation completion in BUF with optional RESULT error string.
RESULT is nil on success, or an error description on failure."
  (setq claude-agent-translate--active-state nil)
  (when (buffer-live-p buf)
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (goto-char (point-max))
        (insert "\n\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n")
        (if result
            (insert (format "[Error: %s]\n" result))
          (insert "[Translation complete]\n")))))
  (message "Translation %s" (if result "failed" "complete")))

*** Core Translation Function

#+BEGIN_SRC elisp
(defun claude-agent-translate (text language &optional extra-options)
  "Translate TEXT to LANGUAGE with streaming output.
EXTRA-OPTIONS is an optional plist to merge with default options.
Each call opens a fresh output buffer named after LANGUAGE so
repeated translations do not overwrite each other."
  (when claude-agent-translate--active-state
    (user-error "Translation already in progress"))
  (let* ((prompt (format "Translate the following text to %s. \
Preserve all formatting, structure, code blocks, and markup. \
Output only the translation without any explanation:\n\n%s"
                         language text))
         (buf (claude-agent-translate--setup-buffer language text))
         (base-options (claude-agent-options :model claude-agent-translate-model))
         (options (if extra-options
                      (apply #'claude-agent-options
                             :model claude-agent-translate-model
                             extra-options)
                    base-options)))
    ;; `lexical-let' captures BUF for the async callbacks — literate-elisp
    ;; evaluates this file with dynamic binding so a plain `let' would not
    ;; survive until the filter/sentinel fires.
    (setq claude-agent-translate--active-state
          (lexical-let ((out-buf buf))
            (claude-agent-query
             prompt
             :options options
             :on-token (lambda (tok)
                         (claude-agent-translate--insert-token out-buf tok))
             :on-error (lambda (err)
                         (claude-agent-translate--complete
                          out-buf (format "%s" err)))
             :on-complete (lambda (res)
                            (claude-agent-translate--complete out-buf res)))))))
