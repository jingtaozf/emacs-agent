* Introduction

This is an Emacs client library for the [[https://claude.ai/code][Claude Code CLI]]. It provides a
literate programming implementation of the Claude Agent SDK, allowing
Emacs users to interact with Claude programmatically.

The library mirrors the Python Claude Agent SDK API, providing:
- ~claude-agent-query~ for one-shot queries
- ~claude-agent-client~ for bidirectional interactive conversations
- Streaming response support via callbacks
- Basic hook support (PreToolUse, PostToolUse, Stop)
- Session management (continue, resume)
- Activity mode-line indicator with spinning animation
- Usage mode-line showing API utilization percentages

** Usage Example

#+BEGIN_SRC elisp :load no
;; One-shot query
(claude-agent-query
 "What is 2+2?"
 :on-message (lambda (msg)
               (when (claude-agent-assistant-message-p msg)
                 (message "Response: %s"
                          (claude-agent-extract-text msg))))
 :on-complete (lambda (_) (message "Done!")))

;; Interactive client
(let ((client (claude-agent-client-create
               :options (claude-agent-options
                         :model "claude-sonnet-4-5"
                         :permission-mode 'accept-edits))))
  (claude-agent-client-connect client)
  (claude-agent-client-send
   client "Hello, Claude!"
   :on-response (lambda (msg) (message "Got: %s" msg))))
#+END_SRC

(defun claude-agent-options (&rest args)
  "Create a Claude Agent options plist.

Supported keyword arguments:
  :system-prompt      - System prompt string
  :allowed-tools      - List of allowed tool names
  :disallowed-tools   - List of disallowed tool names
  :max-turns          - Maximum conversation turns (integer)
  :max-budget-usd     - Maximum budget in USD (number)
  :model              - Model name string
  :permission-mode    - One of: default, accept-edits, plan, bypass-permissions
  :continue-conversation - Boolean to continue previous conversation
  :resume             - Session ID string to resume
  :cwd                - Working directory for CLI
  :cli-path           - Path to Claude CLI binary
  :hooks              - Alist of (event . matchers)
  :env                - Alist of environment variables
  :setting-sources    - List of setting sources (\"user\" \"project\" \"local\")

Returns a plist suitable for passing to `claude-agent-query' or
`claude-agent-client-create'."
  ;; Use `list' instead of quoted literal to avoid destructive modification
  (let ((defaults (list :system-prompt nil
                        :allowed-tools nil
                        :disallowed-tools nil
                        :max-turns nil
                        :max-budget-usd nil
                        :model nil
                        :permission-mode nil
                        :continue-conversation nil
                        :resume nil
                        :cwd nil
                        :cli-path nil
                        :hooks nil
                        :env nil
                        :setting-sources nil)))
    (cl-loop for (key val) on args by #'cddr
             do (setq defaults (plist-put defaults key val)))
    defaults))

** Query Context

Query context holds rich metadata for display and tracking in Session Manager.

#+BEGIN_SRC elisp
(cl-defstruct (claude-agent-query-context
               (:constructor claude-agent-make-query-context)
               (:copier nil))
  "Rich context for query display and tracking.
Used by Session Manager to show query identity like 'dev.org#5(1/3)'."
  (instruction-num nil)   ; Integer instruction number (e.g., 5)
  (loop-current 1)        ; Current iteration (1-based)
  (loop-max 1)            ; Total iterations
  (custom-id nil)         ; CUSTOM_ID for navigation
  (tags nil))             ; List of tags like (:implementation :sdd)

(defun claude-agent-query-context-format-id (ctx)
  "Format context for identity display.
Returns instruction number with optional loop status:
  - '5' for single execution
  - '5(1/3)' for loop iteration"
  (when ctx
    (let ((num (claude-agent-query-context-instruction-num ctx)))
      (when num
        (let ((cur (claude-agent-query-context-loop-current ctx))
              (max (claude-agent-query-context-loop-max ctx)))
          (if (= max 1)
              (number-to-string num)
            (format "%d(%d/%d)" num cur max)))))))

(defun claude-agent-query-context-format-label (ctx)
  "Format context for Session Manager label display.
Returns human-readable label:
  - 'Instruction 5' for single execution
  - 'Instruction 5 (1/3)' for loop iteration"
  (when ctx
    (let ((num (claude-agent-query-context-instruction-num ctx)))
      (when num
        (let ((cur (claude-agent-query-context-loop-current ctx))
              (max (claude-agent-query-context-loop-max ctx)))
          (if (= max 1)
              (format "Instruction %d" num)
            (format "Instruction %d (%d/%d)" num cur max)))))))

** Event Dispatch Functions

#+BEGIN_SRC elisp
(defun claude-agent--dispatch-pre-tool-use (parsed state)
  "Dispatch PreToolUse hooks for tool_use blocks in PARSED message.
STATE is the process state struct."
  (when-let* ((message (plist-get parsed :message))
              (content (plist-get message :content)))
    ;; Content must be a list of blocks, not a plain string
    (when (and (listp content) (listp (car-safe content)))
      (dolist (block content)
        (when (equal (plist-get block :type) "tool_use")
          (run-hook-with-args 'claude-agent-pre-tool-use-functions
                              (plist-get block :name)
                              (plist-get block :input)
                              (plist-get block :id)
                              state))))))

(defun claude-agent--dispatch-post-tool-use (parsed state)
  "Dispatch PostToolUse hooks for tool_result blocks in PARSED message.
STATE is the process state struct."
  (when-let ((message (plist-get parsed :message)))
    (let ((content (plist-get message :content))
          (tool-use-result (plist-get parsed :tool_use_result)))
      ;; Content must be a list of blocks, not a plain string
      (when (and content (listp content) (listp (car-safe content)))
        (dolist (block content)
          (when (equal (plist-get block :type) "tool_result")
            (run-hook-with-args 'claude-agent-post-tool-use-functions
                                (plist-get block :tool_use_id)
                                (plist-get block :content)
                                (plist-get block :is_error)
                                tool-use-result
                                state)))))))

** CLI Discovery

#+BEGIN_SRC elisp
(defun claude-agent--find-cli ()
  "Find the Claude Code CLI binary.
Returns the path to the CLI or signals an error if not found."
  (or claude-agent-cli-path
      (executable-find "claude")
      (cl-find-if #'file-executable-p
                  (mapcar #'expand-file-name
                          claude-agent-cli-search-paths))
      (signal 'claude-agent-cli-not-found-error
              (list "Claude Code CLI not found. Install with: npm install -g @anthropic-ai/claude-code"))))

(defun claude-agent--cli-version ()
  "Get the Claude Code CLI version.
Returns version string or nil if unable to determine."
  (condition-case nil
      (let ((output (shell-command-to-string
                     (format "%s -v" (claude-agent--find-cli)))))
        (when (string-match "\\([0-9]+\\.[0-9]+\\.[0-9]+\\)" output)
          (match-string 1 output)))
    (error nil)))

** Docker Container Management

When =claude-agent-cli-path= or the =:cli-path= option is a Docker command
(e.g., "docker compose exec claude claude"), these functions handle automatic
container startup and management.

*** Docker Command Detection

#+BEGIN_SRC elisp
(defun claude-agent--docker-command-p (cmd)
  "Return non-nil if CMD is a Docker command.
Detects 'docker run', 'docker-compose', and 'docker compose' patterns."
  (and cmd
       (stringp cmd)
       (string-match-p
        (rx bos (* space)
            (or "docker run"
                "docker-compose"
                "docker compose"))
        cmd)))

(defun claude-agent--split-command (cmd)
  "Split CMD string into a list of arguments.
Handles quoted strings and basic shell syntax.
For non-Docker commands, returns a single-element list."
  (if (claude-agent--docker-command-p cmd)
      (split-string-and-unquote cmd)
    (list cmd)))

(defun claude-agent--extract-compose-file-from-command (cmd)
  "Extract compose file path from Docker command CMD.
Looks for -f or --file flag followed by the path.
Returns the compose file path or nil if not found."
  (when (and cmd (stringp cmd))
    (let ((args (split-string-and-unquote cmd)))
      (cl-loop for (arg next) on args
               when (and (member arg '("-f" "--file"))
                         next)
               return next))))
#+END_SRC

*** Docker Compose Directory Discovery

#+BEGIN_SRC elisp
(defun claude-agent--find-compose-dir (&optional cli-path project-root)
  "Find directory containing docker-compose.yml for Claude.

Optional CLI-PATH is the Docker command which may contain -f flag.
Optional PROJECT-ROOT is the project directory for resolving relative paths.

Search order:
1. Extract from CLI-PATH -f flag (absolute or relative to PROJECT-ROOT)
2. .devcontainer/ in PROJECT-ROOT
3. .devcontainer/ in directory hierarchy from PROJECT-ROOT
4. docker-compose.yml directly in PROJECT-ROOT
5. docker-compose.yml in directory hierarchy from PROJECT-ROOT

Returns the directory path or nil if not found."
  (let ((search-dir (or project-root default-directory)))
    (or
     ;; 1. Extract from CLI command -f flag
     (when-let ((compose-file (claude-agent--extract-compose-file-from-command cli-path)))
       (let* ((expanded (if (file-name-absolute-p compose-file)
                            compose-file
                          (expand-file-name compose-file (or project-root default-directory))))
              (dir (file-name-directory expanded)))
         (when (file-exists-p expanded)
           dir)))
     ;; 2. Look for .devcontainer/docker-compose.yml from project-root
     (when project-root
       (let ((devcontainer (expand-file-name ".devcontainer" project-root)))
         (when (file-exists-p (expand-file-name "docker-compose.yml" devcontainer))
           devcontainer)))
     ;; 3. Look for .devcontainer in directory hierarchy

*** Container Startup

#+BEGIN_SRC elisp
(defun claude-agent--ensure-docker-container (&optional cli-path project-root)
  "Ensure the Claude Docker container is running.
Starts the container if not already running.

Optional CLI-PATH is the Docker command which may contain -f flag.
Optional PROJECT-ROOT is the project directory for resolving relative paths.

Returns t if container is ready, signals error on failure."
  (if (claude-agent--docker-container-running-p cli-path project-root)
      t
    (let ((compose-dir (claude-agent--find-compose-dir cli-path project-root)))
      (unless compose-dir
        (error "Cannot find docker-compose.yml for Claude container"))
      (message "Starting Claude Docker container...")
      (let* ((default-directory compose-dir)
             (output-buffer (get-buffer-create "*docker-compose-output*"))
             (exit-code (with-current-buffer output-buffer
                          (erase-buffer)
                          (call-process "docker" nil t nil "compose" "up" "-d"))))
        (unless (zerop exit-code)
          (error "Docker compose up failed (exit code %d). See *docker-compose-output* buffer"
                 exit-code)))
      ;; Wait for container to be ready (includes Claude Code installation)
      (let ((retries 30)  ; 30 seconds max
            (ready nil))
        (while (and (> retries 0) (not ready))
          (sleep-for 1)
          (setq ready (claude-agent--docker-container-running-p cli-path project-root))
          (cl-decf retries))
        (if ready
            (progn
              (message "Claude Docker container ready")
              t)
          (error "Failed to start Claude Docker container (timeout)"))))))

(defun claude-agent--build-args (options &optional streaming-p)
  "Build CLI arguments from OPTIONS plist.
If STREAMING-P is non-nil, use streaming input mode."
  (let ((args (list "--output-format" "stream-json" "--verbose")))
    ;; === Execution Limits ===
    (let ((system-prompt (plist-get options :system-prompt)))
      (when system-prompt
        (setq args (append args (list "--system-prompt" system-prompt)))))
    (let ((max-turns (plist-get options :max-turns)))

        (setq args (append args (list "--max-turns" (number-to-string max-turns))))))

      (when max-budget
        (setq args (append args (list "--max-budget-usd" (number-to-string max-budget))))))

    ;; === Permission Mode ===
    (let ((perm-mode (plist-get options :permission-mode)))
      (when perm-mode
        (let ((mode-str (claude-agent--permission-mode-to-string perm-mode)))
          (when mode-str
            (setq args (append args (list "--permission-mode" mode-str)))))))

    ;; === Permission Callback (requires streaming mode for control protocol) ===
    (when (and streaming-p claude-agent-permission-functions)
      (setq args (append args (list "--permission-prompt-tool" "stdio"))))

    ;; === Tool Controls ===
    (let ((allowed-tools (plist-get options :allowed-tools)))
      (when allowed-tools
        (setq args (append args (list "--allowedTools"
                                      (string-join allowed-tools ","))))))
    (let ((disallowed-tools (plist-get options :disallowed-tools)))
      (when disallowed-tools
        (setq args (append args (list "--disallowedTools"
                                      (string-join disallowed-tools ","))))))
    (let ((mcp-config (plist-get options :mcp-config)))
      (when mcp-config
        (setq args (append args (list "--mcp-config" mcp-config)))))

    ;; === Session Management ===
    (when (plist-get options :continue-conversation)
      (setq args (append args (list "--continue"))))
    (let ((resume (plist-get options :resume)))
      (when resume
        (setq args (append args (list "--resume" resume)))))
    (let ((setting-sources (plist-get options :setting-sources)))
      (when setting-sources
        (setq args (append args (list "--setting-sources"
                                      (string-join setting-sources ","))))))

    ;; === Output Format ===
    (if streaming-p
        (setq args (append args (list "--input-format" "stream-json")))
      (setq args (append args (list "--print"))))
    args))

(defun claude-agent--remove-hash-entries-by-value (table value)
  "Remove all entries from TABLE whose value equals VALUE.
Collects keys first to avoid modify-during-iterate."
  (let (keys)
    (maphash (lambda (k v)
               (when (equal v value) (push k keys)))
             table)
    (dolist (k keys) (remhash k table))))

(defun claude-agent-registry-cleanup-process (state)
  "Per-process cleanup for STATE. Removes only entries owned by this process.
Never calls clrhash — preserves parallel sessions.
Also unregisters from `claude-agent--active-queries' to ensure the
*Claude Queries* buffer and mode-line reflect the terminal state."
  (let* ((req-id (claude-agent--process-state-request-id state))
         (reg claude-agent--registry))
    ;; Remove from active-states list
    (setf (claude-agent--registry-active-states reg)
          (delq state (claude-agent--registry-active-states reg)))
    ;; Remove owned entries from all registry tables
    (when req-id
      (claude-agent--remove-hash-entries-by-value
       (claude-agent--registry-background-tasks reg) req-id)
      (claude-agent--remove-hash-entries-by-value
       (claude-agent--registry-control-requests reg) req-id)
      (remhash req-id (claude-agent--registry-queries reg))
      ;; Also unregister from active-queries hash table.
      ;; This is critical for the abnormal exit (recovery) path where
      ;; sentinel-cleanup is NOT called, so without this the old dead
      ;; entry would persist in active-queries indefinitely.
      (claude-agent--unregister-query req-id))))

*** Error Detection

Detect and extract error messages from various JSON response formats.
The Claude CLI can signal errors in multiple ways - result messages with
=is_error=true=, assistant messages with error fields, etc.

#+NAME: filter-extract-error
#+BEGIN_SRC elisp
(defconst claude-agent--known-error-subtypes
  '(("error_during_execution" . "Execution error (Claude encountered an issue during tool use)")
    ("invalid_request" . "Invalid request error")
    ("rate_limit" . "Rate limit exceeded"))
  "Alist mapping known error subtype strings to human-readable messages.")

(defun claude-agent--extract-result-error (parsed)
  "Extract error from a result message in PARSED JSON.
Checks :errors array, known subtypes, error_details, field values,
and deep-search as fallback.  Returns error string or nil."
  (let ((subtype (plist-get parsed :subtype))
        (errors (plist-get parsed :errors)))
    (cond
     ;; :errors array with specific error messages
     ((and errors (listp errors) (car errors) (stringp (car errors)))
      (car errors))
     ;; Known subtype with human-readable message
     ((cdr (assoc subtype claude-agent--known-error-subtypes)))
     ;; error_details field
     ((plist-get parsed :error_details)
      (let ((details (plist-get parsed :error_details)))
        (if (stringp details) details (format "%S" details))))
     ;; Try :result, :error, :message — skip values that duplicate subtype names
     ((let ((subtype-values (mapcar #'car claude-agent--known-error-subtypes)))
        (or (let ((v (plist-get parsed :result)))
              (and v (not (member v subtype-values)) v))
            (let ((v (plist-get parsed :error)))
              (and v (not (member v subtype-values)) v))
            (let ((v (plist-get parsed :message)))
              (and v (not (member v subtype-values)) v)))))
     ;; Other subtype values — format nicely
     ((and subtype (not (string= subtype "")))
      (format "Error: %s" subtype))
     ;; Deep search for meaningful error strings
     (t
      (or
       (cl-labels ((find-error-string (obj)
                     (cond
                      ((stringp obj)
                       (when (and (> (length obj) 20)
                                  (string-match-p "\\(?:error\\|Error\\|failed\\|not found\\)" obj))
                         obj))
                      ((listp obj)
                       (cl-some #'find-error-string obj))
                      (t nil))))
         (find-error-string parsed))
       (format "Error (fields: %s)"
               (mapcar #'car (seq-partition (cdr parsed) 2))))))))

(defun claude-agent--extract-assistant-error (parsed)
  "Extract error from an assistant message in PARSED JSON.
Checks for :message with non-empty :error field, extracts text
content or falls back to error string.  Returns error string or nil."
  (let* ((msg (plist-get parsed :message))
         (err (plist-get msg :error)))
    (when (and err (stringp err) (not (string-empty-p err)))
      (let* ((content (plist-get msg :content))
             (text-block (car-safe (seq-filter
                                    (lambda (b) (string= (plist-get b :type) "text"))
                                    content))))
        (or (plist-get text-block :text)
            err
            "API error")))))

(defun claude-agent--extract-json-error (parsed)
  "Extract error message from PARSED JSON if it indicates an error.
Returns the error message string if an error is detected, nil otherwise.
Delegates to sub-extractors for result and assistant message types."
  (let ((type (plist-get parsed :type)))
    (cond
     ((and (string= type "result")
           (claude-agent--json-true-p (plist-get parsed :is_error)))
      (claude-agent--extract-result-error parsed))
     ((string= type "assistant")
      (claude-agent--extract-assistant-error parsed))
     (t nil))))
#+END_SRC

*** Normal Message Processing

Helper function to process non-control messages (user, assistant, result, etc.).
Extracted to avoid duplication in the event dispatch logic.

#+BEGIN_SRC elisp
(defun claude-agent--process-normal-message (parsed state)
  "Process a normal (non-control) message.
PARSED is the JSON plist, STATE is process state.
Callbacks and process are extracted from STATE."
  (let ((callback (claude-agent--process-state-callback state))
        (token-callback (claude-agent--process-state-token-callback state))
        (error-callback (claude-agent--process-state-error-callback state))
        (process (claude-agent--process-state-process state))
        (json-error (claude-agent--extract-json-error parsed))
        (is-result (string= (plist-get parsed :type) "result")))
    ;; Set got-result flag for ANY result message (error or success)
    ;; This prevents infinite recovery loops when CLI returns error results
    (when is-result
      (setf (claude-agent--process-state-got-result state) t))
    (if json-error
        ;; Error detected in JSON - call error callback
        (progn
          (claude-agent--debug-log "JSON error detected: %s" json-error)
          (when error-callback
            (funcall error-callback
                     (list 'claude-agent-api-error
                           :message json-error
                           :parsed parsed))))
      ;; Normal message processing (else-branch is implicit progn)
      ;; Extract text for token callback (streaming)
      (when token-callback
        (claude-agent--extract-tokens parsed token-callback))
      ;; Call message callback with parsed message
      (let ((message (claude-agent--parse-message parsed)))
        (when message
          ;; Capture session-id from system/init message (available at start)
          ;; This is critical for recovery - result message comes too late
          (when (and (claude-agent-system-message-p message)
                     (equal (claude-agent-system-message-subtype message) "init"))
            (when-let ((sid (plist-get (claude-agent-system-message-data message) :session_id)))
              (setf (claude-agent--process-state-session-id state) sid)))
          ;; Set got-result flag when result message received
          (when (claude-agent-result-message-p message)
            (setf (claude-agent--process-state-got-result state) t))
          ;; Write to verbose buffer if session-key is set
          (claude-agent--verbose-write-message
           (claude-agent--process-state-session-key state) message)
          ;; Call user callback
          (when callback
            (funcall callback message))))
      ;; Close stdin when result is received (streaming mode)
      (claude-agent--maybe-close-stdin process parsed))))

*** JSON Buffer Processing

Process accumulated JSON data line by line. Each complete line (terminated by
newline) is parsed and dispatched to the appropriate handler - either control
requests for the bidirectional protocol, or normal message callbacks.

#+BEGIN_SRC elisp
(defun claude-agent--try-parse-json (string)
  "Try to parse STRING as JSON, returning plist or nil on failure.
Uses standard json settings: plist objects, list arrays, keyword keys."
  (condition-case nil
      (let ((json-object-type 'plist)
            (json-array-type 'list)
            (json-key-type 'keyword))
        (json-read-from-string string))
    ((json-parse-error json-readtable-error json-end-of-file
                       json-unknown-keyword error)
     nil)))

#+NAME: filter-process-json-buffer
#+BEGIN_SRC elisp
(defun claude-agent--process-json-buffer (process)
  "Process accumulated JSON for PROCESS.
PROCESS has state stored via `process-get'.
Parses complete newline-delimited JSON and invokes callbacks."
  (let* ((state (process-get process 'claude-agent-state))
         (buf (claude-agent--process-state-json-buffer state))
         (error-callback (claude-agent--process-state-error-callback state)))
    (while (string-match "^\\([^\n]+\\)\n" buf)
      (let* ((line (match-string 1 buf))
             (rest (substring buf (match-end 0))))
        (setf (claude-agent--process-state-json-buffer state) rest)
        (setq buf rest)
        (when (and line (not (string-empty-p (string-trim line))))
          (let ((parsed (claude-agent--try-parse-json line)))
            (if parsed
                (let ((msg-type (plist-get parsed :type)))
                  (claude-agent--debug-log "Parsed JSON: %S" msg-type)

                  ;; Control requests use direct dispatch (bidirectional protocol)
                  (if (equal msg-type "control_request")
                      (claude-agent--handle-control-request process parsed)
                    ;; All other types: generic dispatch via cl-defmethod
                    (claude-agent-handle-message
                     (intern msg-type) parsed state)))
              ;; Line is not valid JSON - could be plain text error from CLI
              (claude-agent--debug-log "Non-JSON output: %s" line)
              (when error-callback
                (funcall error-callback
                         (list 'claude-agent-cli-error
                               :message line
                               :original-error nil))))))))))

*** Process Filter Entry Point

The filter function is called by Emacs whenever data arrives from the CLI
subprocess. It accumulates data and triggers JSON processing.

#+NAME: filter-main
#+BEGIN_SRC elisp
(defun claude-agent--process-filter (process output)
  "Process filter function for PROCESS.
OUTPUT is the data received from the process.
Uses `process-get' to retrieve state instead of closures (for dynamic binding compat)."
  (let ((state (process-get process 'claude-agent-state)))
    (when (and state (claude-agent--process-state-ready state))
      ;; Log stdout to verbose buffer for debugging
      (when-let ((session-key (claude-agent--process-state-session-key state)))
        (claude-agent--verbose-insert session-key
                                      (format "<<< STDOUT: %s" output)))
      (let* ((json-buf (or (claude-agent--process-state-json-buffer state) ""))
             (new-buf (concat json-buf output)))
        (if (> (length new-buf) claude-agent-max-json-buffer-size)
            ;; Buffer overflow — clear buffer, signal error, kill process
            (progn
              ;; Clear buffer FIRST to prevent sentinel from re-parsing garbage
              (setf (claude-agent--process-state-json-buffer state) "")
              (let ((err-cb (claude-agent--process-state-error-callback state)))
                (when err-cb
                  (funcall err-cb
                           (list :error
                                 (format "JSON buffer overflow: %d bytes exceeds %d byte limit"
                                         (length new-buf)
                                         claude-agent-max-json-buffer-size)))))
              (when (process-live-p process)
                (delete-process process)))
          (setf (claude-agent--process-state-json-buffer state) new-buf)
          (claude-agent--process-json-buffer process))))))

** Process Sentinel

The sentinel handles process completion and cleanup.

#+BEGIN_SRC elisp
(defun claude-agent--sentinel-handle-normal-exit (state event)
  "Handle normal (non-recovery) exit for STATE with EVENT.
Calls the complete callback with nil for successful exit, or with
an error plist for abnormal exit codes.
Skips the complete callback entirely if the query was cancelled,
since the cancel path already handled cleanup."
  ;; When a query is cancelled, claude-org-cancel (or cancel-query) already
  ;; cleans up the session state (frees marker, sets status, stops spinner).
  ;; Calling complete-callback here would cause duplicate cleanup and fire
  ;; completion hooks with the wrong status.
  (unless (claude-agent--process-state-cancelled state)
    (let ((complete-cb (claude-agent--process-state-complete-callback state)))
      (cond
       ((string-match-p "finished" event)
        (when complete-cb
          (funcall complete-cb nil)))
       ((string-match-p "exited abnormally" event)
        (let ((exit-code (when (string-match "code \\([0-9]+\\)" event)
                           (string-to-number (match-string 1 event)))))
          (when complete-cb
            (funcall complete-cb
                     (list 'claude-agent-process-error
                           :message "Process exited abnormally"
                           :exit-code exit-code)))))
       (t
        (when complete-cb
          (funcall complete-cb
                   (list 'claude-agent-process-error
                         :message (format "Process event: %s" event)))))))))

(defun claude-agent--sentinel-handle-abnormal-exit (process state event)
  "Handle abnormal exit (recovery path) for PROCESS with STATE and EVENT.
Inserts recovery message, kills orphaned children, does minimal cleanup,
then resumes the session."
  (when-let ((session-key (claude-agent--process-state-session-key state)))
    (claude-agent--verbose-insert session-key
                                  (format ">>> ABNORMAL EXIT DETECTED - initiating recovery")))
  (claude-agent--insert-recovery-message state event)
  (claude-agent--kill-child-processes (process-id process))
  ;; Registry-aware cleanup (removes from active-states, queries, etc.)
  (claude-agent-registry-cleanup-process state)
  ;; Kill output buffer after registry cleanup
  (when-let* ((buf (claude-agent--process-state-buffer state)))
    (when (buffer-live-p buf)
      (kill-buffer buf)))
  (claude-agent--resume-session state))

(defun claude-agent--sentinel-cleanup (process state)
  "Clean up after PROCESS exits with STATE.
Kills orphaned children, removes from active-states, unregisters
query, clears Docker mappings, and kills output buffer."
  ;; Kill any orphaned child processes (delegate to existing helper)
  (claude-agent--kill-child-processes (process-id process))
  ;; Remove from active states
  (setq claude-agent--active-states
        (delq state claude-agent--active-states))
  ;; Per-process cleanup and unregister (single req-id extraction)
  (when-let* ((req-id (claude-agent--process-state-request-id state)))
    (claude-agent--cleanup-process-entries req-id)
    (claude-agent--unregister-query req-id))
  ;; Clear Docker path mappings if this process owned them
  (when (claude-agent--process-state-docker-mode state)
    (setq emacs-mcp-server-path-mappings nil))
  (when-let* ((buf (claude-agent--process-state-buffer state)))
    (when (buffer-live-p buf)
      (kill-buffer buf))))

(defun claude-agent--process-sentinel (process event)
  "Process sentinel function for PROCESS.
EVENT is the process event string.
Delegates to sub-handlers for normal exit, abnormal exit (recovery),
and cleanup."
  (let ((state (process-get process 'claude-agent-state)))
    (claude-agent--debug-log "Process event: %s" (string-trim event))
    (when state
      ;; Always log process exit to verbose buffer for diagnostics
      (when-let ((session-key (claude-agent--process-state-session-key state)))
        (claude-agent--verbose-insert session-key
                                      (format ">>> PROCESS EXIT: %s (pending-bg=%d pending-ctrl=%d)"
                                              (string-trim event)
                                              (hash-table-count claude-agent--pending-background-tasks)
                                              (hash-table-count claude-agent--pending-control-requests))))
      (setf (claude-agent--process-state-closed state) t)
      (setf (claude-agent--process-state-ready state) nil)
      ;; Process any remaining JSON in buffer
      (claude-agent--process-json-buffer process)
      ;; Handle any remaining content without trailing newline
      (let ((remaining (claude-agent--process-state-json-buffer state)))
        (when (and remaining (not (string-empty-p (string-trim remaining))))
          (let ((trimmed (string-trim remaining))
                (error-callback (claude-agent--process-state-error-callback state))
                (session-key (claude-agent--process-state-session-key state)))
            (claude-agent--debug-log "Remaining output without newline: %s" trimmed)
            (let ((parsed (claude-agent--try-parse-json trimmed)))
              (if parsed
                  (let ((msg-type (plist-get parsed :type)))
                    (claude-agent--debug-log "Parsed remaining JSON: %S" msg-type)
                    (if (equal msg-type "control_request")
                        (progn
                          (when session-key
                            (claude-agent--verbose-insert
                             session-key
                             (format ">>> WARNING: control_request received without newline (CLI died?) - cannot respond")))
                          (message "Claude: control_request received but CLI died before we could respond"))
                      (when session-key
                        (claude-agent--verbose-insert
                         session-key
                         (format ">>> INFO: Final message without newline: %s" msg-type)))))
                (when error-callback
                  (funcall error-callback
                           (list 'claude-agent-cli-error
                                 :message trimmed
                                 :original-error nil))))))))
      ;; Dispatch to normal or abnormal exit handler
      (if (claude-agent--is-abnormal-exit-p event state)
          (claude-agent--sentinel-handle-abnormal-exit process state event)
        (claude-agent--sentinel-handle-normal-exit state event)
        (claude-agent--sentinel-cleanup process state)))))

** Pipe Deadlock Prevention

When using pipes for bidirectional communication, deadlock can occur if:
1. The CLI writes to stdout faster than Emacs reads
2. The pipe buffer fills up (64KB Linux, 16KB macOS)
3. The CLI blocks waiting for buffer space
4. Emacs blocks waiting for CLI to accept stdin

Solution: Always drain pending output before writing to stdin, and chunk
large messages with interleaved draining.

#+BEGIN_SRC elisp
(defun claude-agent--drain-output (process &optional timeout-ms)
  "Drain pending output from PROCESS before sending input.
TIMEOUT-MS is milliseconds to wait per drain cycle, defaults to 50ms.
Returns t if any output was drained, nil otherwise."
  (let ((timeout-secs (/ (or timeout-ms 50) 1000.0))
        (drained nil))
    (while (and (process-live-p process)
                (accept-process-output process timeout-secs nil t))
      (setq drained t))
    drained))

(defun claude-agent--send-json-safe (process json-str session-key)
  "Send JSON-STR to PROCESS with deadlock prevention.
Drains output before sending, chunks large messages.
SESSION-KEY is used for verbose logging."
  ;; Log to verbose buffer
  (when session-key
    (claude-agent--verbose-insert session-key
                                  (format ">>> STDIN: %s" json-str)))
  (claude-agent--debug-log "Sending to stdin: %s"
                           (substring json-str 0 (min 200 (length json-str))))
  ;; Only send if process is alive
  (when (process-live-p process)
    ;; CRITICAL: Drain stdout first to prevent deadlock
    (claude-agent--drain-output process)
    ;; Send in chunks with interleaved draining for large messages
    (let ((len (length json-str))
          (pos 0)
          (chunk-size claude-agent-stdin-chunk-size))
      (while (< pos len)
        ;; Drain between chunks (except first)
        (when (> pos 0)
          (claude-agent--drain-output process 10))
        (let ((chunk (substring json-str pos (min (+ pos chunk-size) len))))
          (process-send-string process chunk))
        (setq pos (+ pos chunk-size))))))

(defun claude-agent--configure-process-io (process)
  "Configure PROCESS for optimal I/O throughput.
Sets buffer sizes and encoding for reliable bidirectional communication."
  (with-current-buffer (process-buffer process)
    ;; Increase read chunk size (Emacs 27+)
    (when (boundp 'read-process-output-max)
      (setq-local read-process-output-max (* 1024 1024)))
    ;; Disable adaptive buffering for lower latency
    (setq-local process-adaptive-read-buffering nil))
  ;; UTF-8 encoding (already set in make-process, but be explicit)
  (set-process-coding-system process 'utf-8-unix 'utf-8-unix))
#+END_SRC

** Background Task Tracking

Track background tasks (launched with =run_in_background: true=) to keep the
process alive until all tasks complete. The tracker is registered on the
=claude-agent-post-tool-use-functions= hook.

Two types of background tasks exist:
1. *Task tool* (subagents): Returns =isAsync: true= with =agentId=
2. *Bash tool*: Returns =backgroundTaskId= field (no =isAsync= flag)

Both must be tracked to prevent premature stdin close.

#+BEGIN_SRC elisp
(defvar claude-agent--pending-background-tasks (make-hash-table :test 'equal)
  "Hash table of task-id -> owner-request-id for pending background tasks.
Tracks both Task subagents (agentId) and Bash background tasks (backgroundTaskId).
Values are the owning process's request-id for per-process cleanup.")

(defun claude-agent--background-task-tracker (_tool-use-id _result _is-error
                                              tool-use-result state)
  "Track background tasks from PostToolUse hook.
STATE is the process-state struct (used to identify the owning process).
TOOL-USE-RESULT contains:
- For Task tool: isAsync=t, agentId (e.g., \"a4b3ecf\")
- For Bash tool: backgroundTaskId (e.g., \"b8406fe\")
- For TaskOutput: task.task_id, task.status"
  (when tool-use-result
    (let ((owner (and state (claude-agent--process-state-request-id state))))
      ;; Case 1: Task tool async launch - isAsync=t with agentId
      (when (and (eq (plist-get tool-use-result :isAsync) t)
                 (plist-get tool-use-result :agentId))
        (let ((agent-id (plist-get tool-use-result :agentId)))
          (puthash agent-id (or owner t) claude-agent--pending-background-tasks)
          (claude-agent--debug-log "Background Task started: %s (owner: %s)"
                                   agent-id owner)))
      ;; Case 2: Bash tool background - backgroundTaskId field
      (when-let ((bg-task-id (plist-get tool-use-result :backgroundTaskId)))
        (puthash bg-task-id (or owner t) claude-agent--pending-background-tasks)
        (claude-agent--debug-log "Background Bash started: %s (owner: %s)"
                                 bg-task-id owner))
      ;; Task completion: task.status="completed"
      (when-let* ((task (plist-get tool-use-result :task))
                  (task-id (plist-get task :task_id))
                  (status (plist-get task :status)))
        (when (equal status "completed")
          (remhash task-id claude-agent--pending-background-tasks)
          (claude-agent--debug-log "Background task completed: %s" task-id))))))

(defun claude-agent--has-pending-background-tasks-p (&optional owner-request-id)
  "Return non-nil if there are pending background tasks.
When OWNER-REQUEST-ID is non-nil, only count tasks owned by that process.
When nil, checks the global count (any owner).  Callers should guard
with (and owner-req-id (has-pending-...)) to avoid stale-task blocking."
  (if owner-request-id
      (catch 'found
        (maphash (lambda (_k v)
                   (when (equal v owner-request-id)
                     (throw 'found t)))
                 claude-agent--pending-background-tasks)
        nil)
    (> (hash-table-count claude-agent--pending-background-tasks) 0)))

;; Register tracker on PostToolUse hook
(add-hook 'claude-agent-post-tool-use-functions
          #'claude-agent--background-task-tracker)

** Closing Stdin After Result

We need to close stdin when we receive the result message to signal completion.
This is handled in the process filter when a result message is received.
If there are pending background tasks, we keep stdin open.

*** Why Delayed Stdin Close is Necessary

The Claude Code CLI uses a bidirectional protocol over stdin/stdout:
- We send: user prompts, control responses (permissions, hooks)
- CLI sends: messages, control requests (can_use_tool, mcp_message, hook_callback)

When we receive a =result= message, it signals the end of the conversation turn.
However, closing stdin immediately can cause issues:

1. *Process Filter Timing*: The =result= message is processed inside
   =claude-agent--process-filter=. If we call =process-send-eof= synchronously,
   it may trigger process termination before the current filter completes.

2. *MCP Server Communication*: With =emacs-mcp-server= active, there may be
   pending =mcp_message= control requests or responses in flight.

3. *Hook Callbacks*: Any pending hook callbacks need to complete before
   the connection closes.

The Python SDK handles this by waiting for the result event before closing
stdin when MCP servers or hooks are present (see =stream_input= in query.py).

Our solution: Use =run-at-time= with a small delay (0.1s) to:
- Let the current process filter invocation complete
- Allow any pending filter calls to run
- Give MCP communication time to settle
- Then close stdin cleanly

#+BEGIN_SRC elisp
(defcustom claude-agent-stdin-close-delay 0.1
  "Delay in seconds before closing stdin after receiving result message.
This delay allows pending process filter invocations and MCP communication
to complete before signaling end-of-input to the Claude CLI.

The Python SDK uses a similar approach, waiting for the result event
before closing stdin when MCP servers are active.

Set to 0 to close immediately (original behavior, may cause early exit issues).
Set to nil to disable automatic stdin close (for debugging)."
  :type '(choice (number :tag "Delay in seconds")
                 (const :tag "Disable auto-close" nil))
  :group 'claude-agent)

(defun claude-agent--maybe-close-stdin (process parsed)
  "Close stdin if PARSED is a result message and no pending work.
PROCESS is the CLI process.

Checks for both:
1. Pending background tasks owned by THIS process (not stale tasks from dead processes)
2. Pending control requests awaiting responses (permissions, hooks)

This function schedules stdin closure with a small delay to ensure:
1. The current process filter completes processing
2. Any pending MCP control requests/responses finish
3. Hook callbacks have time to execute

The delay is controlled by `claude-agent-stdin-close-delay'."
  (when (equal (plist-get parsed :type) "result")
    (let* ((state (and process (processp process)
                      (process-get process 'claude-agent-state)))
           (owner-req-id (and state
                              (claude-agent--process-state-request-id state))))
      (cond
       ;; Check for pending background tasks owned by THIS process
       ;; When owner-req-id is nil (no state), skip — don't let stale global tasks block
       ((and owner-req-id
             (claude-agent--has-pending-background-tasks-p owner-req-id))
        (claude-agent--debug-log "Result received, background tasks pending for %s"
                                 owner-req-id))
       ;; Check for pending control requests awaiting responses
       ((claude-agent--has-pending-control-requests-p)
        (claude-agent--debug-log "Result received, %d control requests pending response"
                                 (hash-table-count claude-agent--pending-control-requests)))
       ;; No pending work - schedule stdin close
       (t
        (let ((delay claude-agent-stdin-close-delay))
          (unless delay
            (claude-agent--debug-log "Result received, stdin auto-close disabled"))
          (when delay
            (claude-agent--debug-log "Result received, scheduling stdin close in %.1fs" delay)
            (if (zerop delay)
              ;; Immediate close (original behavior)
              (when (process-live-p process)
                (process-send-eof process))
            ;; Delayed close - let filter complete and MCP settle
            ;; Use lexical-let to capture process (literate-elisp ignores lexical-binding: t)
            (lexical-let ((proc process)
                          (req-id owner-req-id))
              (run-at-time
               delay nil
               (lambda ()
                 ;; Re-check for pending work before closing
                 ;; When req-id is nil, skip background task check (no owner to match)
                 (when (and (process-live-p proc)
                            (not (and req-id
                                      (claude-agent--has-pending-background-tasks-p req-id)))
                            (not (claude-agent--has-pending-control-requests-p)))
                   (claude-agent--debug-log "Closing stdin after delay")
                   (process-send-eof proc)))))))))))))

(defcustom claude-agent-bun-memory-limit nil
  "Memory limit in bytes for Bun's JSC heap via BUN_JSC_forceRAMSize.
When non-nil, this value is injected into the subprocess environment,
telling Bun's JavaScriptCore engine to treat available RAM as this many
bytes.  This constrains JS heap growth and triggers GC earlier, helping
prevent OOM kills (signal 9) on macOS.

Recommended value: (* 3 1024 1024 1024) for 3 GB.
Set to nil to disable (default)."
  :type '(choice (integer :tag "Bytes")
                 (const :tag "Disabled" nil))
  :group 'claude-agent)

(defvar claude-agent--stripped-env-vars '("CLAUDECODE")
  "Environment variables to strip when spawning Claude CLI subprocesses.
Claude CLI refuses to launch when CLAUDECODE is set (nested session detection).
Our SDK must unset it so Emacs users can run queries from within a Claude Code session.")

(defun claude-agent--build-process-environment (env-vars cli-dir)
  "Build process environment for a Claude CLI subprocess.
ENV-VARS is an alist of (NAME . VALUE) pairs to add.
CLI-DIR is the directory of the CLI binary (prepended to PATH), or nil.
Strips variables listed in `claude-agent--stripped-env-vars'.
Injects BUN_JSC_forceRAMSize when `claude-agent-bun-memory-limit' is set."
  (let ((base-env (cl-remove-if
                   (lambda (entry)
                     (cl-some (lambda (var)
                                (string-prefix-p (concat var "=") entry))
                              claude-agent--stripped-env-vars))
                   process-environment)))
    (append
     (when cli-dir
       (list (concat "PATH=" cli-dir ":" (getenv "PATH"))))
     ;; Bun memory tuning: cap JSC heap to prevent OOM kills
     (when claude-agent-bun-memory-limit
       (list (format "BUN_JSC_forceRAMSize=%d" claude-agent-bun-memory-limit)))
     (mapcar (lambda (pair)
               (format "%s=%s" (car pair) (cdr pair)))
             env-vars)
     base-env)))

(defun claude-agent--kill-child-processes (pid)
  "Kill all child processes of process with PID.
This prevents orphaned shells from surviving after the main process exits.
PID is guaranteed to be a number from `process-id', so shell injection is not possible."
  (when (and pid (> pid 0))
    (let ((child-pids (split-string
                      (string-trim
                       (shell-command-to-string
                        (format "pgrep -P %d 2>/dev/null" pid)))
                      "\n" t)))
      (dolist (child-pid child-pids)
        (let ((cpid (string-to-number child-pid)))
          (when (> cpid 0)
            (signal-process cpid 9)))))))

(cl-defun claude-agent--start-process (prompt options &key callback error-callback
                                            complete-callback token-callback
                                            request-id session-key source-buffer
                                            query-context source-location-fn
                                            source-location-args)
  "Start Claude CLI process with PROMPT and OPTIONS.

All other arguments are keyword arguments:
  :CALLBACK is called with each parsed message.
  :ERROR-CALLBACK is called on errors.
  :COMPLETE-CALLBACK is called when process completes.
  :TOKEN-CALLBACK is called with each text token for streaming.
  :REQUEST-ID is a unique identifier for this request (auto-generated if nil).
  :SESSION-KEY identifies the session for verbose output grouping.
  :SOURCE-BUFFER is the buffer that initiated this query.
  :QUERY-CONTEXT is a claude-agent-query-context struct for display.
  :SOURCE-LOCATION-FN is a function symbol to call for navigation.
  :SOURCE-LOCATION-ARGS is a list of args for SOURCE-LOCATION-FN.

Uses streaming mode by default to enable bidirectional control protocol
for features like canUseTool permission callback.

Returns a process state object."
  (let* ((raw-cli-path (or (plist-get options :cli-path)
                           (claude-agent--find-cli)))
         ;; For Docker commands, split into args; for regular paths, expand
         (cli-parts (if (claude-agent--docker-command-p raw-cli-path)
                        (claude-agent--split-command raw-cli-path)
                      (list (expand-file-name raw-cli-path))))
         ;; Always use streaming mode for bidirectional control protocol
         (args (claude-agent--build-args options t))
         (cwd (plist-get options :cwd))
         (env-vars (plist-get options :env))
         (req-id (or request-id (claude-agent--generate-request-id)))
         (docker-mode-p (claude-agent--docker-command-p raw-cli-path))
         (state (claude-agent--make-process-state
                 :options options
                 :json-buffer ""
                 :callback callback
                 :token-callback token-callback
                 :error-callback error-callback
                 :complete-callback complete-callback
                 :request-id req-id
                 :start-time (float-time)
                 :session-key session-key
                 :source-buffer source-buffer
                 :query-context query-context
                 :source-location-fn source-location-fn
                 :source-location-args source-location-args
                 :docker-mode docker-mode-p
                 :ready nil
                 :closed nil))
         (default-directory (or cwd default-directory))
         ;; When cli-path is explicit, prepend its directory to exec-path
         ;; This ensures #!/usr/bin/env node finds the correct node version
         ;; (e.g., when using nvm with claude installed under a specific node)
         (cli-dir (when (and raw-cli-path
                             (not (claude-agent--docker-command-p raw-cli-path)))
                    (file-name-directory (expand-file-name raw-cli-path))))
         (exec-path (if cli-dir
                        (cons cli-dir exec-path)
                      exec-path))
         ;; Build process-environment: custom vars, CLI PATH, stripped nesting vars
         (process-environment (claude-agent--build-process-environment
                               env-vars cli-dir)))
    ;; Build command; for Docker commands, cli-parts is already a list
    (let* ((command (append cli-parts args))
           (process (make-process
                    :name "claude-agent"
                    :command command
                    :buffer (generate-new-buffer " *claude-agent*")
                    :connection-type 'pipe  ;; Use pipe, not PTY
                    :filter #'claude-agent--process-filter
                    :sentinel #'claude-agent--process-sentinel
                    :coding 'utf-8-unix
                    :noquery t)))
      ;; Store state on process for filter/sentinel to access
      ;; (avoids closure issues with dynamic binding)
      (process-put process 'claude-agent-state state)
      (setf (claude-agent--process-state-process state) process)
      (setf (claude-agent--process-state-buffer state) (process-buffer process))
      ;; Configure I/O for optimal throughput and deadlock prevention
      (claude-agent--configure-process-io process)
      ;; Register query for cancellation tracking
      (claude-agent--register-query req-id state)
      ;; Initialize verbose buffer for session if session-key provided
      (when session-key
        (claude-agent--get-session-verbose-buffer session-key)
        (claude-agent--verbose-insert-query-header session-key prompt
                                                   (string-join cli-parts " ")
                                                   args default-directory env-vars))
      ;; Streaming mode: Send initialization and prompt via stdin
      (claude-agent--streaming-init process state prompt)
      (push state claude-agent--active-states)
      state)))

(defun claude-agent--prepare-query-options (options cli-path project-root)

(cl-defun claude-agent-query (prompt &key options on-message on-token on-error on-complete
                                     session-key source-buffer query-context
                                     source-location-fn source-location-args
                                     cli-path path-mappings project-root
                                     org-section)
  "Send a one-shot query to Claude.

PROMPT is the query string to send.

OPTIONS is an optional plist created with `claude-agent-options'.
ORG-SECTION is the name of the org section/story for display in permission prompts.

ON-MESSAGE is called with each message as it arrives.
  Function signature: (lambda (message) ...)
  where MESSAGE is one of:
    - `claude-agent-user-message'
    - `claude-agent-assistant-message'
    - `claude-agent-system-message'
    - `claude-agent-result-message'

ON-TOKEN is called with each text token for streaming display.
  Function signature: (lambda (text) ...)
  This is useful for real-time streaming UIs like claude-org mode.

ON-ERROR is called if an error occurs.
  Function signature: (lambda (error) ...)

ON-COMPLETE is called when the query finishes.
  Function signature: (lambda (result-or-error) ...)

SESSION-KEY identifies the session for verbose output grouping (optional).
  When provided, messages will be written to a per-session verbose buffer.

SOURCE-BUFFER is the buffer that initiated this query (optional).
  Used for display in session manager and mode-line.

QUERY-CONTEXT is a claude-agent-query-context struct for display (optional).
  Used together with SOURCE-BUFFER for display.

SOURCE-LOCATION-FN is a function symbol to call for navigation (optional).
  When non-nil, the query will be shown as a navigable link in session manager.
  The function is called with SOURCE-LOCATION-ARGS in SOURCE-BUFFER.

SOURCE-LOCATION-ARGS is a list of args for SOURCE-LOCATION-FN (optional).
  Passed to the location function when navigating to source.

CLI-PATH overrides `claude-agent-cli-path' for this query (optional).
  Use for Docker execution, e.g., \"docker compose exec claude claude\".
  When a Docker command is detected and `claude-agent-docker-auto-start' is t,
  the container will be started automatically.

PATH-MAPPINGS is an alist of (HOST-PATH . CONTAINER-PATH) for Docker mode (optional).
  These mappings are let-bound to `emacs-mcp-server-path-mappings' during query
  execution, enabling path translation between container and host.

PROJECT-ROOT is the project directory path (optional).
  Used to resolve relative paths in Docker compose commands and for finding
  docker-compose.yml. Typically set from org property PROJECT_ROOT or
  CLAUDE_HOST_PATH.

Returns the process state object for potential interruption.
The process state includes a request-id that can be used with
`claude-agent-cancel-query' for cancellation.

Example:
  (claude-agent-query
   \"What is 2+2?\"
   :on-message (lambda (msg)
                 (when (claude-agent-assistant-message-p msg)
                   (message \"Response: %s\"
                            (claude-agent-extract-text msg))))
   :on-token (lambda (text)
               (insert text))  ; For streaming display
   :on-complete (lambda (result)
                  (message \"Done!\"))
   :session-key \"my-session\"
   :source-buffer (current-buffer)
   :query-context (claude-agent-make-query-context :instruction-num 5))

Docker example:
  (claude-agent-query
   \"List files\"
   :cli-path \"docker compose exec claude claude\"
   :path-mappings \\='((\"/home/user/project\" . \"/workspace\")))"
  (unless prompt
    (error "Prompt is required"))
  (let* ((prepared (claude-agent--prepare-query-options options cli-path project-root))
         (effective-cli-path (plist-get prepared :cli-path))
         (opts (plist-get prepared :options)))
    ;; Set path mappings for MCP server (for Docker mode path translation).
    ;; This must be setq (not let-bound) because MCP requests arrive
    ;; asynchronously via HTTP, outside any let-binding scope.
    ;; The variable is cleared on query completion.
    (setq emacs-mcp-server-path-mappings path-mappings)
    ;; Execute query
    (let ((claude-agent-cli-path effective-cli-path))
      (let ((state (claude-agent--start-process
                    prompt opts
                    :callback on-message
                    :error-callback on-error
                    :complete-callback on-complete
                    :token-callback on-token
                    :session-key session-key
                    :source-buffer source-buffer
                    :query-context query-context
                    :source-location-fn source-location-fn
                    :source-location-args source-location-args)))
        ;; Store org-section on the process for permission callbacks
        (when (and state org-section)
          (let ((process (claude-agent--process-state-process state)))
            (when process
              (process-put process :org-section org-section))))
        state))))

(defun claude-agent--kill-process-tree (process &optional signal)
  "Kill PROCESS and all its descendant processes.
SIGNAL defaults to SIGKILL (9).  On Unix, child shells (zsh, bash) create
their own process groups via setsid/setpgid, so killing just the parent or
its process group leaves grandchildren orphaned.  This function uses pgrep
to find all descendants and kills them bottom-up."
  (when (process-live-p process)
    (let ((pid (process-id process))
          (sig (or signal 9)))
      (when pid
        ;; Kill descendants first (children, grandchildren, etc.)
        ;; pgrep -P finds direct children; we recurse via the shell
        (let ((desc-pids (split-string
                          (string-trim
                           (shell-command-to-string
                            (format "pgrep -P %d 2>/dev/null" pid)))
                          "\n" t)))
          (dolist (child-pid desc-pids)
            (let ((child-pid-num (string-to-number child-pid)))
              (when (> child-pid-num 0)
                ;; Recursively kill grandchildren first
                (let ((grandchild-pids
                       (split-string
                        (string-trim
                         (shell-command-to-string
                          (format "pgrep -P %s 2>/dev/null" child-pid)))
                        "\n" t)))
                  (dolist (gc-pid grandchild-pids)
                    (let ((gc-pid-num (string-to-number gc-pid)))
                      (when (> gc-pid-num 0)
                        (signal-process gc-pid-num sig)))))
                (signal-process child-pid-num sig)))))
        ;; Finally kill the main process
        (kill-process process)))))

(defun claude-agent-query-interrupt (state)
  "Interrupt an ongoing query by STATE.
STATE is the process state returned by `claude-agent-query'.
Sets the cancelled flag to prevent automatic recovery."
  (setf (claude-agent--process-state-cancelled state) t)
  (when-let ((process (claude-agent--process-state-process state)))
    (when (process-live-p process)
      (interrupt-process process))))

(defun claude-agent-query-kill (state)
  "Kill an ongoing query by STATE.
Unlike `claude-agent-query-interrupt', this sends SIGKILL which cannot be ignored.
Kills the entire process tree to prevent orphaned shell processes.
Sets the cancelled flag to prevent automatic recovery.
STATE is the process state returned by `claude-agent-query'."
  (setf (claude-agent--process-state-cancelled state) t)
  (when-let ((process (claude-agent--process-state-process state)))
    (when (process-live-p process)
      (claude-agent--kill-process-tree process))))

(defun claude-agent-kill-all-processes ()
  "Kill all active Claude agent processes and their children.
This is useful for cleanup after tests or when processes are stuck.
Returns the number of processes killed."
  (interactive)
  (let ((killed 0))
    ;; Kill via active-states tracking
    (dolist (state claude-agent--active-states)
      (when-let ((process (claude-agent--process-state-process state)))
        (when (process-live-p process)
          (claude-agent--kill-process-tree process)
          (cl-incf killed))))
    ;; Also kill via active-queries hash table
    (maphash (lambda (_req-id state)
               (when-let ((process (claude-agent--process-state-process state)))
                 (when (process-live-p process)
                   (claude-agent--kill-process-tree process)
                   (cl-incf killed))))
             claude-agent--active-queries)
    ;; Clear state tracking
    (setq claude-agent--active-states nil)
    (clrhash claude-agent--active-queries)
    (when (called-interactively-p 'any)
      (message "Killed %d Claude process(es)" killed))
    killed))

(defun claude-agent-extract-text (message)

   ((claude-agent-assistant-message-p message) 'assistant)
   ((claude-agent-system-message-p message) 'system)
   ((claude-agent-result-message-p message) 'result)
   (t 'unknown)))

                      (claude-agent-assistant-message-content message))))

  "Extract thinking content from an assistant MESSAGE.
Returns concatenated thinking text, or nil if none."
  (when (claude-agent-assistant-message-p message)
    (let ((thoughts nil))
      (dolist (block (claude-agent-assistant-message-content message))
        (when (claude-agent-thinking-block-p block)
          (push (claude-agent-thinking-block-thinking block) thoughts)))
      (when thoughts
        (string-join (nreverse thoughts) "\n")))))

(defun claude-agent--extract-error-message (error)
  "Extract human-readable message from ERROR.
ERROR can be (symbol :message ...) plist, a string, or other."
  (cond
   ;; CLI error: (claude-agent-cli-error :message "..." ...) - plist starts at cdr
   ((and (listp error) (symbolp (car error)) (plist-get (cdr error) :message))
    (plist-get (cdr error) :message))
   ;; Plain plist with :message
   ((and (listp error) (plist-get error :message))
    (plist-get error :message))
   ;; String error
   ((stringp error) error)
   ;; Fall back to prin1-to-string
   (t (prin1-to-string error))))

(defun claude-agent-format-elapsed-time (start-time &optional format-style)
  "Format elapsed time since START-TIME as human-readable string.
FORMAT-STYLE can be:
  'compact   - \"5s\", \"10m\", \"2h\" (default for mode-line)
  'verbose   - \"started 5 seconds ago\"
  nil        - defaults to 'compact

Returns \"unknown\" if START-TIME is nil."
  (if (null start-time)
      "unknown"
    (let* ((elapsed (- (float-time) start-time))
           (style (or format-style 'compact)))
      (cond
       ;; Less than 60 seconds
       ((< elapsed 60)
        (if (eq style 'verbose)
            (format "started %d seconds ago" (round elapsed))
          (format "%ds" (round elapsed))))
       ;; Less than 1 hour
       ((< elapsed 3600)
        (if (eq style 'verbose)
            (format "started %d minutes ago" (round (/ elapsed 60)))
          (format "%dm" (round (/ elapsed 60)))))
       ;; 1 hour or more
       (t
        (if (eq style 'verbose)
            (format "started %d hours ago" (round (/ elapsed 3600)))
          (format "%dh" (round (/ elapsed 3600)))))))))

(defun claude-agent-client-connect (client &optional initial-prompt)
  "Connect CLIENT to Claude with optional INITIAL-PROMPT.

Starts the CLI process in streaming mode."
  (when (claude-agent-client-connected-p client)
    (error "Client is already connected"))
  (let* ((options (claude-agent-client-options client))
         (prompt (or initial-prompt ""))
         (on-msg (claude-agent-client-on-message client))
         (on-err (claude-agent-client-on-error client)))
    (setf (claude-agent-client-state client)
          (claude-agent--start-process
           prompt options
           :callback (lambda (msg)
                       ;; Track session ID from result messages
                       (when (claude-agent-result-message-p msg)
                         (setf (claude-agent-client-session-id client)
                               (claude-agent-result-message-session-id msg)))
                       ;; Call user callback
                       (when on-msg
                         (funcall on-msg msg)))
           :error-callback on-err
           :complete-callback (lambda (result)
                                (setf (claude-agent-client-connected-p client) nil)
                                (when (and result on-err)
                                  (funcall on-err result)))))
    (setf (claude-agent-client-connected-p client) t)
    client))

(cl-defun claude-agent-client-send (client message &key on-response on-error)
  "Send MESSAGE through CLIENT (legacy - starts new process).

MESSAGE is a string to send to Claude.
ON-RESPONSE is called with each response message.
ON-ERROR is called if an error occurs.

Note: This creates a new query process for each message.
For persistent bidirectional streaming, use `claude-agent-client-send-message'."
  (unless (claude-agent-client-connected-p client)
    (error "Client is not connected"))
  ;; Legacy: start a new process for each message
  (let* ((options (claude-agent-client-options client))
         ;; Continue from previous session if we have one
         (session-id (claude-agent-client-session-id client))
         (new-options (if session-id
                          (plist-put (copy-sequence options) :resume session-id)
                        options)))
    (setf (claude-agent-client-state client)
          (claude-agent--start-process
           message new-options
           :callback (or on-response (claude-agent-client-on-message client))
           :error-callback (or on-error (claude-agent-client-on-error client))
           :complete-callback
           (lambda (result)
             (when (and result (claude-agent-client-on-error client))
               (funcall (claude-agent-client-on-error client) result)))))))

(cl-defun claude-agent-client-send-message (client message &key on-message on-error
                                                   on-complete token-callback)
  "Send MESSAGE through CLIENT's persistent connection.

This sends the message through the existing process without spawning a new one.
Requires the client to be connected with `claude-agent-client-connect'.

MESSAGE is the prompt string to send.
ON-MESSAGE is called for each response message (overrides client default).
ON-ERROR is called if an error occurs (overrides client default).
ON-COMPLETE is called when the response completes (overrides client default).
TOKEN-CALLBACK is called with each text token for streaming display.

Returns non-nil if the message was sent successfully."
  (unless (claude-agent-client-connected-p client)
    (error "Client is not connected"))
  (let* ((state (claude-agent-client-state client))
         (process (claude-agent--process-state-process state))
         (session-key (claude-agent-client-session-key client)))
    (unless (and process (process-live-p process))
      (error "Client process is not alive"))
    ;; Update client callbacks via public API
    (claude-agent-update-state-callbacks state
     :callback (when on-message
                 (lambda (msg)
                   ;; Track session ID from result messages
                   (when (claude-agent-result-message-p msg)
                     (setf (claude-agent-client-session-id client)
                           (claude-agent-result-message-session-id msg))
                     ;; Call on-complete when we get result
                     (when (or on-complete (claude-agent-client-on-complete client))
                       (funcall (or on-complete (claude-agent-client-on-complete client)) msg)))
                   (funcall on-message msg)))
     :error-callback on-error
     :token-callback token-callback)
    ;; Update client metadata
    (cl-incf (claude-agent-client-query-count client))
    (setf (claude-agent-client-last-activity client) (float-time))
    ;; Send user message through existing process
    (let ((user-message (list :type "user"
                              :message (list :role "user"
                                            :content message))))
      (claude-agent--send-stdin process user-message session-key))
    t))

(defun claude-agent-client-disconnect (client)
  "Disconnect CLIENT from Claude.

Cleans up process and resources, calls on-disconnect callback if set."
  (when (claude-agent-client-connected-p client)
    (let ((on-disconnect (claude-agent-client-on-disconnect client))
          (session-key (claude-agent-client-session-key client)))
      (when-let ((state (claude-agent-client-state client)))
        (when-let ((process (claude-agent--process-state-process state)))
          (when (process-live-p process)
            (claude-agent--kill-process-tree process)))
        (when-let ((buf (claude-agent--process-state-buffer state)))
          (when (buffer-live-p buf)
            (kill-buffer buf))))
      (setf (claude-agent-client-connected-p client) nil)
      (setf (claude-agent-client-state client) nil)
      ;; Call on-disconnect callback
      (when on-disconnect
        (funcall on-disconnect client)))))

(defun claude-agent-client-interrupt (client)
  "Send interrupt signal through CLIENT."
  (when-let ((state (claude-agent-client-state client)))
    (when-let ((process (claude-agent--process-state-process state)))
      (when (process-live-p process)
        (interrupt-process process)))))

(defun claude-agent-continue-session (client)
  "Configure CLIENT to continue from the previous session.
Must be called before `claude-agent-client-connect'."
  (let ((options (claude-agent-client-options client)))
    (setf (claude-agent-client-options client)
          (plist-put options :continue-conversation t))))

(defun claude-agent-resume-session (client session-id)

Must be called before `claude-agent-client-connect'."
  (let ((options (claude-agent-client-options client)))
    (setf (claude-agent-client-options client)
          (plist-put options :resume session-id))))

(defun claude-agent-get-session-id (client)
  "Get the current session ID from CLIENT.
Returns nil if not yet available."
  (claude-agent-client-session-id client))

(defun claude-agent-client-alive-p (client)
  "Return non-nil if CLIENT's process is alive and responsive."
  (and (claude-agent-client-connected-p client)
       (when-let ((state (claude-agent-client-state client)))
         (when-let ((process (claude-agent--process-state-process state)))
           (process-live-p process)))))

(defun claude-agent-client-busy-p (client)
  "Return non-nil if CLIENT is currently processing a query.
A client is busy if it has sent a message and hasn't received a result yet."
  (and (claude-agent-client-alive-p client)
       (when-let ((state (claude-agent-client-state client)))
         ;; Check if we're waiting for a result
         (and (claude-agent--process-state-ready state)
              (not (claude-agent--process-state-closed state))
              ;; Also check if there are pending responses
              (not (claude-agent--process-state-got-result state))))))

** Control Request Handling

Handle incoming control requests from Claude CLI and send responses.

#+BEGIN_SRC elisp
(defun claude-agent--send-control-response (process request-id subtype response-data)
  "Send control protocol response to PROCESS.

REQUEST-ID: string matching the request
SUBTYPE: \"success\" or \"error\"
RESPONSE-DATA: plist with response fields or :error string

Also untracks the request from pending control requests."
  (let* ((response (list :type "control_response"
                        :response (list :subtype subtype
                                       :request_id request-id
                                       :response response-data)))
         (json-object-type 'plist)
         (json-array-type 'list)
         (json-key-type 'keyword)
         (json (concat (json-encode response) "\n"))
         (state (process-get process 'claude-agent-state))
         (session-key (and state (claude-agent--process-state-session-key state))))
    (claude-agent--debug-log "Sending control response: %s"
                            (substring json 0 (min 200 (length json))))
    ;; Log to verbose buffer for debugging
    (when session-key
      (claude-agent--verbose-insert session-key
                                    (format ">>> STDIN (control_response): %s"
                                            (substring json 0 (min 300 (length json))))))
    ;; Untrack the request now that we're sending the response
    (claude-agent--untrack-control-request request-id)
    ;; Only send if process is still alive
    (if (process-live-p process)
        (progn
          ;; CRITICAL: Drain pending output before sending to prevent deadlock
          (claude-agent--drain-output process)
          (process-send-string process json))
      ;; Log when we can't send the response - this indicates the CLI died unexpectedly
      (claude-agent--debug-log "WARNING: Cannot send control_response - process is dead (id=%s)"
                               request-id)
      (when session-key
        (claude-agent--verbose-insert session-key
                                      (format ">>> ERROR: control_response DROPPED (process dead) id=%s"
                                              request-id)))
      (message "Claude: control_response dropped - CLI process died (request-id: %s)" request-id))))

(defun claude-agent--handle-control-request (process request)
  "Handle control_request from Claude CLI.

PROCESS: The Claude CLI process
REQUEST: Parsed control_request plist

Invokes appropriate Emacs hooks and sends control_response.
Tracks pending requests to prevent premature stdin close."
  (let* ((request-id (plist-get request :request_id))
         (request-data (plist-get request :request))
         (subtype (plist-get request-data :subtype)))

    (claude-agent--debug-log "Control request: %s (id=%s)" subtype request-id)

    ;; Track this request as pending (will be untracked when response is sent)
    (let ((owner (when-let* ((state (process-get process 'claude-agent-state)))
                   (claude-agent--process-state-request-id state))))
      (claude-agent--track-control-request request-id owner))

    (condition-case err
        (cond
         ;; Hook callback (reserved for future use)
         ((equal subtype "hook_callback")
          ;; Currently not used - return empty success response
          (claude-agent--send-control-response
           process request-id "success"
           (list)))

         ;; Tool permission - run permission functions hook
         ((equal subtype "can_use_tool")
          (let* ((tool-name (plist-get request-data :tool_name))
                 (tool-input (plist-get request-data :input))
                 (session-id (process-get process :session-id))
                 (org-section (process-get process :org-section))
                 (context (list :session-id session-id
                               :org-section org-section
                               :suggestions (plist-get request-data :permission_suggestions))))
            (claude-agent--debug-log "Running permission functions for %s (id=%s)"
                                     tool-name request-id)
            (let ((result (claude-agent--run-permission-functions
                          tool-name tool-input context)))
              (claude-agent--debug-log "Permission result for %s: %S (id=%s)"
                                       tool-name result request-id)
              (claude-agent--send-control-response
               process request-id "success"
               (list :behavior (or (plist-get result :behavior) "allow")
                     :updatedInput (or (plist-get result :updated-input) tool-input)
                     :message (plist-get result :message))))))

         (t
          ;; Unsupported subtype
          (error "Unsupported control request subtype: %s" subtype)))

      (error
       ;; Send error response
       (claude-agent--send-control-response
        process request-id "error"
        (list :error (error-message-string err)))))))
#+END_SRC

* Query Cancellation
:PROPERTIES:
:DEPENDS_ON: Data Structures, Process Management
:DEPENDED_BY: Core API, Activity Mode-Line, claude-org.org
:END:

This section provides query cancellation support by tracking active
queries and allowing them to be interrupted.

** Active Query Tracking

#+BEGIN_SRC elisp
(defvar claude-agent--active-queries (make-hash-table :test 'equal)
  "Map request-id to process state for cancellation support.")

(defvar claude-agent--request-counter 0
  "Counter for generating unique request IDs.")

(defun claude-agent--generate-request-id ()
  "Generate a unique request ID for query tracking."
  (format "req-%d-%.0f" (cl-incf claude-agent--request-counter) (* 1000 (float-time))))

(defun claude-agent--register-query (request-id state)
  "Register STATE for REQUEST-ID in active queries table."
  (when request-id
    (puthash request-id state claude-agent--active-queries)))

(defun claude-agent--unregister-query (request-id)
  "Remove REQUEST-ID from active queries table."
  (when request-id
    (remhash request-id claude-agent--active-queries)))

(defun claude-agent--get-active-query (request-id)
  "Get process state for REQUEST-ID."
  (gethash request-id claude-agent--active-queries))

(defcustom claude-agent-query-timeout 300
  "Seconds after which a query is considered stale and should be cleaned up.
Queries running longer than this are assumed to be zombie processes from
network issues or other failures. Set to nil to disable timeout cleanup."
  :type '(choice (integer :tag "Seconds")
                 (const :tag "Disable" nil))
  :group 'claude-agent)

(defun claude-agent--query-timed-out-p (state)
  "Return t if STATE represents a query that has timed out.
Uses `claude-agent-query-timeout' to determine staleness."
  (and claude-agent-query-timeout
       (claude-agent--process-state-p state)
       (claude-agent--process-state-start-time state)
       (> (- (float-time) (claude-agent--process-state-start-time state))
          claude-agent-query-timeout)))

(defun claude-agent--cleanup-stale-queries ()
  "Remove stale entries from active-queries hash table and active-states list.
Called after code reload to clean up entries with dead processes,
outdated structure, or queries that have exceeded timeout."
  (let ((stale-keys '())
        (timed-out-keys '()))
    (maphash
     (lambda (key state)
       (cond
        ;; Invalid state, dead process, or closed
        ((or (not (claude-agent--process-state-p state))
             (and (claude-agent--process-state-process state)
                  (not (process-live-p
                        (claude-agent--process-state-process state))))
             (claude-agent--process-state-closed state))
         (push key stale-keys))
        ;; Timed out - kill process and mark for removal
        ((claude-agent--query-timed-out-p state)
         (let ((proc (claude-agent--process-state-process state)))
           (when (and proc (process-live-p proc))
             (claude-agent--debug-log "Killing timed-out query: %s (age: %.0fs)"
                                      key
                                      (- (float-time)
                                         (claude-agent--process-state-start-time state)))
             (delete-process proc)))
         (push key timed-out-keys))))
     claude-agent--active-queries)
    ;; Remove all stale and timed-out entries from hash table
    (dolist (key (append stale-keys timed-out-keys))
      (remhash key claude-agent--active-queries))
    (when stale-keys
      (claude-agent--debug-log "Cleaned up %d stale queries" (length stale-keys)))
    (when timed-out-keys
      (claude-agent--debug-log "Cleaned up %d timed-out queries" (length timed-out-keys))))
  ;; Also clean up active-states list (may accumulate dead entries)
  (let ((initial-count (length claude-agent--active-states)))
    (setq claude-agent--active-states
          (cl-remove-if
           (lambda (state)
             (or (not (claude-agent--process-state-p state))
                 (claude-agent--process-state-closed state)
                 (let ((proc (claude-agent--process-state-process state)))
                   (and proc (not (process-live-p proc))))))
           claude-agent--active-states))
    (let ((removed (- initial-count (length claude-agent--active-states))))
      (when (> removed 0)
        (claude-agent--debug-log "Cleaned up %d stale states" removed)))))

;; Run cleanup on load
(claude-agent--cleanup-stale-queries)
#+END_SRC

** Cancellation Functions

#+BEGIN_SRC elisp
(defun claude-agent-cancel-query (request-id)
  "Cancel query by REQUEST-ID.
Always marks as closed and cancelled, then unregisters, even if process is dead.
Sets the cancelled flag to prevent automatic recovery.
Returns t if found and cancelled, nil if not found."
  (let ((state (claude-agent--get-active-query request-id)))
    (when state
      ;; Mark as closed and cancelled to prevent auto-recovery
      (setf (claude-agent--process-state-closed state) t)
      (setf (claude-agent--process-state-cancelled state) t)
      ;; Interrupt process if still alive
      (when-let ((process (claude-agent--process-state-process state)))
        (when (process-live-p process)
          (interrupt-process process)))
      ;; Always unregister from hash table
      (claude-agent--unregister-query request-id)
      (claude-agent--debug-log "Query cancelled: %s" request-id)
      t)))

(defun claude-agent-cancel-all-queries ()
  "Cancel all active queries."
  (maphash (lambda (id _state)
             (claude-agent-cancel-query id))
           claude-agent--active-queries))

(defun claude-agent--query-active-p (state)
  "Return t if STATE represents a truly active query.
Checks both closed state and process liveness."
  (and state
       (claude-agent--process-state-p state)
       (not (claude-agent--process-state-closed state))
       ;; Process must be nil (not yet started) or alive
       (let ((proc (claude-agent--process-state-process state)))
         (or (null proc) (process-live-p proc)))))

(defun claude-agent-cleanup ()
  "Clean up all Claude agent state.
Cancels active queries and removes stale entries."
  (interactive)
  (claude-agent-cancel-all-queries)
  (claude-agent--cleanup-stale-queries))

(cl-defun claude-agent-update-state-callbacks (state &key callback error-callback
                                                     token-callback complete-callback)
  "Update callbacks on STATE.
This is the public API for external modules to modify process state callbacks
without reaching into private struct internals.
Only provided non-nil keyword arguments are updated."
  (when (and state (claude-agent--process-state-p state))
    (when callback
      (setf (claude-agent--process-state-callback state) callback))
    (when error-callback
      (setf (claude-agent--process-state-error-callback state) error-callback))
    (when token-callback
      (setf (claude-agent--process-state-token-callback state) token-callback))
    (when complete-callback
      (setf (claude-agent--process-state-complete-callback state) complete-callback))))

(defun claude-agent-close-process-state (state)
  "Mark STATE as closed and unregister its query from active tracking.

that a process state is terminated without reaching into private internals."
  (when (and state (claude-agent--process-state-p state))
    (setf (claude-agent--process-state-closed state) t)
    (when-let* ((req-id (claude-agent--process-state-request-id state)))
      (claude-agent--unregister-query req-id))))

(defun claude-agent-queries-goto-source ()
  "Jump to the source location for query at point.
For navigable queries (shown as links), jumps to the source heading.
For non-navigable queries, jumps to the source buffer only."
  (interactive)
  (let* ((query-id (get-text-property (point) 'query-id))
         (navigable (get-text-property (point) 'navigable)))
    (if (null query-id)
        (message "No query on this line")
      (let ((state (gethash query-id claude-agent--active-queries)))
        (if (null state)
            (message "Query not found (may have completed)")
          (let ((buf (claude-agent--process-state-source-buffer state))
                (loc-fn (claude-agent--process-state-source-location-fn state))
                (loc-args (claude-agent--process-state-source-location-args state)))

             ((null buf)
              (message "No source buffer for this query"))
             ((not (buffer-live-p buf))
              (message "Source buffer no longer exists"))
             ((null loc-fn)
              ;; No location function - just jump to buffer
              (pop-to-buffer buf)
              (message "Jumped to buffer (no specific location)"))
             (t
              ;; Has location function - switch buffer and call it
              (pop-to-buffer buf)
              (if (apply loc-fn loc-args)
                  (message "Jumped to source")
                (message "Location function failed to find target"))))))))))

(defun claude-agent-usage-callback (status)
  "Process usage response and update mode-line. STATUS contains error info."
  (let ((error-msg nil))
    ;; Check for connection/HTTP errors
    (when (plist-get status :error)
      (setq error-msg (format "Connection error: %s" (plist-get status :error))))
    ;; Parse response
    (unless error-msg
      (goto-char (point-min))
      (when (re-search-forward "^$" nil t)
        (condition-case err
            (let* ((json (json-parse-buffer :object-type 'alist))
                   (api-error (alist-get 'error json))
                   (api-detail (alist-get 'detail json)))
              (cond
               ;; API returned error
               ((or api-error api-detail)
                (setq error-msg (or api-error api-detail "API error")))
               ;; Success - parse usage data
               (t
                (let* ((five-hr-data (alist-get 'five_hour json))
                       (seven-day-data (alist-get 'seven_day json))
                       (sonnet-data (alist-get 'seven_day_sonnet json))
                       (five-hr (alist-get 'utilization five-hr-data))
                       (seven-day (alist-get 'utilization seven-day-data))
                       (sonnet (alist-get 'utilization sonnet-data))
                       (five-reset (alist-get 'resets_at five-hr-data))
                       (seven-reset (alist-get 'resets_at seven-day-data))
                       (sonnet-reset (alist-get 'resets_at sonnet-data))
                       (display (format " [C:%d|%d|%d]"
                                        (truncate (or five-hr 0))
                                        (truncate (or seven-day 0))
                                        (truncate (or sonnet 0))))
                       (tooltip (format "Claude Usage\n─────────────────────────\n5-hour:       %3d%% (resets in %s)\n7-day:        %3d%% (resets in %s)\n7-day Sonnet: %3d%% (resets in %s)\n\nClick to refresh"
                                        (truncate (or five-hr 0))
                                        (claude-agent-usage-format-reset-time five-reset)
                                        (truncate (or seven-day 0))
                                        (claude-agent-usage-format-reset-time seven-reset)
                                        (truncate (or sonnet 0))
                                        (claude-agent-usage-format-reset-time sonnet-reset))))
                  (setq claude-agent-usage-data json)
                  (setq claude-agent-usage-string
                        (propertize display
                                    'help-echo tooltip
                                    'mouse-face 'mode-line-highlight
                                    'local-map (let ((map (make-sparse-keymap)))
                                                 (define-key map [mode-line mouse-1] #'claude-agent-usage-fetch)
                                                 map)))))))
          (error (setq error-msg (format "Parse error: %s" (error-message-string err)))))))
    ;; Handle errors
    (when error-msg
      (setq claude-agent-usage-string
            (propertize " [C:ERR]"
                        'face '(:weight bold :underline t)
                        'help-echo (format "Claude usage error:\n%s\n\nClick to retry" error-msg)
                        'mouse-face 'mode-line-highlight
                        'local-map (let ((map (make-sparse-keymap)))
                                     (define-key map [mode-line mouse-1] #'claude-agent-usage-fetch)
                                     map))))
    (force-mode-line-update t))
  (kill-buffer))

(defun claude-agent--verbose-insert (session-key text)
  "Insert TEXT into verbose buffer for SESSION-KEY.
When `claude-agent-verbose-buffer-max-size' is set, trims oldest content
if buffer exceeds the limit after insertion."
  (when-let ((buf (gethash session-key claude-agent--session-verbose-buffers)))
    (when (buffer-live-p buf)
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          ;; Trim if over max size before inserting
          (when (and claude-agent-verbose-buffer-max-size
                     (> (buffer-size) claude-agent-verbose-buffer-max-size))
            (let ((trim-to (/ claude-agent-verbose-buffer-max-size 2)))
              (goto-char (+ (point-min) (- (buffer-size) trim-to)))
              ;; Advance to next line boundary for clean trim
              (forward-line 1)
              (delete-region (point-min) (point))
              (goto-char (point-min))
              (insert "[...verbose output trimmed...]\n")))
          (goto-char (point-max))
          (insert text)
          (unless (string-suffix-p "\n" text)
            (insert "\n")))))))

(defun claude-agent--get-session-verbose-buffer (session-key)
  "Get or create verbose buffer for SESSION-KEY."
  (or (let ((buf (gethash session-key claude-agent--session-verbose-buffers)))
        (and buf (buffer-live-p buf) buf))
      (let ((buf (generate-new-buffer
                  (format "*Claude Session[%s]*"
                          (if (string-match "::\\(.+\\)$" session-key)
                              (match-string 1 session-key)
                            (file-name-nondirectory session-key))))))
        (with-current-buffer buf
          (claude-agent-verbose-mode)
          (let ((inhibit-read-only t))
            (insert (format "═══ Claude Session ═══\n"))
            (insert (format "Session: %s\n" session-key))
            (insert (format "Started: %s\n\n" (format-time-string "%Y-%m-%d %H:%M:%S")))))
        (puthash session-key buf claude-agent--session-verbose-buffers)
        buf)))

(defun claude-agent--verbose-insert-query-header (session-key prompt &optional cli-path args cwd env-vars)
  "Insert query header in verbose buffer for SESSION-KEY with PROMPT.
CLI-PATH is the path to claude CLI, ARGS are command line arguments,
CWD is working directory, ENV-VARS is alist of environment variables."
  (when-let ((buf (gethash session-key claude-agent--session-verbose-buffers)))
    (when (buffer-live-p buf)
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          (goto-char (point-max))
          (insert "\n")
          (insert "════════════════════════════════════════════════════════════════\n")
          (insert (format "▸ Query [%s]\n" (format-time-string "%H:%M:%S")))
          ;; Show CLI command if provided
          (when (and cli-path args)
            (insert "\n")
            (insert (propertize "Command:\n" 'face 'font-lock-keyword-face))
            (insert (format "  %s \\\n" cli-path))
            (let ((arg-pairs nil)
                  (i 0))
              ;; Group args into option-value pairs for better display
              (while (< i (length args))
                (let ((arg (nth i args)))
                  (if (string-prefix-p "-" arg)
                      (if (and (< (1+ i) (length args))
                               (not (string-prefix-p "-" (nth (1+ i) args))))
                          (progn
                            (push (cons arg (nth (1+ i) args)) arg-pairs)
                            (setq i (+ i 2)))
                        (push (cons arg nil) arg-pairs)
                        (setq i (1+ i)))
                    (push (cons arg nil) arg-pairs)
                    (setq i (1+ i)))))
              (dolist (pair (nreverse arg-pairs))
                (let ((opt (car pair))
                      (val (cdr pair)))
                  (if val
                      ;; Show full values (including system prompts) for debugging
                      (insert (format "    %s %s \\\n"
                                      (propertize opt 'face 'font-lock-constant-face)
                                      val))
                    (insert (format "    %s \\\n"
                                    (propertize opt 'face 'font-lock-constant-face))))))))
          ;; Show working directory
          (when cwd
            (insert (format "\n%s %s\n"
                            (propertize "Working directory:" 'face 'font-lock-keyword-face)
                            cwd)))
          ;; Show environment variables if any
          (when env-vars
            (insert (format "\n%s\n" (propertize "Environment:" 'face 'font-lock-keyword-face)))
            (dolist (pair env-vars)
              (let* ((name (car pair))
                     (value (cdr pair))
                     ;; Mask sensitive values (tokens/keys)
                     (display-value (if (string-match-p "TOKEN\\|KEY\\|SECRET\\|PASSWORD" name)
                                        (concat (substring value 0 (min 8 (length value))) "...")
                                      value)))
                (insert (format "  %s=%s\n"
                                (propertize name 'face 'font-lock-variable-name-face)
                                display-value)))))
          (insert "\n")
          (insert (propertize "Prompt:\n" 'face 'font-lock-keyword-face))
          (insert prompt)
          (insert "\n\n"))))))

(defun claude-agent--verbose-format-tool-input (input)
  "Format tool INPUT for display, handling different types."
  (cond
   ((null input) "")
   ((stringp input) input)
   ((listp input)
    ;; Convert plist or alist to readable format
    (let ((json-str (condition-case nil
                        (json-encode input)
                      (error (format "%S" input)))))
      json-str))
   (t (format "%S" input))))

(defun claude-agent--verbose-format-content-block (block)
  "Format a single content BLOCK for verbose display."
  (cond
   ;; Tool use block - show full tool call with input
   ((claude-agent-tool-use-block-p block)
    (let* ((name (claude-agent-tool-use-block-name block))
           (input (claude-agent-tool-use-block-input block))
           (input-str (claude-agent--verbose-format-tool-input input)))
      (concat
       (format "\n%s %s\n"
               (propertize "▶ Tool:" 'face 'font-lock-keyword-face)
               (propertize name 'face 'font-lock-function-name-face))
       (when (and input-str (not (string-empty-p input-str)))
         (format "%s\n%s\n"
                 (propertize "  Input:" 'face 'font-lock-comment-face)
                 input-str)))))

   ;; Tool result block - show content
   ((claude-agent-tool-result-block-p block)
    (let* ((is-error (claude-agent-tool-result-block-is-error block))
           (content (claude-agent-tool-result-block-content block))
           (content-str (if (stringp content)
                            content
                          (claude-agent--verbose-format-tool-input content))))
      (concat
       (format "%s %s\n"
               (propertize "◀ Result:" 'face 'font-lock-keyword-face)
               (if is-error
                   (propertize "ERROR" 'face 'error)
                 (propertize "OK" 'face 'success)))
       (when (and content-str (not (string-empty-p content-str)))
         (format "%s\n" content-str)))))

   ;; Thinking block - show actual thinking content
   ((claude-agent-thinking-block-p block)
    (let ((thinking (claude-agent-thinking-block-thinking block)))
      (concat
       (format "\n%s\n" (propertize "💭 Thinking:" 'face 'font-lock-comment-face))
       (when (and thinking (not (string-empty-p thinking)))
         (format "%s\n" (propertize thinking 'face 'font-lock-doc-face))))))

   ;; Text block
   ((claude-agent-text-block-p block)
    (claude-agent-text-block-text block))

   (t nil)))

(defun claude-agent--verbose-format-message (msg)
  "Format MSG for verbose buffer display."
  (cond
   ;; Assistant message - iterate through content blocks
   ((claude-agent-assistant-message-p msg)
    (let ((content (claude-agent-assistant-message-content msg))
          (parts nil))
      (claude-agent--debug-log "Assistant content blocks: %d items" (length content))
      (dolist (block content)
        (claude-agent--debug-log "  Block type: %s"
                                 (cond
                                  ((claude-agent-tool-use-block-p block) "tool_use")
                                  ((claude-agent-thinking-block-p block) "thinking")
                                  ((claude-agent-text-block-p block) "text")
                                  ((claude-agent-tool-result-block-p block) "tool_result")
                                  (t (format "unknown: %S" (type-of block)))))
        (when-let ((formatted (claude-agent--verbose-format-content-block block)))
          (push formatted parts)))
      (when parts
        (mapconcat #'identity (nreverse parts) ""))))

   ;; User message - show tool results if any
   ((claude-agent-user-message-p msg)
    (let ((content (claude-agent-user-message-content msg)))
      (cond
       ;; Content is a list of blocks (tool results)
       ((and (listp content) (listp (car content)))
        (let ((parts nil))
          (dolist (block content)
            (when-let ((formatted (claude-agent--verbose-format-content-block
                                   (claude-agent--parse-content-block block))))
              (push formatted parts)))
          (when parts
            (mapconcat #'identity (nreverse parts) ""))))
       ;; Content is just a string
       ((stringp content)
        nil)  ; Don't show user input (already shown in header)
       (t nil))))

   ;; System message
   ((claude-agent-system-message-p msg)
    (let* ((subtype (claude-agent-system-message-subtype msg))
           (data (claude-agent-system-message-data msg))
           (message-text (plist-get data :message)))
      (cond
       ;; Show error messages with message text
       ((string= subtype "error")
        (format "❌ Error: %s\n" (or message-text subtype)))
       ;; Show init messages
       ((string= subtype "init")
        (format "ℹ️ %s\n" (or message-text "Initializing...")))
       ;; Show any message that has text (like "Compacting conversation...")
       (message-text
        (format "ℹ️ %s\n" message-text))
       ;; For other subtypes, just show the subtype
       (subtype
        (format "ℹ️ %s\n" subtype))
       (t nil))))

   ;; Result message
   ((claude-agent-result-message-p msg)
    (format "\n%s\n"
            (propertize
             (format "════ Complete: %.1fs | $%.4f ════"
                     (/ (or (claude-agent-result-message-duration-ms msg) 0) 1000.0)
                     (or (claude-agent-result-message-total-cost-usd msg) 0))
             'face 'font-lock-comment-face)))

   (t nil)))

(defun claude-agent--verbose-write-message (session-key msg)
  "Write MSG to verbose buffer for SESSION-KEY."
  (claude-agent--debug-log "Verbose write: session=%s msg-type=%s"
                           session-key
                           (cond
                            ((claude-agent-assistant-message-p msg) "assistant")
                            ((claude-agent-user-message-p msg) "user")
                            ((claude-agent-system-message-p msg) "system")
                            ((claude-agent-result-message-p msg) "result")

  (when session-key
    (when-let* ((buf (gethash session-key claude-agent--session-verbose-buffers))
                (formatted (claude-agent--verbose-format-message msg)))
      (when (and buf (buffer-live-p buf) (not (string-empty-p formatted)))
        (with-current-buffer buf
          (let ((inhibit-read-only t))
            (goto-char (point-max))
            (insert formatted)))))))

    (pcase type

                             rear-nonsticky t
                             read-only t)))
    (when proc
      (set-marker (process-mark proc) (point)))))

(defun claude-agent-chat--send-input (input)
  "Send INPUT to Claude via the backend."
  (when (and input (not (string-empty-p (string-trim input))))
    (if claude-agent-chat--waiting
        (message "Still waiting for Claude's response...")
      (setq claude-agent-chat--waiting t)
      (setq claude-agent-chat--current-response "")
      (claude-agent-chat--insert-response-start)
      (claude-agent-chat--insert "[Waiting...]" 'claude-agent-chat-thinking-face)
      ;; Ensure backend exists
      (unless claude-agent-chat--backend
        (setq claude-agent-chat--backend (claude-agent-chat--make-backend)))
      ;; Build options with session continuation if we have a session
      (let* ((base-opts (or claude-agent-chat--options (claude-agent-options)))
             (opts (if claude-agent-chat--session-id
                       (plist-put (plist-put (copy-sequence base-opts)
                                             :continue-conversation t)
                                  :resume claude-agent-chat--session-id)
                     base-opts))
             (opts (if claude-agent-chat--system-prompt
                       (plist-put (copy-sequence opts)
                                  :system-prompt claude-agent-chat--system-prompt)
                     opts))
             (opts (if claude-agent-chat--mcp-config
                       (plist-put (copy-sequence opts)
                                  :mcp-config claude-agent-chat--mcp-config)
                     opts))
             (system-reminder (when claude-agent-chat-include-ide-context
                                (claude-agent-get-system-reminder)))
             (actual-prompt (if (and system-reminder
                                     (not (string-empty-p system-reminder)))
                                (concat system-reminder "\n" input)
                              input)))
        ;; Build callbacks plist for backend-query
        (lexical-let ((chat-buf (current-buffer)))
          (let ((callbacks
                 (list
                  :on-token (lambda (token)
                              (when (buffer-live-p chat-buf)
                                (with-current-buffer chat-buf
                                  (claude-agent-chat--handle-token token))))
                  :on-message (lambda (msg)
                                (when (buffer-live-p chat-buf)
                                  (with-current-buffer chat-buf
                                    (claude-agent-chat--handle-message msg))))
                  :on-error (lambda (err)
                              (when (buffer-live-p chat-buf)
                                (with-current-buffer chat-buf
                                  (claude-agent-chat--handle-error err))))
                  :on-complete (lambda (result)
                                 (when (buffer-live-p chat-buf)
                                   (with-current-buffer chat-buf
                                     (claude-agent-chat--handle-complete result)))))))
            (setq claude-agent-chat--query-handle
                  (claude-agent-backend-query
                   claude-agent-chat--backend
                   actual-prompt
                   callbacks
                   :options opts))))))))

(defun claude-agent-chat-interrupt ()
  "Interrupt the current Claude request via the backend."
  (interactive)
  (when claude-agent-chat--backend
    (claude-agent-backend-cancel claude-agent-chat--backend
                                 claude-agent-chat--query-handle))
  (setq claude-agent-chat--query-handle nil)
  (setq claude-agent-chat--waiting nil)
  (claude-agent-chat--insert "\n[Interrupted]\n" 'claude-agent-chat-error-face)
  (claude-agent-chat--show-prompt))

(defun claude-agent-chat-clear ()
  "Clear the chat buffer and reset the session."
  (interactive)
  (when (yes-or-no-p "Clear chat history and start new session? ")
    ;; Clean up backend
    (when claude-agent-chat--backend
      (claude-agent-backend-cleanup claude-agent-chat--backend)
      (setq claude-agent-chat--backend nil)
      (setq claude-agent-chat--query-handle nil))
    (let ((inhibit-read-only t))
      (erase-buffer)
      (setq claude-agent-chat--session-id nil)
      (setq claude-agent-chat--total-cost 0.0)
      (setq claude-agent-chat--waiting nil)
      (claude-agent-chat--initialize-buffer))))

(defun claude-agent-chat-new-session ()
  "Start a new conversation session without clearing history.
For json-stream backend, just resets session-id.
For claude-cli backend, cleans up terminal so a fresh one is created."

  (if (and claude-agent-chat--backend
           (claude-agent-claude-backend-p claude-agent-chat--backend))

        (claude-agent-backend-cleanup claude-agent-chat--backend)

        (setq claude-agent-chat--query-handle nil))
    ;; JSON stream: just nil session-id for fresh conversation
    (setq claude-agent-chat--session-id nil))
  (claude-agent-chat--insert "\n--- New Session ---\n" 'claude-agent-chat-info-face)
  (message "Started new session"))

#+BEGIN_SRC elisp
(defun claude-agent--format-messages-for-title (messages)
  "Format MESSAGES for title generation prompt.
MESSAGES is a list of (role . content) pairs."
  (let ((formatted nil)
        (count 0))
    (dolist (msg (reverse messages))
      (when (< count claude-agent-title-max-messages)
        (let ((role (car msg))
              (content (cdr msg)))
          ;; Truncate long content
          (when (> (length content) 200)
            (setq content (concat (substring content 0 197) "...")))
          (push (format "%s: %s" (capitalize (symbol-name role)) content) formatted)
          (cl-incf count))))
    (mapconcat #'identity formatted "\n\n")))

(cl-defun claude-agent-query-accumulate (prompt &key options on-result on-error
                                              session-key)
  "Run a query that accumulates all response text, then calls ON-RESULT.
This is a convenience wrapper around `claude-agent-query' for fire-and-forget
queries that just need the full text response (e.g., title generation,
prompt refinement).

PROMPT is the query prompt.
OPTIONS is an options plist (from `claude-agent-options').
ON-RESULT is called with the accumulated text string (trimmed).
ON-ERROR is called on error.
SESSION-KEY is an optional session key."
  (lexical-let ((collected-text "")
                (on-result-cb on-result))
    (claude-agent-query
     prompt
     :options options
     :session-key session-key
     :on-message (lambda (msg)
                   (when (claude-agent-assistant-message-p msg)
                     (when-let ((text (claude-agent-extract-text msg)))
                       (setq collected-text (concat collected-text text)))))
     :on-error on-error
     :on-complete (lambda (_result)
                    (when on-result-cb
                      (funcall on-result-cb (string-trim collected-text)))))))

(cl-defun claude-agent-generate-title (messages &key on-title on-error)
  "Generate a short title for conversation MESSAGES.

MESSAGES is a list of (role . content) pairs, e.g.:
  \\='((user . \"How do I sort a list?\")
    (assistant . \"You can use the sort function...\"))

ON-TITLE is called with the generated title string.
  Function signature: (lambda (title) ...)

ON-ERROR is called if an error occurs.
  Function signature: (lambda (error) ...)

Uses Haiku model for low cost and fast response.
Each call uses a fresh session (no conversation continuity).

Example:
  (claude-agent-generate-title
   \\='((user . \"Help me write a Python script\")
     (assistant . \"Sure, what would you like the script to do?\")
     (user . \"Parse JSON files\"))
   :on-title (lambda (title) (message \"Title: %s\" title)))"
  (let* ((formatted-messages (claude-agent--format-messages-for-title messages))
         (prompt (format "Generate a short title (%d chars max) for this conversation:

%s

IMPORTANT: Output ONLY the title text. No quotes, no explanation, no response to the conversation.
Example good outputs: \"Python JSON Parsing\" or \"Fix Login Bug\" or \"Database Schema Design\"
Example bad outputs: \"I'll help you with...\" or \"Here's a title:\" or \"The title is:\""
                         claude-agent-title-max-chars
                         formatted-messages))
         (system-prompt (format "You are a title generator. Your ONLY job is to output a short descriptive title.
NEVER respond to or engage with the conversation content.
NEVER start with \"I'll\", \"I can\", \"Here's\", \"The title\", or similar phrases.
Output ONLY the title text, nothing else. Maximum %d characters."
                                claude-agent-title-max-chars))
         (options (claude-agent-options
                   :model claude-agent-title-model
                   :system-prompt system-prompt
                   :disallowed-tools '("*")  ; No tools needed for titling
                   :max-turns 1)))
    (lexical-let ((on-title-cb on-title)
                  (max-chars claude-agent-title-max-chars))
      (claude-agent-query-accumulate
       prompt
       :options options
       :on-error on-error
       :on-result
       (lambda (text)
         (when on-title-cb
           (let ((title text))
             ;; Remove surrounding quotes if present
             (when (and (>= (length title) 2)
                        (string-match-p "^[\"'].*[\"']$" title))
               (setq title (substring title 1 -1)))
             ;; Remove common bad prefixes
             (when (string-match "^\\(?:Title:\\s-*\\|Here's\\s-*\\(?:a\\|the\\)\\s-*title:\\s-*\\)" title)
               (setq title (substring title (match-end 0))))
             ;; If title looks like a response, skip it
             (unless (string-match-p "^I'\\(?:ll\\|m\\|ve\\)\\s-\\|^I\\s-+\\(?:can\\|will\\|would\\)" title)
               ;; Truncate if too long
               (when (> (length title) max-chars)
                 (setq title (concat (substring title 0 (- max-chars 3)) "...")))
               (funcall on-title-cb title)))))))))

(defun claude-agent-generate-title-from-text (text &optional callback)

TEXT is a string containing the conversation or content to title.

This is a convenience wrapper for simple use cases."

     :on-title cb

                 (message "Title generation error: %s" err)))))
