** Overview
:PROPERTIES:
:CUSTOM_ID: pi-ui-overview
:END:

Phase 1 of the *Pi TUI → Emacs migration* described in =draft.org=.
Gives the user every routine Pi-TUI control reachable from one
transient submenu — =C-c C-/ p= — so day-to-day Pi usage never
requires opening Pi's terminal UI.

Phase 1 is *RPC-only*: each command maps to one or more existing
=pi --mode rpc= commands.  Multimodal input, tool-call inline
rendering, session-tree browsing, and widget overlays land in
subsequent phases (=draft.org= §pi-mig-phase-2/3/4).

(defcustom claude-agent-pi-ui-rpc-timeout 6.0
  "Seconds to wait for a synchronous RPC response.
Used by `claude-agent-pi-ui--call' for menu commands like
`cycle_model'.  Most Pi commands respond in well under a second; the
generous timeout is for the rare slow-start case (cold provider
handshake).  Raise this if you frequently hit `Pi RPC timeout' errors."
  :type 'number
  :group 'claude-agent)

** Backend lookup + synchronous RPC call
:PROPERTIES:
:CUSTOM_ID: pi-ui-helpers
:END:

#+BEGIN_SRC elisp
(defun claude-agent-pi-ui--current-backend ()
  "Return the Pi backend for the session at point, or signal `user-error'."
  (let* ((sk (and (fboundp 'code-agent-org--current-session-key)
                  (code-agent-org--current-session-key)))
         (b (and sk (fboundp 'code-agent-org--session-get)
                 (code-agent-org--session-get sk :backend))))
    (cond
     ((null sk)
      (user-error "claude-agent-pi-ui: not in a code-agent-org session"))
     ((null b)
      (user-error
       "claude-agent-pi-ui: no active backend.  Run C-c C-c on an ai block first to spawn pi"))
     ((not (claude-agent-pi-backend-p b))
      (user-error
       "claude-agent-pi-ui: active backend is %s, not Pi.  Set :CLAUDE_BACKEND: pi on this workspace"
       (type-of b)))
     (t b))))

(defun claude-agent-pi-ui--call (backend command-alist &optional timeout)
  "Send COMMAND-ALIST to BACKEND with auto-generated id and wait for response.
Returns the parsed response object (alist).  Signals `user-error' on timeout."
  (let* ((id (claude-agent-pi--gen-id backend "ui"))
         (cmd (cons (cons 'id id) command-alist))
         (result nil)
         (limit (+ (float-time) (or timeout claude-agent-pi-ui-rpc-timeout))))
    (puthash id (list :on-response (lambda (obj) (setq result obj)))
             (claude-agent-pi-backend-pending-by-id backend))
    (claude-agent-pi--send backend cmd)
    (while (and (null result) (< (float-time) limit))
      (accept-process-output nil 0.05))
    (when (null result)
      (remhash id (claude-agent-pi-backend-pending-by-id backend))
      (user-error "Pi RPC timeout waiting for response to %s"
                  (or (cdr (assq 'type command-alist)) "<unknown>")))
    result))

(defun claude-agent-pi-ui--unwrap (response)
  "Extract data from an RPC RESPONSE alist; signal on success=false."
  (if (eq (map-elt response 'success) t)
      (map-elt response 'data)
    (user-error "Pi RPC %s failed: %s"
                (map-elt response 'command)
                (or (map-elt response 'error) "no detail"))))

(defun claude-agent-pi-ui-cycle-model ()
  "Cycle to the next model in the user's `models' settings list."
  (interactive)
  (let* ((b (claude-agent-pi-ui--current-backend))
         (resp (claude-agent-pi-ui--call b '((type . "cycle_model"))))
         (data (claude-agent-pi-ui--unwrap resp)))
    (if data
        (message "Pi model: %s (thinking %s)"
                 (map-elt (map-elt data 'model) 'id)
                 (map-elt data 'thinkingLevel))
      (message "Pi: no scoped models configured (use `pi config' to add)"))))

(defun claude-agent-pi-ui-pick-model ()
  "Prompt for a model via `completing-read', then set it on Pi."
  (interactive)
  (let* ((b (claude-agent-pi-ui--current-backend))
         (resp (claude-agent-pi-ui--call b '((type . "get_available_models"))))
         (data (claude-agent-pi-ui--unwrap resp))
         (models (and data (map-elt data 'models)))
         (choices
          (mapcar (lambda (m)
                    (let ((provider (map-elt m 'provider))
                          (id (map-elt m 'id))
                          (name (map-elt m 'name)))
                      (cons (format "%s/%s — %s" provider id name)
                            (cons provider id))))
                  (if (sequencep models) (append models nil) nil))))
    (unless choices (user-error "Pi reported zero available models"))
    (let* ((picked (completing-read "Pi model: " (mapcar #'car choices) nil t))
           (entry (cdr (assoc picked choices)))
           (provider (car entry))
           (id (cdr entry)))
      (claude-agent-pi-ui--unwrap
       (claude-agent-pi-ui--call b `((type . "set_model")
                                     (provider . ,provider)
                                     (modelId . ,id))))
      (message "Pi model set to %s/%s" provider id))))

*** Session resume + tree (Phase 3)
:PROPERTIES:
:CUSTOM_ID: pi-ui-cmd-resume
:END:

Pi stores sessions as JSONL files under =~/.pi/agent/sessions/<cwd-hash>/=
where the cwd hash is the absolute path with =/= → =-= substitution.
Each file's first line is a =SessionHeader= entry containing the id +
parent-session pointer + creation timestamp.

The resume picker scans the current workspace's session directory,
shows them via =completing-read=, and dispatches =switch_session=.
=session-tree= is the BFS over parent links, rendered into a
read-only =*Pi Sessions: <cwd>*= buffer with =org-mode= folding.

#+BEGIN_SRC elisp
(defcustom claude-agent-pi-ui-sessions-dir
  (expand-file-name "~/.pi/agent/sessions/")
  "Root directory under which Pi stores per-cwd session subdirs."
  :type 'directory
  :group 'claude-agent)

(defun claude-agent-pi-ui--cwd-hash (cwd)
  "Return Pi's filesystem-encoded cwd subdir name for CWD.
Pi replaces =/= with =-= and wraps with dashes; e.g.
=/Users/jt/proj/= → =--Users-jt-proj--=."
  (let ((normalised (expand-file-name cwd)))
    (concat "-"
            (replace-regexp-in-string "/" "-" normalised)
            "-")))

(defun claude-agent-pi-ui--list-session-files (&optional cwd)
  "Return absolute paths of all session JSONL files for CWD (default
=default-directory=).  Newest first."
  (let* ((root (file-name-as-directory claude-agent-pi-ui-sessions-dir))
         (sub (claude-agent-pi-ui--cwd-hash (or cwd default-directory)))
         (dir (expand-file-name sub root)))
    (when (file-directory-p dir)
      (sort (directory-files dir t "\\.jsonl\\'" t)
            (lambda (a b)
              (time-less-p (nth 5 (file-attributes b))
                            (nth 5 (file-attributes a))))))))

(defun claude-agent-pi-ui--read-session-header (path)
  "Parse the first JSONL line of PATH; return the header alist or nil."
  (with-temp-buffer
    (condition-case nil
        (progn
          (insert-file-contents path nil 0 4096)
          (goto-char (point-min))
          (let ((line (buffer-substring-no-properties
                       (point) (line-end-position))))
            (json-parse-string line :object-type 'alist :null-object nil :false-object nil)))
      (error nil))))

(defun claude-agent-pi-ui-resume-session ()
  "Pick + resume a Pi session for the current workspace.
Scans `claude-agent-pi-ui-sessions-dir' for the cwd subdir, parses
each JSONL header to extract id + name + timestamp, presents via
`completing-read', dispatches `switch_session' on the chosen path."
  (interactive)
  (let* ((b (claude-agent-pi-ui--current-backend))
         (files (claude-agent-pi-ui--list-session-files default-directory))
         (entries
          (mapcar
           (lambda (path)
             (let* ((hdr (claude-agent-pi-ui--read-session-header path))
                    (id (and hdr (map-elt hdr 'id)))
                    (ts (and hdr (map-elt hdr 'timestamp)))
                    (label (format "%s  %s"
                                    (or ts "(no timestamp)")
                                    (or id (file-name-base path)))))
               (cons label path)))
           files)))
    (unless entries
      (user-error "No Pi sessions found under %s for cwd %s"
                  claude-agent-pi-ui-sessions-dir default-directory))
    (let* ((pick (completing-read "Resume Pi session: "
                                   (mapcar #'car entries) nil t))
           (path (cdr (assoc pick entries))))
      (claude-agent-pi-ui--unwrap
       (claude-agent-pi-ui--call b `((type . "switch_session")
                                      (sessionPath . ,path))
                                  30.0))
      (message "Pi: switched to %s" (file-name-base path)))))

(defun claude-agent-pi-ui-session-tree ()
  "Open a buffer showing the session DAG for the current workspace.
Each session row links to its parent via the JSONL header's
=parentSession= field; we render the tree as indented org headings
in a read-only =*Pi Sessions: <cwd>*= buffer."
  (interactive)
  (let* ((files (claude-agent-pi-ui--list-session-files default-directory))
         (by-id (make-hash-table :test 'equal))
         (children (make-hash-table :test 'equal))
         (roots nil))
    (unless files
      (user-error "No Pi sessions found for cwd %s" default-directory))
    ;; Index every session by id; build parent → children index.
    (dolist (path files)
      (let* ((hdr (claude-agent-pi-ui--read-session-header path))
             (id (and hdr (map-elt hdr 'id)))
             (parent (and hdr (map-elt hdr 'parentSession))))
        (when id
          (puthash id (list :path path :hdr hdr) by-id)
          (if parent
              (push id (gethash parent children))
            (push id roots)))))
    (let ((buf (get-buffer-create
                (format "*Pi Sessions: %s*"
                        (abbreviate-file-name default-directory)))))
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          (erase-buffer)
          (org-mode)
          (insert (format "#+TITLE: Pi sessions for %s\n\n"
                          (abbreviate-file-name default-directory)))
          (claude-agent-pi-ui--render-session-subtree roots by-id children 1))
        (goto-char (point-min))
        (setq buffer-read-only t))
      (pop-to-buffer buf))))

(defun claude-agent-pi-ui--render-session-subtree (ids by-id children depth)
  "Render the IDS at DEPTH as org headings; recurse into children."
  (dolist (id (nreverse ids))
    (let* ((entry (gethash id by-id))
           (hdr (and entry (plist-get entry :hdr)))
           (ts (or (and hdr (map-elt hdr 'timestamp)) ""))
           (path (and entry (plist-get entry :path))))
      (insert (format "%s %s — %s\n"
                       (make-string (1+ depth) ?\*)
                       (or id "(unknown)")
                       ts))
      (when path
        (insert (format "  :PROPERTIES:\n  :PATH: %s\n  :END:\n" path)))
      (let ((kids (gethash id children)))
        (when kids
          (claude-agent-pi-ui--render-session-subtree
           kids by-id children (1+ depth)))))))
#+END_SRC

** Extension UI handlers
:PROPERTIES:
:CUSTOM_ID: pi-ui-extension-ui
:END:

Replaces =claude-agent-pi--default-ui-handler= (auto-cancel stub) with
real implementations.

#+BEGIN_SRC elisp
(defun claude-agent-pi-ui--respond (backend id payload)
  "Send `extension_ui_response' for ID with PAYLOAD plist members."
  (claude-agent-pi--send
   backend
   `((type . "extension_ui_response")
     (id . ,id)
     ,@payload)))

(defun claude-agent-pi-ui--extension-ui-handler (backend obj)
  "Real handler for Pi `extension_ui_request' events."
  (let* ((id (map-elt obj 'id))
         (method (map-elt obj 'method)))
    (cond
     ((null id)
      (when (equal method "notify")
        (message "Pi extension notify: %s" (map-elt obj 'message))))
     ((equal method "select")
      (let* ((title (map-elt obj 'title))
             (opts (map-elt obj 'options))
             (opts-list (if (sequencep opts) (append opts nil) nil))
             (pick (and opts-list
                        (completing-read (format "%s: " title) opts-list nil t))))
        (if pick
            (claude-agent-pi-ui--respond backend id `((value . ,pick)))
          (claude-agent-pi-ui--respond backend id '((cancelled . t))))))
     ((equal method "confirm")
      (let* ((title (map-elt obj 'title))
             (msg (map-elt obj 'message))
             (answer (yes-or-no-p (format "%s: %s " title msg))))
        (claude-agent-pi-ui--respond backend id `((confirmed . ,(if answer t :json-false))))))
     ((equal method "input")
      (let* ((title (map-elt obj 'title))
             (placeholder (or (map-elt obj 'placeholder) ""))
             (val (read-string (format "%s: " title) placeholder)))
        (claude-agent-pi-ui--respond backend id `((value . ,val)))))
     ((equal method "editor")
      (let* ((title (map-elt obj 'title))
             (prefill (or (map-elt obj 'prefill) ""))
             (val (read-string (format "%s (one line, RET to submit): " title)
                                prefill)))
        (claude-agent-pi-ui--respond backend id `((value . ,val)))))
     ((equal method "notify")
      (let ((msg (map-elt obj 'message))
            (kind (or (map-elt obj 'notifyType) "info")))
        (cond
         ((equal kind "error")
          (display-warning 'claude-agent-pi msg :error))
         ((equal kind "warning")
          (display-warning 'claude-agent-pi msg :warning))
         (t (message "Pi: %s" msg)))
        (claude-agent-pi-ui--respond backend id '((value . "")))))
     ((equal method "setStatus")
      ;; Track per-key extension status; surface to user as a message
      ;; so a "loaded N rules" notification doesn't vanish into the
      ;; verbose buffer.
      (let ((status-key (map-elt obj 'statusKey))
            (status-text (map-elt obj 'statusText)))
        (when (and status-text (stringp status-text)
                   (> (length status-text) 0))
          (message "Pi ext [%s]: %s" status-key status-text))
        (claude-agent-pi--verbose-log
         backend (format "ext setStatus %s = %S" status-key status-text)))
      (claude-agent-pi-ui--respond backend id '((value . ""))))
     ((equal method "setWidget")
      ;; Phase 4 widget surface: write the widget lines as a header
      ;; section into the verbose buffer.  A real overlay zone
      ;; above/below the ai block is future work; the data is
      ;; preserved + visible here so an extension can experiment.
      (let* ((wkey (map-elt obj 'widgetKey))
             (lines (map-elt obj 'widgetLines))
             (placement (or (map-elt obj 'widgetPlacement) "above"))
             (lines-str (cond
                         ((null lines) "(cleared)")
                         ((sequencep lines)
                          (mapconcat #'identity (append lines nil) "\n  "))
                         (t (format "%S" lines)))))
        (claude-agent-pi--verbose-log
         backend
         (format "ext setWidget[%s/%s]:\n  %s" wkey placement lines-str))
        (claude-agent-pi-ui--respond backend id '((value . "")))))
     ((member method '("setTitle" "set_editor_text"))
      (claude-agent-pi--verbose-log
       backend
       (format "extension_ui %s (ack only)" method))
      (claude-agent-pi-ui--respond backend id '((value . ""))))
     (t
      (claude-agent-pi--verbose-log
       backend (format "extension_ui_request unknown method %S — cancelled" method))
      (claude-agent-pi-ui--respond backend id '((cancelled . t)))))))
