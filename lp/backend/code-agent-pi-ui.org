** Overview
:PROPERTIES:
:CUSTOM_ID: pi-ui-overview
:END:

Phase 1 of the *Pi TUI → Emacs migration* described in [[file:../_drafts/draft.org][draft.org]].
Gives the user every routine Pi-TUI control reachable from one
transient submenu — =C-c C-/ p= — so day-to-day Pi usage never
requires opening Pi's terminal UI.

Phase 1 is *RPC-only*: each command maps to one or more existing
=pi --mode rpc= commands.  Multimodal input, tool-call inline
rendering, session-tree browsing, and widget overlays land in
subsequent phases ([[file:../_drafts/draft.org][draft.org]] §pi-mig-phase-2/3/4).

#+BEGIN_SRC elisp
(defun code-agent-pi-ui--current-backend ()
  "Return the Pi backend for the session at point, or signal `user-error'."
  (let* ((sk (and (fboundp 'code-agent-org--current-session-key)
                  (code-agent-org--current-session-key)))
         (b (and sk (fboundp 'code-agent-org--session-get)
                 (code-agent-org--session-get sk :backend))))
    (cond
     ((null sk)
      (user-error "code-agent-pi-ui: not in a code-agent-org session"))
     ((null b)
      (user-error
       "code-agent-pi-ui: no active backend.  Run C-c C-c on an ai block first to spawn pi"))
     ((not (code-agent-pi-backend-p b))
      (user-error
       "code-agent-pi-ui: active backend is %s, not Pi.  Set :CLAUDE_BACKEND: pi on this workspace"
       (type-of b)))
     (t b))))

(defun code-agent-pi-ui--call (backend command-alist &optional timeout)
  "Send COMMAND-ALIST to BACKEND with auto-generated id and wait for response.
Returns the parsed response object (alist).  Signals `user-error' on timeout.

`lexical-let' binds RESULT so the on-response closure captures THIS
call's slot under literate-elisp's dynamic-binding default (H1 from
the 2026-05-26 review).  `unwind-protect' around the send +
event-loop pump guarantees the pending-by-id entry is reaped even if
`--send' signals (e.g. process died after we puthash'd; C2 from the
same review)."
  (lexical-let* ((id (code-agent-pi--gen-id backend "ui"))
                 (cmd (cons (cons 'id id) command-alist))
                 (result nil)
                 (limit (+ (float-time) (or timeout code-agent-pi-ui-rpc-timeout)))
                 (pending (code-agent-pi-backend-pending-by-id backend)))
    (puthash id (list :on-response (lambda (obj) (setq result obj))) pending)
    (unwind-protect
        (progn
          (code-agent-pi--send backend cmd)
          (while (and (null result) (< (float-time) limit))
            (accept-process-output nil 0.05)))
      ;; Always reap; if send raised or we timed out, the entry stays
      ;; stale forever otherwise.
      (when (null result)
        (remhash id pending)))
    (when (null result)
      (user-error "Pi RPC timeout (or send failed) for %s"
                  (or (cdr (assq 'type command-alist)) "<unknown>")))
    result))

(defun code-agent-pi-ui--unwrap (response)
  "Extract data from an RPC RESPONSE alist; signal on success=false."
  (if (eq (map-elt response 'success) t)
      (map-elt response 'data)
    (user-error "Pi RPC %s failed: %s"
                (map-elt response 'command)
                (or (map-elt response 'error) "no detail"))))

#+BEGIN_SRC elisp
(defun code-agent-pi-ui-cycle-model ()
  "Cycle to the next model in the user's `models' settings list."
  (interactive)
  (let* ((b (code-agent-pi-ui--current-backend))
         (resp (code-agent-pi-ui--call b '((type . "cycle_model"))))
         (data (code-agent-pi-ui--unwrap resp)))
    (if data
        (message "Pi model: %s (thinking %s)"
                 (map-elt (map-elt data 'model) 'id)
                 (map-elt data 'thinkingLevel))
      (message "Pi: no scoped models configured (use `pi config' to add)"))))

(defun code-agent-pi-ui-pick-model ()
  "Prompt for a model via `completing-read', then set it on Pi."
  (interactive)
  (let* ((b (code-agent-pi-ui--current-backend))
         (resp (code-agent-pi-ui--call b '((type . "get_available_models"))))
         (data (code-agent-pi-ui--unwrap resp))
         (models (and data (map-elt data 'models)))
         (choices
          (mapcar (lambda (m)
                    (let ((provider (map-elt m 'provider))
                          (id (map-elt m 'id))
                          (name (map-elt m 'name)))
                      (cons (format "%s/%s — %s" provider id name)
                            (cons provider id))))
                  (if (sequencep models) (append models nil) nil))))
    (unless choices (user-error "Pi reported zero available models"))
    (let* ((picked (completing-read "Pi model: " (mapcar #'car choices) nil t))
           (entry (cdr (assoc picked choices)))
           (provider (car entry))
           (id (cdr entry)))
      (code-agent-pi-ui--unwrap
       (code-agent-pi-ui--call b `((type . "set_model")
                                     (provider . ,provider)
                                     (modelId . ,id))))
      (message "Pi model set to %s/%s" provider id))))

*** Session resume + tree (Phase 3)
:PROPERTIES:
:CUSTOM_ID: pi-ui-cmd-resume
:END:

Pi stores sessions as JSONL files under =~/.pi/agent/sessions/<cwd-hash>/=
where the cwd hash is the absolute path with =/= → =-= substitution.
Each file's first line is a =SessionHeader= entry containing the id +
parent-session pointer + creation timestamp.

The resume picker scans the current workspace's session directory,
shows them via =completing-read=, and dispatches =switch_session=.
=session-tree= is the BFS over parent links, rendered into a
read-only =*Pi Sessions: <cwd>*= buffer with =org-mode= folding.

#+BEGIN_SRC elisp
(defcustom code-agent-pi-ui-sessions-dir
  (expand-file-name "~/.pi/agent/sessions/")
  "Root directory under which Pi stores per-cwd session subdirs."
  :type 'directory
  :group 'code-agent)

(defun code-agent-pi-ui--cwd-hash (cwd)
  "Return Pi's filesystem-encoded cwd subdir name for CWD.
Pi replaces =/= with =-= verbatim then wraps with leading + trailing
dash, so =/Users/jt/proj/= → =--Users-jt-proj--= and =/Users/jt/proj=
(no trailing slash) → =--Users-jt-proj-= (single).  Emacs's
`default-directory' always carries a trailing slash, so the common
case maps to the double-dash form Pi creates at spawn time.  M3
from the 2026-05-26 review proposed normalising via
`directory-file-name', but that BREAKS the encoding for the
trailing-slash inputs we always see — kept as-is."
  (let ((normalised (expand-file-name cwd)))
    (concat "-"
            (replace-regexp-in-string "/" "-" normalised)
            "-")))

(defun code-agent-pi-ui--list-session-files (&optional cwd)
  "Return absolute paths of all session JSONL files for CWD (default
=default-directory=).  Newest first."
  (let* ((root (file-name-as-directory code-agent-pi-ui-sessions-dir))
         (sub (code-agent-pi-ui--cwd-hash (or cwd default-directory)))
         (dir (expand-file-name sub root)))
    (when (file-directory-p dir)
      (sort (directory-files dir t "\\.jsonl\\'" t)
            (lambda (a b)
              (time-less-p (nth 5 (file-attributes b))
                            (nth 5 (file-attributes a))))))))

(defun code-agent-pi-ui--read-session-header (path)
  "Parse the first JSONL line of PATH; return the header alist or nil."
  (with-temp-buffer
    (condition-case nil
        (progn
          (insert-file-contents path nil 0 4096)
          (goto-char (point-min))
          (let ((line (buffer-substring-no-properties
                       (point) (line-end-position))))
            (json-parse-string line :object-type 'alist :null-object nil :false-object nil)))
      (error nil))))

(defun code-agent-pi-ui--scan-session-summary (path &optional max-bytes max-turns)
  "Scan PATH (a session JSONL file) and return a plist with rich
display metadata for resume / tree pickers.

Returned keys:
  :path                  absolute file path
  :id                    session UUID (from header)
  :name                  user-set session name if any
  :timestamp             ISO8601 from header
  :mtime                 file modification time (decoded)
  :parent-session        absolute path of parent session, if any
  :model                 model id of last assistant message scanned
  :provider              provider of last assistant message scanned
  :turn-count            count of `assistantMessage' entries scanned
  :first-user-msg        first user message text (clipped to 80 chars)

MAX-BYTES caps how much of the file we read (default 65536) — for
big sessions we only sample the first chunk, so turn-count is a
lower bound on truly large sessions.  MAX-TURNS caps the parse
loop (default 200)."
  (let* ((max-bytes (or max-bytes 65536))
         (max-turns (or max-turns 200))
         (hdr nil) (id nil) (name nil) (timestamp nil) (parent nil)
         (model nil) (provider nil) (turn-count 0) (first-user-msg nil))
    (with-temp-buffer
      (condition-case nil
          (insert-file-contents path nil 0 max-bytes)
        (error nil))
      (goto-char (point-min))
      (let ((scanned 0))
        (while (and (not (eobp)) (< scanned max-turns))
          (let ((line (buffer-substring-no-properties
                       (point) (line-end-position))))
            (when (and line (not (string-empty-p line)))
              (condition-case nil
                  (let* ((obj (json-parse-string
                               line :object-type 'alist
                               :null-object nil :false-object nil))
                         (otype (map-elt obj 'type)))
                    (cond
                     ((and (null hdr) (equal otype "session"))
                      (setq hdr obj
                            id (map-elt obj 'id)
                            name (map-elt obj 'name)
                            timestamp (map-elt obj 'timestamp)
                            parent (map-elt obj 'parentSession)))
                     ((equal otype "assistantMessage")
                      (cl-incf turn-count)
                      (let* ((m (map-elt obj 'message))
                             (mm (or (map-elt obj 'model)
                                     (and m (map-elt m 'model))))
                             (pp (or (map-elt obj 'provider)
                                     (and m (map-elt m 'provider)))))
                        (when mm (setq model mm))
                        (when pp (setq provider pp))))
                     ((and (null first-user-msg)
                           (equal otype "userMessage"))
                      (let* ((m (map-elt obj 'message))
                             (c (and m (map-elt m 'content))))
                        (setq first-user-msg
                              (cond
                               ((stringp c) c)
                               ((sequencep c)
                                (let ((first (and (> (length c) 0) (elt c 0))))
                                  (and first (map-elt first 'text))))
                               (t nil)))))))
                (error nil))))
          (forward-line 1)
          (cl-incf scanned))))
    (let ((clipped (and first-user-msg
                        (truncate-string-to-width
                         (replace-regexp-in-string
                          "\\s-+" " " (string-trim first-user-msg))
                         80 nil nil "…"))))
      (list :path path :id id :name name :timestamp timestamp
            :mtime (nth 5 (file-attributes path))
            :parent-session parent
            :model model :provider provider
            :turn-count turn-count
            :first-user-msg clipped))))

(defun code-agent-pi-ui--format-session-line (summary)
  "Format a SUMMARY plist as a single completing-read display line.
Layout: `<short-mtime>  <provider/model>  <turns>t  [<id-prefix>] «name» \"<first-msg>\"'."
  (let* ((mtime (plist-get summary :mtime))
         (short-mtime (and mtime (format-time-string "%Y-%m-%d %H:%M" mtime)))
         (provider (plist-get summary :provider))
         (model (plist-get summary :model))
         (provmodel (cond
                     ((and provider model) (format "%s/%s" provider model))
                     (model (format "%s" model))
                     (t "(no model yet)")))
         (turns (plist-get summary :turn-count))
         (id (plist-get summary :id))
         (id-short (and id (substring id 0 (min 8 (length id)))))
         (name (plist-get summary :name))
         (msg (plist-get summary :first-user-msg)))
    (format "%s  %-30s %3dt  [%s]%s  %S"
            (or short-mtime "(no mtime)      ")
            provmodel
            (or turns 0)
            (or id-short "????????")
            (if name (format " «%s»" name) "")
            (or msg ""))))

(defun code-agent-pi-ui-resume-session ()
  "Pick + resume a Pi session for the current workspace.
Scans `code-agent-pi-ui-sessions-dir' for the cwd subdir, parses
each JSONL with `--scan-session-summary' to extract id + name +
timestamp + provider/model + turn count + first user message,
presents via `completing-read', dispatches `switch_session' on
the chosen path."
  (interactive)
  (let* ((b (code-agent-pi-ui--current-backend))
         (files (code-agent-pi-ui--list-session-files default-directory))
         (entries
          (mapcar
           (lambda (path)
             (let ((summary (code-agent-pi-ui--scan-session-summary path)))
               (cons (code-agent-pi-ui--format-session-line summary)
                     path)))
           files)))
    (unless entries
      (user-error "No Pi sessions found under %s for cwd %s"
                  code-agent-pi-ui-sessions-dir default-directory))
    (let* ((pick (completing-read "Resume Pi session: "
                                   (mapcar #'car entries) nil t))
           (path (cdr (assoc pick entries))))
      (code-agent-pi-ui--unwrap
       (code-agent-pi-ui--call b `((type . "switch_session")
                                      (sessionPath . ,path))
                                  30.0))
      (message "Pi: switched to %s" (file-name-base path)))))

(defun code-agent-pi-ui--session-path-at-point ()

(defun code-agent-pi-ui-session-tree-resume-at-point ()
  "Switch to the session at point in the *Pi Sessions* tree."
  (interactive)
  (let ((path (code-agent-pi-ui--session-path-at-point)))
    (unless path (user-error "No :PATH: property at point"))
    (let ((b (code-agent-pi-ui--current-backend)))
      (code-agent-pi-ui--unwrap
       (code-agent-pi-ui--call b `((type . "switch_session")
                                      (sessionPath . ,path))
                                  30.0))
      (message "Pi: switched to %s" (file-name-base path))
      (code-agent-pi-ui-session-tree-refresh))))

(defun code-agent-pi-ui-session-tree-fork-at-point ()
  "Fork from the session at point.
Asks Pi to load the at-point session, then requests `clone' to start a
fresh branch.  This is `at-session-level' forking — for in-session
branch-from-entry use the Phase 3.1 fork-messages picker (TBD)."
  (interactive)
  (let ((path (code-agent-pi-ui--session-path-at-point)))
    (unless path (user-error "No :PATH: property at point"))
    (unless (yes-or-no-p (format "Fork (clone) Pi session %s? "
                                  (file-name-base path)))
      (user-error "Aborted"))
    (let ((b (code-agent-pi-ui--current-backend)))
      ;; First switch to the source session
      (code-agent-pi-ui--unwrap
       (code-agent-pi-ui--call b `((type . "switch_session")
                                      (sessionPath . ,path))
                                  30.0))
      ;; Then clone it into a new branch
      (let* ((resp (code-agent-pi-ui--call b '((type . "clone")) 15.0))
             (data (code-agent-pi-ui--unwrap resp)))
        (if (and data (map-elt data 'cancelled))
            (message "Pi: clone cancelled")
          (message "Pi: cloned %s into a new branch" (file-name-base path))
          (code-agent-pi-ui-session-tree-refresh))))))

(defun code-agent-pi-ui-session-tree-refresh ()
  "Re-render the current *Pi Sessions* buffer from disk."
  (interactive)
  (let ((cwd (or code-agent-pi-ui--session-tree-cwd default-directory)))
    (code-agent-pi-ui-session-tree cwd)))

(defun code-agent-pi-ui-session-tree (&optional cwd)
  "Open a buffer showing the session DAG for CWD (default `default-directory').
Each session row links to its parent via the JSONL header's
=parentSession= field; we render the tree as indented org headings
in a =*Pi Sessions: <cwd>*= buffer with keys:
  RET   resume the session at point (switch_session)
  f     fork (clone) the session at point as a new branch
  g     refresh from disk
  q     quit window

The currently-active session id (from the live backend, if any) is
visually marked with `▶' in its heading."
  (interactive)
  (let* ((cwd (or cwd default-directory))
         (files (code-agent-pi-ui--list-session-files cwd))
         (by-id (make-hash-table :test 'equal))
         (children (make-hash-table :test 'equal))
         (roots nil)
         (active-id (let ((b (ignore-errors (code-agent-pi-ui--current-backend))))
                      (and b (code-agent-pi-backend-session-id b)))))
    (unless files
      (user-error "No Pi sessions found for cwd %s" cwd))
    (dolist (path files)
      (let* ((summary (code-agent-pi-ui--scan-session-summary path))
             (id (plist-get summary :id))
             (parent-path (plist-get summary :parent-session)))
        (when id
          (puthash id summary by-id)
          (if parent-path
              ;; parent is path → resolve to id via reverse lookup later
              (push (list :child id :parent-path parent-path)
                    (gethash :pending-parents children))
            (push id roots)))))
    ;; Resolve parent paths → parent ids
    (let ((pending (gethash :pending-parents children)))
      (remhash :pending-parents children)
      (dolist (link pending)
        (let* ((child-id (plist-get link :child))
               (parent-path (plist-get link :parent-path))
               (parent-id (cl-loop for k being the hash-keys of by-id
                                   for v = (gethash k by-id)
                                   when (equal (plist-get v :path) parent-path)
                                   return k)))
          (if parent-id
              (push child-id (gethash parent-id children))
            ;; parent not in our cwd dir → treat child as root
            (push child-id roots)))))
    (let ((buf (get-buffer-create
                (format "*Pi Sessions: %s*"
                        (abbreviate-file-name cwd)))))
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          (erase-buffer)
          (org-mode)
          (insert (format "#+TITLE: Pi sessions for %s\n\n"
                          (abbreviate-file-name cwd)))
          (insert "Keys: RET=resume  f=fork(clone)  g=refresh  q=quit\n")
          (when active-id
            (insert (format "▶ marks active session (id=%s)\n"
                            (substring active-id 0 (min 8 (length active-id))))))
          (insert "\n")
          (code-agent-pi-ui--render-session-subtree
           roots by-id children 1 active-id nil))
        (setq-local code-agent-pi-ui--session-tree-cwd cwd)
        (use-local-map code-agent-pi-ui-session-tree-mode-map)
        (goto-char (point-min))
        (setq buffer-read-only t))
      (pop-to-buffer buf))))

(defun code-agent-pi-ui--render-session-subtree (ids by-id children depth active-id &optional visited)
  "Render IDS at DEPTH as org headings; recurse into children.
VISITED is an internal hash-set of already-rendered ids, used to
break cycles in case a Pi session JSONL on disk has a malformed
parentSession pointer (M1 from the 2026-05-26 review).

ACTIVE-ID is the currently-active backend session id; matching
nodes get a `▶' marker.

Uses `reverse' (not `nreverse') so callers can re-use the hash-table
values without observing the destructive side-effect."
  (let ((seen (or visited (make-hash-table :test 'equal))))
    (dolist (id (reverse ids))
      (unless (gethash id seen)
        (puthash id t seen)
        (let* ((summary (gethash id by-id))
               (path (and summary (plist-get summary :path)))
               (active (and active-id (equal id active-id)))
               (marker (if active "▶ " "  "))
               (display-line (and summary
                                  (code-agent-pi-ui--format-session-line summary))))
          (insert (format "%s %s%s\n"
                          (make-string (1+ depth) ?\*)
                          marker
                          (or display-line (format "%s (no summary)" id))))
          (when path
            (insert (format "   :PROPERTIES:\n   :PATH: %s\n   :ID: %s\n   :END:\n"
                            path (or id ""))))
          (let ((kids (gethash id children)))
            (when kids
              (code-agent-pi-ui--render-session-subtree
               kids by-id children (1+ depth) active-id seen))))))))
#+END_SRC

** Widget store
:PROPERTIES:
:CUSTOM_ID: pi-ui-widget-store
:END:

Phase 4 widget surface — see [[file:../draft.org::#pi-mig-phase-4][draft.org §pi-mig-phase-4]].
Pi extensions emit =extension_ui_request method=setWidget= with
=widgetKey= + =widgetLines: string[] | undefined= +
=widgetPlacement: aboveEditor|belowEditor=.

This module *stores* the live widget set per Emacs session (the
key the verbose buffer + backend use to scope everything else) and
renders snapshots on demand via =M-x code-agent-pi-ui-show-widgets=
(also bound as =w= in the transient menu).

Why store-and-on-demand instead of inline overlay on the ai block:
- *Cross-buffer simplicity*: the backend doesn't carry an ai-block
  position (only =session-key=).  An overlay would need that.
- *Replace-by-key semantics*: subsequent =setWidget= calls with the
  same =widgetKey= replace prior content.  The store is the
  natural data structure.
- *Per-session isolation*: cmux concurrent sessions never see each
  other's widgets (the store is keyed by =session-key=).

#+BEGIN_SRC elisp
(defvar code-agent-pi-ui--widget-store nil
  "Alist mapping SESSION-KEY (string) → widget-alist.

Each widget-alist is itself an alist mapping WIDGET-KEY (string) →
plist (:lines LIST :placement STRING :ts TIME).

Updates flow:
  - LINES non-nil: replace or insert the entry, refresh =:ts=.
  - LINES nil: drop the entry (matches pi spec where
    `widgetLines: undefined' clears the widget).")

(defun code-agent-pi-ui--widget-store-update (session-key widget-key lines placement)
  "Replace or remove WIDGET-KEY entry under SESSION-KEY in the store.
LINES nil removes the entry; non-nil LINES replaces or inserts.

When LINES is nil AND the session has no prior entries, this is a
no-op — we don't push a stale `(SESSION-KEY . nil)' cell (review A5).
Defensive: skips entirely if WIDGET-KEY is nil/empty (review A1)."
  (when (and session-key
             (stringp widget-key)
             (not (string-empty-p widget-key)))
    (let* ((session-entry (assoc session-key code-agent-pi-ui--widget-store))
           (current-alist (cdr session-entry)))
      (cond
       ;; Add / replace
       (lines
        (setf (alist-get widget-key current-alist nil nil #'equal)
              (list :lines lines
                    :placement (or placement "belowEditor")
                    :ts (current-time)))
        (if session-entry
            (setcdr session-entry current-alist)
          (push (cons session-key current-alist)
                code-agent-pi-ui--widget-store)))
       ;; Remove (only if we actually have something to remove)
       (session-entry
        (setcdr session-entry
                (assoc-delete-all widget-key current-alist #'equal)))
       ;; Else: clear request for a never-seen session → no-op
       ))))

(defun code-agent-pi-ui--widget-snapshot-buffer-name (session-key)
  "Return the canonical buffer name for SESSION-KEY's widget snapshot."
  (format "*Pi Widgets: %s*" (or session-key "(no session)")))

(defun code-agent-pi-ui--widget-snapshot-refresh (session-key)
  "If `*Pi Widgets: <session-key>*' is visible in any window, re-render it.
Called from the setWidget handler so a visible snapshot tracks live."
  (let ((buf (get-buffer (code-agent-pi-ui--widget-snapshot-buffer-name
                          session-key))))
    (when (and buf (get-buffer-window buf t))
      (with-current-buffer buf
        (code-agent-pi-ui--widget-snapshot-render session-key)))))

(defun code-agent-pi-ui--widget-snapshot-render (session-key)
  "Render the current widget store for SESSION-KEY into the current buffer.
Buffer is replaced wholesale (idempotent).  Sort widgets by
=widgetKey= for stable display."
  (let ((inhibit-read-only t)
        (widget-alist (cdr (assoc session-key
                                  code-agent-pi-ui--widget-store))))
    (erase-buffer)
    (org-mode)
    (insert (format "#+TITLE: Pi widgets for session %s\n" session-key))
    (insert (format "#+STARTUP: showall\n\n"))
    (if (null widget-alist)
        (insert "(no active widgets — extensions can register some via setWidget)\n")
      (insert (format "%d active widget(s):\n\n" (length widget-alist)))
      (dolist (entry (sort (copy-sequence widget-alist)
                           (lambda (a b) (string< (car a) (car b)))))
        (let* ((wkey (car entry))
               (plist (cdr entry))
               (lines (plist-get plist :lines))
               (placement (plist-get plist :placement))
               (ts (plist-get plist :ts)))
          (insert (format "* %s\n" wkey))
          (insert (format "  :PROPERTIES:\n"))
          (insert (format "  :placement: %s\n" placement))
          (insert (format "  :updated:   %s\n"
                          (format-time-string "%Y-%m-%d %H:%M:%S" ts)))
          (insert (format "  :END:\n\n"))
          (dolist (line lines)
            (insert (format "  %s\n" line)))
          (insert "\n"))))
    (goto-char (point-min))
    (setq buffer-read-only t)))

(defun code-agent-pi-ui-show-widgets ()
  "Pop a snapshot buffer showing the current Pi widgets for this session.
Idempotent: re-running refreshes the same buffer."
  (interactive)
  (let* ((b (code-agent-pi-ui--current-backend))
         (sk (code-agent-pi-backend-session-key b))
         (buf-name (code-agent-pi-ui--widget-snapshot-buffer-name sk))
         (buf (get-buffer-create buf-name)))
    (with-current-buffer buf
      (code-agent-pi-ui--widget-snapshot-render sk))
    (pop-to-buffer buf)))

(defun code-agent-pi-ui-clear-all-widgets ()
  "Drop all widgets for the current session.  Useful when an extension
leaves stale entries after a crash."
  (interactive)
  (let* ((b (code-agent-pi-ui--current-backend))
         (sk (code-agent-pi-backend-session-key b)))
    (setf (alist-get sk code-agent-pi-ui--widget-store
                     nil nil #'equal)
          nil)
    (code-agent-pi-ui--widget-snapshot-refresh sk)
    (message "Pi: cleared all widgets for session %s" sk)))
#+END_SRC

** Extension UI handlers
:PROPERTIES:
:CUSTOM_ID: pi-ui-extension-ui
:END:

Replaces =code-agent-pi--default-ui-handler= (auto-cancel stub) with
real implementations.

#+BEGIN_SRC elisp
(defun code-agent-pi-ui--respond (backend id payload)
  "Send `extension_ui_response' for ID with PAYLOAD plist members."
  (code-agent-pi--send
   backend
   `((type . "extension_ui_response")
     (id . ,id)
     ,@payload)))

(defun code-agent-pi-ui--extension-ui-handler (backend obj)
  "Real handler for Pi `extension_ui_request' events."
  (let* ((id (map-elt obj 'id))
         (method (map-elt obj 'method)))
    (cond
     ((null id)
      (when (equal method "notify")
        (message "Pi extension notify: %s" (map-elt obj 'message))))
     ((equal method "select")
      (let* ((title (map-elt obj 'title))
             (opts (map-elt obj 'options))
             (opts-list (if (sequencep opts) (append opts nil) nil))
             (pick (and opts-list
                        (completing-read (format "%s: " title) opts-list nil t))))
        (if pick
            (code-agent-pi-ui--respond backend id `((value . ,pick)))
          (code-agent-pi-ui--respond backend id '((cancelled . t))))))
     ((equal method "confirm")
      (let* ((title (map-elt obj 'title))
             (msg (map-elt obj 'message))
             (answer (yes-or-no-p (format "%s: %s " title msg))))
        (code-agent-pi-ui--respond backend id `((confirmed . ,(if answer t :json-false))))))
     ((equal method "input")
      (let* ((title (map-elt obj 'title))
             (placeholder (or (map-elt obj 'placeholder) ""))
             (val (read-string (format "%s: " title) placeholder)))
        (code-agent-pi-ui--respond backend id `((value . ,val)))))
     ((equal method "editor")
      (let* ((title (map-elt obj 'title))
             (prefill (or (map-elt obj 'prefill) ""))
             (val (read-string (format "%s (one line, RET to submit): " title)
                                prefill)))
        (code-agent-pi-ui--respond backend id `((value . ,val)))))
     ((equal method "notify")
      (let ((msg (map-elt obj 'message))
            (kind (or (map-elt obj 'notifyType) "info")))
        (cond
         ((equal kind "error")
          (display-warning 'code-agent-pi msg :error))
         ((equal kind "warning")
          (display-warning 'code-agent-pi msg :warning))
         (t (message "Pi: %s" msg)))
        (code-agent-pi-ui--respond backend id '((value . "")))))
     ((equal method "setStatus")
      ;; Track per-key extension status; surface to user as a message
      ;; so a "loaded N rules" notification doesn't vanish into the
      ;; verbose buffer.
      (let ((status-key (map-elt obj 'statusKey))
            (status-text (map-elt obj 'statusText)))
        (when (and status-text (stringp status-text)
                   (> (length status-text) 0))
          (message "Pi ext [%s]: %s" status-key status-text))
        (code-agent-pi--verbose-log
         backend (format "ext setStatus %s = %S" status-key status-text)))
      (code-agent-pi-ui--respond backend id '((value . ""))))
     ((equal method "setWidget")
      ;; Phase 4: maintain a per-session widget store keyed by widgetKey.
      ;; `widgetLines = undefined' (= nil here) removes the entry.
      ;; Render is on-demand via M-x `code-agent-pi-ui-show-widgets'
      ;; (also bound under the transient menu — `w' below).
      (let* ((sk (code-agent-pi-backend-session-key backend))
             (wkey (map-elt obj 'widgetKey))
             (lines (map-elt obj 'widgetLines))
             (lines-list (cond
                          ((null lines) nil)
                          ((sequencep lines) (append lines nil))
                          (t (list (format "%S" lines)))))
             (placement (or (map-elt obj 'widgetPlacement) "belowEditor")))
        (code-agent-pi-ui--widget-store-update sk wkey lines-list placement)
        ;; Snapshot buffer auto-refresh: if `*Pi Widgets: <sk>*' is
        ;; visible, re-render it so the user sees updates live.
        (code-agent-pi-ui--widget-snapshot-refresh sk)
        ;; Verbose log keeps the actual content (review B1) so the
        ;; buffer remains the canonical debug surface.  Keep the
        ;; bracket format `[key/placement]' consistent across update
        ;; and clear so downstream greps don't split on word boundary.
        (code-agent-pi--verbose-log
         backend
         (if lines-list
             (format "ext setWidget[%s/%s]: %d line(s)\n  %s"
                     wkey placement (length lines-list)
                     (mapconcat #'identity lines-list "\n  "))
           (format "ext setWidget[%s/%s]: cleared" wkey placement)))
        (code-agent-pi-ui--respond backend id '((value . "")))))
     ((member method '("setTitle" "set_editor_text"))
      (code-agent-pi--verbose-log
       backend
       (format "extension_ui %s (ack only)" method))
      (code-agent-pi-ui--respond backend id '((value . ""))))
     (t
      (code-agent-pi--verbose-log
       backend (format "extension_ui_request unknown method %S — cancelled" method))
      (code-agent-pi-ui--respond backend id '((cancelled . t)))))))

(setq code-agent-pi-extension-ui-handler
      #'code-agent-pi-ui--extension-ui-handler)
