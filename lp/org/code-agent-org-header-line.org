#+TITLE: Code Agent Org — Header Line
#+PROPERTY: literate-load yes

* Overview
:PROPERTIES:
:CUSTOM_ID: org-header-line-overview
:END:

Buffer-local header-line for =code-agent-org-mode= buffers.  Shows
the current file name plus a chain of badges: workspace / story
location, Docker-sandbox flag, permission mode, activity (number
of thinking sessions + queued blocks), project root, and IDE
context (current file + selection).  Click the activity badge to
open the session manager.

Layered on top of [[file:code-agent-org.org][code-agent-org.org]]'s session / queue / project
/ permission helpers — so this module loads *after*
[[file:code-agent-org.org][code-agent-org.org]], not before.  A single 0.2-second timer
refreshes all buffers; a separate 0.2-second debounced timer
recomputes the cached workspace/story name on cursor movement.

* Dependencies

Loaded eagerly at file evaluation time — symbols below assume every required feature is already provided.

#+BEGIN_SRC elisp
(require 'cl-lib)
(require 'code-agent-org)
#+END_SRC

(defvar-local code-agent-org--original-header-line nil

(defvar-local code-agent-org--original-truncate-lines nil
  "Original truncate-lines value before enabling code-agent-org-mode.")

(defconst code-agent-org--spinner-frames '("|" "/" "-" "\\")
  "Frames for the busy spinner animation.")

(defvar code-agent-org--header-timer nil
  "Timer for refreshing header line across all code-agent-org buffers.")

(defvar code-agent-org--session-badge-map
  (let ((map (make-sparse-keymap)))
    (define-key map [header-line mouse-1] #'code-agent-org-list-sessions)
    map)
  "Keymap for clicking on the session badge in the header line.")

  "Cached workspace name for header line (updated on cursor movement).")

(defvar-local code-agent-org--header-story-name nil
  "Cached story name for header line (updated on cursor movement).")

(defvar-local code-agent-org--header-cursor-timer nil
  "Debounce timer for cursor-position-driven header updates.")

* Spinner Control

Drives the spinner frame index forward + invalidates the header-line — called by the per-session activity timer.

#+BEGIN_SRC elisp
(defun code-agent-org--start-spinner (session-key)
  "Start spinner animation for SESSION-KEY."
  (code-agent-org--session-put session-key :spinner 0)
  (code-agent-org--ensure-header-timer))

(defun code-agent-org--stop-spinner (session-key)
  "Stop spinner animation for SESSION-KEY."
  (code-agent-org--session-put session-key :spinner nil))
#+END_SRC

* Timer Management

Lifecycle helpers for timer — create / inspect / dispose.

#+BEGIN_SRC elisp
(defun code-agent-org--ensure-header-timer ()
  "Ensure header refresh timer is running."
  (unless (and code-agent-org--header-timer
               (timerp code-agent-org--header-timer))
    (setq code-agent-org--header-timer
          (run-at-time 0.2 0.2 #'code-agent-org--refresh-all-headers))))

(defun code-agent-org--stop-header-timer ()
  "Stop header refresh timer if no active sessions."
  (when (and code-agent-org--header-timer
             (zerop (code-agent-org--active-session-count)))
    (cancel-timer code-agent-org--header-timer)
    (setq code-agent-org--header-timer nil)))

(defun code-agent-org--refresh-all-headers ()
  "Refresh header line in all code-agent-org buffers."
  (dolist (buf (buffer-list))
    (when (buffer-live-p buf)
      (with-current-buffer buf
        (when (bound-and-true-p code-agent-org-mode)
          ;; Skip indirect buffers to avoid org-element errors
          ;; These can cause "Invalid search bound" errors when org-entry-get
          ;; tries to parse with inheritance across buffer boundaries
          (unless (buffer-base-buffer)
            (code-agent-org--refresh-header-line)
            ;; Advance spinner for busy sessions (guard against nil hash table)
            (when (hash-table-p code-agent-org--sessions)
              (maphash (lambda (_key state)
                         (when (and (code-agent-org--session-state-p state)
                                    (code-agent-org--session-state-get state :busy))
                           (let* ((frame (or (code-agent-org--session-state-get state :spinner) 0))
                                  (next-frame (mod (1+ frame) 4)))
                             (code-agent-org--session-state-set state :spinner next-frame))))
                       code-agent-org--sessions))))))))

(defun code-agent-org--refresh-header-line ()
  "Refresh the header line with current session status."
  (setq header-line-format (code-agent-org--header-line))
  (force-mode-line-update))
#+END_SRC

* Cursor-Position Tracking

The header line shows which workspace and story the cursor is in.
These update on a 0.2-second debounce after cursor movement — fast
enough to feel responsive, slow enough to avoid unnecessary org tree
walks on every keystroke.

#+BEGIN_SRC elisp
(defun code-agent-org--schedule-cursor-header-update ()
  "Schedule a debounced header update after cursor movement.
Added to `post-command-hook' by `code-agent-org-mode'."
  (when code-agent-org--header-cursor-timer
    (cancel-timer code-agent-org--header-cursor-timer))
  (setq code-agent-org--header-cursor-timer
        (run-at-time 0.2 nil #'code-agent-org--update-cursor-header
                     (current-buffer))))

(defun code-agent-org--update-cursor-header (buf)
  "Update cached workspace/story names from cursor position in BUF."
  (when (buffer-live-p buf)
    (with-current-buffer buf
      (setq code-agent-org--header-cursor-timer nil)
      (when (bound-and-true-p code-agent-org-mode)
        (let ((ws (ignore-errors (code-agent-org--find-workspace-heading)))
              (story (ignore-errors (code-agent-org--find-story-heading))))
          (let ((new-ws (and ws (substring-no-properties (car ws))))
                (new-story (and story (substring-no-properties (car story)))))
            (unless (and (equal new-ws code-agent-org--header-workspace-name)
                         (equal new-story code-agent-org--header-story-name))
              (setq code-agent-org--header-workspace-name new-ws
                    code-agent-org--header-story-name new-story)
              (code-agent-org--refresh-header-line))))))))
#+END_SRC

* Tooltip Formatting

Pretty-printing for tooltip — used by the verbose buffer + status surfaces.

#+BEGIN_SRC elisp
(defun code-agent-org--format-session-tooltip ()
  "Format tooltip showing all active sessions."
  (let ((lines (list "Active Sessions" "───────────────────")))
    (maphash
     (lambda (id state)
       (when (and (code-agent-org--session-state-p state)
                  (code-agent-org--session-state-get state :busy))
         (let* ((spinner-idx (or (code-agent-org--session-state-get state :spinner) 0))
                (spinner-char (nth spinner-idx code-agent-org--spinner-frames))
                (display-name (code-agent-org--session-display-name id))
                (elapsed (code-agent-org--format-elapsed
                          (code-agent-org--session-state-get state :start-time))))
           (push (format "● %s  [%s %s]" display-name spinner-char elapsed)
                 lines))))
     code-agent-org--sessions)
    (if (= (length lines) 2)
        "No active sessions\n\nClick to open session manager"
      (concat (string-join (nreverse lines) "\n")
              "\n\nClick to open session manager"))))
#+END_SRC

* Header Line Generation

Each badge is a small dedicated function so they can be composed
or replaced (a UI experiment can drop a single badge without
rewriting the assembler).  =code-agent-org--header-line=
concatenates them in display order.

#+BEGIN_SRC elisp
(defun code-agent-org--header-session-badge ()
  "Return session scope badge string for header line.
Shows [sec:ID...] for section scope, [file:ID...] for file scope,
or [file-only] when no session ID is set."
  (let ((session-id (code-agent-org--get-session-id))
        (session-scope (code-agent-org--get-session-scope)))
    (if session-id
        (propertize (format " [%s:%s...]"
                            (if (eq session-scope 'section) "sec" "file")
                            (substring session-id 0 (min 12 (length session-id))))
                    'face (if (eq session-scope 'section)
                              'font-lock-constant-face
                            'font-lock-type-face))
      (propertize " [file-only]" 'face 'font-lock-comment-face))))

(defun code-agent-org--header-docker-badge ()
  "Return Docker sandbox badge string for header line.
Returns \" [Docker]\" when Docker mode is active, empty string otherwise."
  (let ((docker-p (code-agent-org--docker-mode-p)))
    (if docker-p
        (let ((container-path (ignore-errors
                                (org-entry-get nil code-agent-org-container-path-property t))))
          (propertize " [Docker]"
                      'face '(:weight bold)
                      'help-echo (format "Docker sandbox mode\nCLI: %s\nContainer path: %s"
                                         (or (code-agent-org--get-cli-path) "default")
                                         (or container-path "/workspace"))))
      "")))

(defun code-agent-org--header-permission-badge ()
  "Return permission mode badge string for header line."
  (propertize (format " [%s]" (code-agent-org--permission-mode-short))
              'face 'font-lock-warning-face
              'help-echo (format "Permission: %s (C-c C-p to change)"
                                 (code-agent-org--get-permission-mode-display))))

(defun code-agent-org--header-activity-badge ()
  "Return activity badge string for header line.
Shows thinking count and queue depth when sessions are active."
  (let ((active-count (code-agent-org--active-session-count))
        (total-queued (code-agent-org--total-queue-count)))
    (if (> active-count 0)
        (let ((text (if (> total-queued 0)
                        (format " [%d thinking... Q:%d]" active-count total-queued)
                      (format " [%d thinking...]" active-count))))
          (propertize text
                      'face '(:weight bold)
                      'help-echo (code-agent-org--format-session-tooltip)
                      'mouse-face 'mode-line-highlight
                      'local-map code-agent-org--session-badge-map))
      "")))

(defun code-agent-org--header-project-badge ()
  "Return project root badge string for header line."
  (let ((project-root (code-agent-org--get-project-root)))
    (if project-root
        (propertize (format " [%s]"
                            (file-name-nondirectory
                             (directory-file-name project-root)))
                    'face 'font-lock-keyword-face)
      "")))

(defun code-agent-org--header-ide-context ()
  "Return IDE context string (current file + selection) for header line."
  (let* ((ide-context (code-agent-collect-ide-context))
         (current-file (plist-get ide-context :current-file))
         (file-str (if current-file
                       (propertize (format " | %s%s"
                                           (or (plist-get current-file :name) "?")
                                           (if (plist-get current-file :modified) "*" ""))
                                   'face 'font-lock-string-face)
                     ""))
         (selection (plist-get ide-context :selection))
         (sel-str (if (and selection (plist-get selection :text))
                      (let* ((start-line (plist-get selection :start-line))
                             (end-line (plist-get selection :end-line))
                             (text-len (length (plist-get selection :text))))
                        (propertize (format " | sel:%d-%d (%d)" start-line end-line text-len)
                                    'face 'font-lock-comment-face))
                    "")))
    (concat file-str sel-str)))

(defun code-agent-org--header-location-badge ()
  "Return workspace/story location badge for header line.
Reads from cached values updated by `code-agent-org--update-cursor-header'."
  (let ((ws code-agent-org--header-workspace-name)
        (story code-agent-org--header-story-name))
    (cond
     ((and ws story)
      (concat (propertize (format " %s" ws) 'face 'font-lock-type-face)
              (propertize " / " 'face 'font-lock-comment-face)
              (propertize story 'face 'font-lock-constant-face)))
     (ws
      (propertize (format " %s" ws) 'face 'font-lock-type-face))
     (t ""))))

(defun code-agent-org--header-line ()
  "Generate the header line content for Claude Org buffer.
Assembles components from dedicated badge functions."
  (let ((file-name (if buffer-file-name
                       (file-name-nondirectory buffer-file-name)
                     "untitled")))
    (concat " Claude Org: "
            (propertize file-name 'face 'font-lock-function-name-face)
            (code-agent-org--header-location-badge)
            (code-agent-org--header-docker-badge)
            (code-agent-org--header-permission-badge)
            (code-agent-org--header-activity-badge)
            (code-agent-org--header-project-badge)
            (code-agent-org--header-ide-context))))
#+END_SRC
