* Overview
:PROPERTIES:
:CUSTOM_ID: org-queue-overview
:END:

When a Claude query is in flight on a buffer and the user submits a
second =#+begin_src ai= block (or the same block again), the second
submission queues here instead of racing the first.  Without this
queue, concurrent queries would interleave tokens into each other's
response sections and clobber each other's marker positions.

Invariant: the queue is per-session (keyed by session-key), FIFO,
capped by =code-agent-org-max-queue-depth=, and deduped by
=:CUSTOM_ID:= when present so a re-submission of the same block does
not double-count.  =code-agent-org--queue-pop= is the only path that
advances; the session's =:busy= flag must be nil before a pop
materialises a new in-flight query.

** Queue Operations

Three primitives drive the queue: =--same-block-p= for dedup,
=--queue-block= for enqueue with depth check + dedup, and
=--queue-pop= for FIFO dequeue.  Each session has its own pending
queue stored on the lifecycle sub-struct's =pending-queue= slot.

#+BEGIN_SRC elisp
(defun code-agent-org--same-block-p (id-a marker-a id-b marker-b)
  "Return non-nil if block A and block B refer to the same block.
Compares by custom-id when both are non-nil, otherwise by marker position."
  (if (and id-a id-b)
      (equal id-a id-b)
    (let ((pos-a (and (code-agent-org--marker-valid-p marker-a)
                      (marker-position marker-a)))
          (pos-b (and (code-agent-org--marker-valid-p marker-b)
                      (marker-position marker-b))))
      (and pos-a pos-b (= pos-a pos-b)))))

(defun code-agent-org--queue-block (session-key block-info)
  "Add BLOCK-INFO to pending queue for SESSION-KEY.
BLOCK-INFO is a plist with :custom-id, :content, :marker, :instr-num.
Duplicate detection uses custom-id.  When custom-id is unavailable on
either side, marker position is compared instead.
Returns:
  'queued    - block was successfully added to queue
  'running   - block is already running (cannot queue)
  'in-queue  - block is already in the queue"
  (code-agent-with-span "queue-block"
      (list :span-kind "PRODUCER"
            :custom-id (plist-get block-info :custom-id)) nil
  (let* ((queue (code-agent-org--session-get session-key :pending-queue))
         (new-custom-id (plist-get block-info :custom-id))
         (running-custom-id (code-agent-org--session-get session-key :custom-id))
         (new-marker (plist-get block-info :marker))
         (running-marker (code-agent-org--session-get session-key :marker))
         ;; Determine if the new block matches the running block
         (is-running
          (and (code-agent-org--session-get session-key :busy)
               (code-agent-org--same-block-p new-custom-id new-marker
                                         running-custom-id running-marker)))
         ;; Determine if the new block is already in queue
         (in-queue
          (cl-some (lambda (queued)
                     (code-agent-org--same-block-p
                      new-custom-id new-marker
                      (plist-get queued :custom-id) (plist-get queued :marker)))
                   queue)))
    (cond
     (is-running 'running)   ; Cannot queue the running block
     (in-queue 'in-queue)    ; Already queued
     ((>= (length queue) code-agent-org-max-queue-depth)
      (message "Claude: Queue full (%d blocks). Cannot queue more."
               (length queue))
      'queue-full)
     (t
      (code-agent-org--session-put session-key :pending-queue
                               (append queue (list block-info)))
      'queued)))))

(defun code-agent-org--dequeue-block (session-key)
  "Remove and return next block from SESSION-KEY queue.
Returns nil if queue is empty."
  (let ((queue (code-agent-org--session-get session-key :pending-queue)))
    (when queue
      (code-agent-org--session-put session-key :pending-queue (cdr queue))
      (car queue))))

(defun code-agent-org--clear-queue (session-key)
  "Clear all pending blocks for SESSION-KEY."
  (code-agent-org--session-put session-key :pending-queue nil))

(defun code-agent-org--queue-count (session-key)
  "Return number of queued blocks for SESSION-KEY."
  (length (code-agent-org--session-get session-key :pending-queue)))

(defun code-agent-org--total-queue-count ()
  "Return total number of queued blocks across all sessions in current buffer."
  (let ((total 0))
    (when (hash-table-p code-agent-org--sessions)
      (maphash (lambda (_key state)
                 (when (code-agent-org--session-state-p state)
                   (cl-incf total (length (code-agent-org--session-state-get state :pending-queue)))))
               code-agent-org--sessions))
    total))
