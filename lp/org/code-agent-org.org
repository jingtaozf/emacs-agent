* Overview
:PROPERTIES:
:CUSTOM_ID: org-overview
:END:

This module owns the =code-agent-org-mode= minor mode — the public face
of =code-agent-org= that users enable in org buffers.  It wires together
the workspace-management infrastructure: cmux workspace lifecycle
(launch, restart, cancel, open tab), backend/profile selection, IDE
server management, and permission mode settings, all surfaced through a
=transient= menu (=C-c C-/=).

This file is the integration point — it loads the sub-modules
(customize, session, session-context, env-permission, project-prompts,
workspace, process) and provides the interactive surface.  The actual
workspace orchestration lives in those sub-modules under =lp/org/=.

Invariant: =code-agent-org-mode= turns on iff this module is loaded; no
other module activates the minor mode.  Cross-buffer state (session
table, active queries, mode-line counters) is owned by this file or by
sub-modules whose functions are called through the single-dash public
API surface exported here.

* Mode Definition

** Terminal Dispatchers

Dispatchers called by workspace_bridge.py to route events to the correct
terminal backend.  All use the same pattern: check if the session is
known to cmux, then dispatch accordingly.

#+BEGIN_SRC elisp
(defun code-agent-org--terminal-query-completed (workspace-session-id)
  "Dispatch query-completed to the correct terminal backend.
WORKSPACE-SESSION-ID identifies the session whose query just finished.
Tries cmux (checks hash table, then falls back to request-id file)."
  (cond
   ;; cmux: session-key in hash table (normal path)
   ((and (fboundp 'code-agent-org-cmux--query-completed)
         (gethash workspace-session-id code-agent-org-terminal--workspace-to-session-key))
    (code-agent-org-cmux--query-completed workspace-session-id))
   ;; cmux: hash empty but the org section still exists (Emacs restart /
   ;; session eviction) — recover the mapping by scanning loaded buffers,
   ;; then dispatch.  This must precede the request-id fallback: that path
   ;; re-enters with the hash still empty, so the session goes out of sync
   ;; (busy never clears, loop never continues) unless the mapping is first
   ;; rebuilt from the authoritative org section.
   ((and (fboundp 'code-agent-org-cmux--query-completed)
         (fboundp 'code-agent-org-cmux--recover-session)
         (code-agent-org-cmux--recover-session workspace-session-id))
    (code-agent-org-cmux--query-completed workspace-session-id))
   ;; cmux: hash empty, recovery failed, but request-id file exists
   ;; (module reload / race) — last-resort dispatch.
   ((and (fboundp 'code-agent-org-cmux--query-completed)
         (code-agent-org-terminal--read-request-id workspace-session-id))
    (code-agent-org-cmux--query-completed workspace-session-id))
   (t (message "code-agent-org: no terminal backend loaded for query-completed"))))

(defun code-agent-org--terminal-permission-needed (workspace-session-id tool-name)
  "Dispatch permission notification to the correct terminal backend."
  (cond
   ((and (fboundp 'code-agent-org-cmux--permission-needed)
         (boundp 'code-agent-org-terminal--workspace-to-session-key)
         (gethash workspace-session-id code-agent-org-terminal--workspace-to-session-key))
    (code-agent-org-cmux--permission-needed workspace-session-id tool-name))
   ;; Recover a lost mapping (Emacs restart / eviction) before dropping.
   ((and (fboundp 'code-agent-org-cmux--permission-needed)
         (fboundp 'code-agent-org-cmux--recover-session)
         (code-agent-org-cmux--recover-session workspace-session-id))
    (code-agent-org-cmux--permission-needed workspace-session-id tool-name))
   (t (message "code-agent-org: no terminal backend loaded for permission-needed"))))

(defun code-agent-org--terminal-permission-resolved (workspace-session-id)
  "Dispatch permission resolution to the correct terminal backend."
  (cond
   ((and (fboundp 'code-agent-org-cmux--permission-resolved)
         (boundp 'code-agent-org-terminal--workspace-to-session-key)
         (gethash workspace-session-id code-agent-org-terminal--workspace-to-session-key))
    (code-agent-org-cmux--permission-resolved workspace-session-id))
   ;; Recover a lost mapping (Emacs restart / eviction) before dropping.
   ((and (fboundp 'code-agent-org-cmux--permission-resolved)
         (fboundp 'code-agent-org-cmux--recover-session)
         (code-agent-org-cmux--recover-session workspace-session-id))
    (code-agent-org-cmux--permission-resolved workspace-session-id))
   (t (message "code-agent-org: no terminal backend loaded for permission-resolved"))))

** with-backend macro

Three things every "dispatch on the backend for point's session"
callsite repeats: read =current-session-key=, build the resolve
=options= plist with =:cwd=, and call =--resolve-backend=.  Lifting
that triple into one macro keeps callers focused on the verb they
actually want.

#+BEGIN_SRC elisp
(defmacro code-agent-org-with-backend (backend-var &rest body)
  "Bind BACKEND-VAR to the resolved backend for the current org
session and run BODY.  Resolve uses =current-session-key= +
=(:cwd default-directory)= options, the canonical preamble for
interactive callers.

Callers that already hold a session-key (e.g. =--send-request=
which receives one as a parameter) should NOT use this macro —
call =--resolve-backend= directly so the session-key flows through
explicitly."
  (declare (indent 1) (debug (symbolp body)))
  `(let* ((session-key (code-agent-org-current-session-key))
          (options (list :cwd default-directory))
          (,backend-var (code-agent-org--resolve-backend session-key options)))
     ,@body))
#+END_SRC

** Open cmux tab

Opens the terminal tab for the current session using the cmux backend.

(defun code-agent-org-open-terminal-tab ()
  "Open the terminal for this session based on the configured backend.
Dispatches through Protocol 3 `code-agent-org-backend-open-terminal-tab'
on the backend instance — multiplexer backends override to raise
their workspace / tab.  Agent backends default to a user-error
(there is no terminal to open for a JSON-stream subprocess)."
  (interactive)
  (code-agent-org-with-backend backend
    (code-agent-org-backend-open-terminal-tab backend)))

** Workspace Status Helper

Display the current workspace session identifier and its status.

#+BEGIN_SRC elisp
(defun code-agent-org-cmux-show-status ()
  "Show the status of the current cmux workspace."
  (interactive)
  (let* ((session-key (code-agent-org-current-session-key))
         (workspace-id (code-agent-org-session-get session-key :workspace-session-id)))
    (if workspace-id
        (message "Session %s using workspace %s" session-key workspace-id)
      (message "No workspace associated with current buffer"))))

** New Workspace

Insert a new workspace heading with the standard properties for cmux session management.

#+BEGIN_SRC elisp
(defun code-agent-org-new-workspace (name)
  "Create a new workspace heading named NAME at the end of the buffer.
Inserts a top-level heading with =:CMUX_WORKSPACE:= and
=:CLAUDE_SESSION_ID:= properties, then moves point there."
  (interactive "sWorkspace name: ")
  (save-excursion
    (goto-char (point-max))
    (unless (bolp) (insert "\n"))
    (insert "\n* " name "\n")
    (insert ":PROPERTIES:\n")
    (insert ":CMUX_WORKSPACE: " name "\n")
    (let ((session-id (format "sdd-%s" (format-time-string "%Y%m%d-%H%M%S"))))
      (insert ":CLAUDE_SESSION_ID: " session-id "\n"))
    (insert ":END:\n"))
  (message "Workspace '%s' created. Use C-c C-/ r to launch Claude Code." name))
#+END_SRC

** IDE Server Wrappers

Interactive wrappers for IDE server lifecycle functions that need
session-level context (workspace-session-id, claude-ide-key) that a
bare transient binding cannot provide directly.

#+BEGIN_SRC elisp
(defun code-agent-org-ide-start ()
  "Start the IDE server for the current workspace."
  (interactive)
  (let* ((session-key (code-agent-org-current-session-key))
         (workspace-id (code-agent-org-session-get session-key :workspace-session-id))
         (project-root (code-agent-org-get-project-root)))
    (if (and workspace-id project-root)
        (progn
          (code-agent-org-cmux--ensure-ide-server project-root workspace-id)
          (message "IDE server started for workspace %s" workspace-id))
      (message "No active workspace or project root"))))

(defun code-agent-org-ide-stop ()
  "Stop the IDE server for the current workspace."
  (interactive)
  (let* ((session-key (code-agent-org-current-session-key))
         (claude-ide-key (code-agent-org-session-get session-key :claude-ide-key)))
    (if claude-ide-key
        (progn
          (claude-ide-stop-server claude-ide-key)
          (message "IDE server stopped"))
      (message "No active IDE server for this session"))))
#+END_SRC

** Keymap

The keymap binds a single key to open the transient menu.
Per Emacs convention, =C-c= followed by non-letter is reserved for minor modes.

#+BEGIN_SRC elisp
(defvar code-agent-org-mode-map
  (let ((map (make-sparse-keymap)))
    ;; Primary: Transient menu (C-c C-/ is safe for minor modes per Emacs convention)
    (define-key map (kbd "C-c C-/") #'code-agent-org-menu)
    ;; Note: C-c C-c is org-babel's domain, not bound here
    map)
  "Keymap for `code-agent-org-mode'.
Use \\[code-agent-org-menu] to open the transient command menu.")
#+END_SRC

** Minor Mode

Major / minor mode owning the UI surface named in the heading.

#+BEGIN_SRC elisp
(defun code-agent-org--register-exclusion ()
  "Register code-agent-org buffers as excluded from IDE context."
  (add-to-list 'code-agent-ide-context-exclude-predicates
               (lambda (buf)
                 (with-current-buffer buf
                   (bound-and-true-p code-agent-org-mode)))))

;;;###autoload
(define-minor-mode code-agent-org-mode
  "Minor mode for Claude Code workspace management in org-mode.

Key bindings:
  \\[code-agent-org-menu]     Open transient command menu

\\{code-agent-org-mode-map}"
  :lighter " ClaudeOrg"
  :keymap code-agent-org-mode-map
  :group 'code-agent-org
  (if code-agent-org-mode
      (progn
        ;; Save original header line
        (setq code-agent-org--original-header-line header-line-format)
        ;; Save and disable line truncation (so long lines wrap)
        (setq code-agent-org--original-truncate-lines truncate-lines)
        (setq truncate-lines nil)
        ;; Set up header line
        (code-agent-org-refresh-header-line)
        ;; Track cursor position for workspace badge
        (add-hook 'post-command-hook
                  #'code-agent-org--schedule-cursor-header-update nil t)
        ;; Register exclusion from IDE context
        (code-agent-org--register-exclusion)
        ;; Auto-start MCP server
        (code-agent-org--ensure-mcp-server)
        ;; Proactively start the OTel trace bridge so spans flow from the
        ;; first session rather than only lazily on the first Emacs-side
        ;; traced span — a cmux-only workflow may never emit one, leaving
        ;; the bridge dead (and Phoenix silent) until a manual restart.
        ;; `--ensure-bridge' is idempotent and self-gates on
        ;; `code-agent-trace-enabled'.
        (when (fboundp 'code-agent-trace--ensure-bridge)
          (ignore-errors (code-agent-trace--ensure-bridge)))
        ;; Add buffer kill hook to clean up resources
        (add-hook 'kill-buffer-hook #'code-agent-org--on-buffer-kill nil t))
    ;; Restore original header line
    (setq header-line-format code-agent-org--original-header-line)
    ;; Restore original truncate-lines
    (setq truncate-lines code-agent-org--original-truncate-lines)
    ;; Disconnect persistent clients for this buffer
    (code-agent-org-persistent-registry-disconnect-buffer
     code-agent-org--persistent-registry (current-buffer))
    ;; Stop header timer if no more code-agent-org buffers
    (code-agent-org--stop-header-timer)
    ;; Cancel cursor-position debounce timer
    (when code-agent-org--header-cursor-timer
      (cancel-timer code-agent-org--header-cursor-timer)
      (setq code-agent-org--header-cursor-timer nil))
    (remove-hook 'post-command-hook
                 #'code-agent-org--schedule-cursor-header-update t)
    ;; Remove buffer-local hooks
    (remove-hook 'kill-buffer-hook #'code-agent-org--on-buffer-kill t)))

;;;###autoload
(defun code-agent-org-cleanup ()
  "Clean up code-agent-org resources."
  (interactive)
  (clrhash code-agent-org--sessions))
