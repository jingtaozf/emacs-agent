#+TITLE: Code Agent Org — Customization
#+PROPERTY: literate-load yes

* Overview
:PROPERTIES:
:CUSTOM_ID: overview
:END:

All user-tunable knobs for =code-agent-org=: the =defgroup=, every
=defcustom= and =defconst=, the three execution hooks (=execute= /
=execute-after= / =complete=), and the small set of side-effecting
helpers attached to those hooks (macOS notifier, loop-detection
record/inject/consume).

Extracted out of =code-agent-org.org= so the configuration surface is
readable end-to-end without scrolling through the 5K-line main
module. Loaded *first* in =code-agent-org.org='s module bootstrap so
every subsequent module sees the customs/hooks/property constants
already bound.

This module is invariant-light by design: nothing here decides
control flow at load time. The few =defun=​s in this file are *hook
attachments* — they exist to live alongside the hook =defvar= they
extend, since the hook contract (signature + status symbols) is the
public surface defined here.

** macOS Notifications

Send a desktop notification with sound when a query completes, pulling
the user back even when looking at another app. Hooking
=code-agent-org-complete-hook= rather than each backend means every
backend (JSON stream, cmux, tmux) gets notifications for free with no
per-backend wiring.

One =osascript= call delivers both banner + sound, non-blocking via
=start-process=. Two non-goals accepted: the banner always shows
"Script Editor" as the sender (osascript's identity; would need a
bundled .app to override), and only the completion event is surfaced
— intermediate streaming/tool-use/permission would be noise. Linux /
Windows users get the same =complete-hook= and can layer their own
notifier. The defcustoms above gate behaviour and pick from the
standard macOS sound set (Basso / Blow / Bottle / … / Tink).

#+BEGIN_SRC elisp
(defun code-agent-org--notify (title message &optional sound)
  "Send a macOS desktop notification with optional SOUND. Non-blocking."
  (when (eq system-type 'darwin)
    (let* ((sound-clause (if sound
                             (format " sound name %S" sound)
                           ""))
           (script (format "display notification %S with title %S%s"
                           message title sound-clause)))
      (start-process "claude-notify" nil "osascript" "-e" script))))

(defun code-agent-org--notify-on-complete (session-key _result status)
  "Send macOS notification on query completion.
STATUS is `completed' or `error'.  Added to `code-agent-org-complete-hook'."
  (when code-agent-org-notify-on-complete
    (let* ((title (if (eq status 'error)
                      "Claude: Error"
                    "Claude: Complete"))
           (message (format "%s" (or (buffer-name) "query"))))
      (code-agent-org--notify title message code-agent-org-notify-sound))))

(add-hook 'code-agent-org-complete-hook #'code-agent-org--notify-on-complete)
#+END_SRC

*** Loop Detection Utilities

Helper functions for the loop iteration hook system. These are building
blocks that hook functions can use — no detection logic runs by default.

#+BEGIN_SRC elisp
(defun code-agent-org--record-tool-call (session-key tool-name tool-input)
  "Record a tool call in the session's :tool-call-history.
Appends (TOOL-NAME . ARGS-HASH) to the history list."
  (let* ((args-hash (sxhash tool-input))
         (entry (cons tool-name args-hash))
         (history (code-agent-org--session-get session-key :tool-call-history)))
    (code-agent-org--session-put session-key :tool-call-history
                             (append history (list entry)))))

(defun code-agent-org-loop-inject-warning (session-key message)
  "Inject a warning MESSAGE into the next loop iteration for SESSION-KEY.
The warning will be included in the system prompt and cleared after use."
  (code-agent-org--session-put session-key :loop-warning message))

(defun code-agent-org--consume-loop-warning (session-key)
  "Read and clear the :loop-warning from SESSION-KEY.
Returns the warning string, or nil if none."
  (let ((warning (code-agent-org--session-get session-key :loop-warning)))
    (when warning
      (code-agent-org--session-put session-key :loop-warning nil))
    warning))
#+END_SRC

** Pre-Completion Utilities

Mid-stream insertions (warnings, debug notes, error envelopes) need
to land in the right response section *before* the Stop hook
finalises it.  =append-to-response= finds the response heading by
=:QUERY_ID:= property and appends without disturbing the user's
scroll position.

#+BEGIN_SRC elisp
(defun code-agent-org-append-to-response (session-key query-id text)
  "Append TEXT to the response section identified by QUERY-ID.
Finds the response heading with matching :QUERY_ID: property,
navigates to the end of its subtree, and inserts TEXT.
Preserves window-start (scroll position)."
  (let ((ws (window-start)))
    (prog1
        (save-excursion
          (goto-char (point-min))
          (condition-case nil
              (when (re-search-forward
                     (format ":QUERY_ID: %s" (regexp-quote query-id)) nil t)
                (org-back-to-heading t)
                (org-end-of-subtree t)
                (insert text)
                t)
            (error nil)))
      (set-window-start nil ws t))))
#+END_SRC

** Telemetry Persistence

Token-usage data from the backend's =Stop= event lands here and gets
stamped as =:INPUT_TOKENS:= / =:OUTPUT_TOKENS:= / =:CACHE_READ_TOKENS:=
properties on the response heading.  Persisting in the org buffer
means the user can grep their conversation history for cost data
weeks later without consulting external logs.

#+BEGIN_SRC elisp
(defun code-agent-org--persist-telemetry (session-key usage-plist)
  "Persist token usage from USAGE-PLIST to org properties for SESSION-KEY.
USAGE-PLIST has keys :input-tokens, :output-tokens, :cache-read-tokens."
  (let ((query-id (code-agent-org--session-get session-key :query-id)))
    (when query-id
      (save-excursion
        (goto-char (point-min))
        (when (re-search-forward
               (format ":QUERY_ID: %s" (regexp-quote query-id)) nil t)
          (org-back-to-heading t)
          (let ((input (plist-get usage-plist :input-tokens))
                (output (plist-get usage-plist :output-tokens))
                (cache (plist-get usage-plist :cache-read-tokens)))
            (when input
              (org-entry-put nil "INPUT_TOKENS" (number-to-string input)))
            (when output
              (org-entry-put nil "OUTPUT_TOKENS" (number-to-string output)))
            (when cache
              (org-entry-put nil "CACHE_READ_TOKENS" (number-to-string cache)))))))))

(defun code-agent-org--persist-duration (session-key)
  "Persist query duration to org property for SESSION-KEY.
Computes duration from :start-time to now."
  (let ((start-time (code-agent-org--session-get session-key :start-time))
        (query-id (code-agent-org--session-get session-key :query-id)))
    (when (and start-time query-id)
      (let ((duration (- (float-time) start-time)))
        (save-excursion
          (goto-char (point-min))
          (when (re-search-forward
                 (format ":QUERY_ID: %s" (regexp-quote query-id)) nil t)
            (org-back-to-heading t)
            (org-entry-put nil "DURATION_SECS"
                           (format "%.1f" duration))))))))

(defun code-agent-org--persist-tool-count (session-key)
  "Persist tool call count to org property for SESSION-KEY."
  (let ((count (or (code-agent-org--session-get session-key :tool-call-count) 0))
        (query-id (code-agent-org--session-get session-key :query-id)))
    (when query-id
      (save-excursion
        (goto-char (point-min))
        (when (re-search-forward
               (format ":QUERY_ID: %s" (regexp-quote query-id)) nil t)
          (org-back-to-heading t)
          (org-entry-put nil "TOOL_CALLS" (number-to-string count)))))))
#+END_SRC

** File-Based Prompt Templates

Prompt templates are stored as org files under =prompts/= for easier
editing and hot-reloading. Changes to template files take effect immediately
without restarting Emacs or reloading code.

Directory structure:
#+begin_example
prompts/
├── system/          # System prompts (loaded in sorted order)
│   ├── 00-base.org  # Core org-mode instructions
│   └── 10-links.org # Link resolution prompt
├── tags/            # SDD workflow phases
│   ├── sdd.org      # Container workflow
│   ├── research.org, design.org, planning.org, implementation.org
└── headers/         # Header argument behaviors
    ├── phase/       # Enum-style (subdirectory with value files)
    └── *.org        # Template files (tests.org, coverage.org, etc.)
#+end_example

(defcustom code-agent-org-prompts-directory
  (let* ((from (file-name-directory (or load-file-name
                                        (locate-library "code-agent-org")
                                        buffer-file-name
                                        default-directory)))
         ;; After the 2026-05-26 layout restructure, this file lives

         ;; up to the `code-agent.el` sentinel so the lookup works

  "Directory containing prompt template files.

  system/  - System prompts loaded in sorted order
  tags/    - Tag behavior prompts (TAG.org)
  headers/ - Header argument prompts (HEADER.org or HEADER/VALUE.org)"
  :type 'directory
  :group 'code-agent-org)

*** Prompt File Loader

#+NAME: prompt-file-loader
#+BEGIN_SRC elisp
(defun code-agent-org--load-prompt-file (file)
  "Load prompt content from FILE.
Returns file content as trimmed string, or nil if file doesn't exist."
  (when (file-readable-p file)
    (with-temp-buffer
      (insert-file-contents file)
      (string-trim (buffer-string)))))

(defun code-agent-org--load-system-prompts ()
  "Load all system prompts from system/ directory.
Files are loaded in sorted order (use numeric prefix for ordering).
Returns list of prompt strings."
  (let ((dir (expand-file-name "system" code-agent-org-prompts-directory)))
    (when (file-directory-p dir)
      (let ((files (directory-files dir t "\\.org\\'" t)))
        (delq nil (mapcar #'code-agent-org--load-prompt-file (sort files #'string<)))))))
#+END_SRC
