#+TITLE: Claude Agent Permission Module
#+PROPERTY: literate-load yes

* Overview

This module owns Emacs's side of the *tool permission decision* — when
the agent asks "may I run this Bash / Edit this file / WebFetch this
URL", the answer comes from here. Three layers stack to produce a
verdict:

| Layer | What it provides | When it fires |
|-------+------------------+---------------|
| Pattern lists | Allow / deny / ask globs scoped per tool | Before the interactive prompt; first match wins |
| Cache | Per-session decisions (e.g. "always allow Bash(npm test)") | Between prompts in the same session, so the user isn't asked twice |
| Interactive prompt | =completing-read= y/n/always with the decision threaded back through the agent's =can_use_tool= control-request or ACP =session/request_permission= | Whenever the prior layers don't produce a definite answer |

The dispatch entry point is =code-agent--run-permission-functions= —
which the JSON-stream agent backend calls from its control-request
handler and the ACP backend calls from its permission-request handler.
It runs each function in =code-agent-permission-functions= in order
and returns the first non-nil result as a plist (=:behavior "allow"= /
=:behavior "deny"= plus optional =:message=).

Three invariants the rest of the system relies on:

- *Deny on error.* If a permission function signals, the result is
  =deny=, never =allow=. Tested in =test-permission-round-trip.el=.
- *Cache is per-session-key.* No global cache; the cache lives on
  the session struct so different stories never share permission
  decisions.
- *Pattern hooks are user-extensible.* Project-specific allow/deny
  rules live in =.dir-locals.el= or user init via
  =code-agent-permission-functions=; this module ships defaults
  but never overrides user functions.

* Dependencies

Loaded eagerly at file evaluation time — symbols below assume every required feature is already provided.

** Permission Hook

When Claude CLI requests permission for a tool (via =acceptEdits= mode),
functions in =code-agent-permission-functions= are called in order.
Each function can return:
- =nil= - No decision, continue to next function
- =(:behavior "allow")= - Allow the tool
- =(:behavior "deny" :message "...")= - Deny with message

Built-in functions:
- =code-agent-permission-ask-user-question= - Handle AskUserQuestion tool interactively
- =code-agent-permission-check-patterns= - Pattern-based allow/deny
- =code-agent-permission-prompt= - Interactive y-or-n-p
- =code-agent-permission-auto-allow= - Auto-approve everything

#+BEGIN_SRC elisp
(defvar code-agent-permission-functions
  '(code-agent-permission-ask-user-question
    code-agent-permission-check-patterns
    code-agent-permission-prompt)
  "Hook for permission checking when Claude requests tool access.

Functions are called with three arguments:
  TOOL-NAME: string like \"Read\", \"Write\", \"Bash\"
  TOOL-INPUT: plist with tool-specific parameters
  CONTEXT: plist with :session-id, :context-label, :suggestions

Each function should return:
  nil                            - No decision, try next function
  (:behavior \"allow\")           - Allow the tool
  (:behavior \"deny\" :message ..) - Deny with message

The first non-nil result is used. If all return nil, tool is allowed.

Example - Add custom protection:
  (add-hook 'code-agent-permission-functions
            #'my-protect-sensitive-files t)  ; append

Example - Replace default behavior:
  (setq code-agent-permission-functions
        '(my-custom-checker code-agent-permission-auto-allow))")
(defun code-agent--validate-permission-result (result)
  "Validate that RESULT is a proper permission result.
Returns RESULT if valid, nil if invalid.
Valid results: nil, or plist with :behavior being \"allow\" or \"deny\"."
  (cond
   ;; nil is valid (pass to next function)
   ((null result) nil)
   ;; Must be a list (plist)
   ((not (listp result)) nil)
   ;; Must have :behavior key
   ((not (plist-member result :behavior)) nil)
   ;; :behavior must be string "allow" or "deny"
   ((not (stringp (plist-get result :behavior))) nil)
   ((not (member (plist-get result :behavior) '("allow" "deny"))) nil)
   ;; Valid
   (t result)))

(defun code-agent--run-permission-functions (tool-name tool-input context)
  "Run permission functions hook and return first non-nil result.

TOOL-NAME: string like \"Read\", \"Write\", \"Bash\"
TOOL-INPUT: plist with tool-specific parameters
CONTEXT: plist with :session-id, :context-label, :suggestions

Returns: plist with :behavior (\"allow\" or \"deny\") and optional :message

SECURITY: This function DENIES on any error. This prevents the vulnerability
where permission function crashes could silently allow dangerous operations."
  (let ((had-error nil)
        (error-fn nil)
        (error-msg nil))
    (catch 'permission-result
      (dolist (fn code-agent-permission-functions)
        (when (functionp fn)
          (condition-case err
              (let* ((raw-result (funcall fn tool-name tool-input context))
                     (result (code-agent--validate-permission-result raw-result)))
                ;; Check for invalid return type
                (when (and raw-result (not result))
                  (setq had-error t
                        error-fn fn
                        error-msg (format "Invalid return type: %S" (type-of raw-result)))
                  (message "Permission function %s returned invalid result: %S" fn raw-result))
                (when result
                  (throw 'permission-result result)))
            (error
             (setq had-error t
                   error-fn fn
                   error-msg (error-message-string err))
             (message "Permission function %s error: %s" fn error-msg)))))
      ;; SECURITY: If any error occurred, DENY (not allow)
      (if had-error
          (list :behavior "deny"
                :reason "permission-error"
                :message (format "Permission function error: %s" error-msg))
        ;; Default: allow (only when no errors occurred)
        (list :behavior "allow")))))
(defun code-agent-permission-check-patterns (tool-name tool-input _context)
  "Check TOOL-NAME against allow/deny patterns.
Uses `code-agent-allow-patterns' and `code-agent-deny-patterns'.
Returns permission result or nil to pass to next function."
  (let ((result (code-agent-check-permission
                 tool-name tool-input
                 code-agent-allow-patterns
                 code-agent-deny-patterns)))
    (pcase result
      ('allow (list :behavior "allow"))
      ('deny (list :behavior "deny"
                   :message (format "Tool %s denied by pattern" tool-name)))
      ;; 'ask or nil -> return nil to let next function decide
      (_ nil))))

*** Permission Cache (Per-Session)

Each session's decisions are persisted to a separate file.

#+BEGIN_SRC elisp
(defun code-agent--permission-cache-file (session-id)
  "Return cache file path for SESSION-ID."
  (when session-id
    (expand-file-name (concat session-id ".el") code-agent-permission-cache-dir)))

(defun code-agent--permission-cache-load (session-id)
  "Load permission cache for SESSION-ID. Returns hash-table."
  (let ((file (code-agent--permission-cache-file session-id)))
    (if (and file (file-exists-p file))
        (condition-case nil
            (with-temp-buffer
              (insert-file-contents file)
              (read (current-buffer)))
          (error (make-hash-table :test 'equal)))
      (make-hash-table :test 'equal))))

(defun code-agent--permission-cache-save (session-id cache)
  "Save CACHE hash-table for SESSION-ID."
  (when session-id
    (let ((file (code-agent--permission-cache-file session-id)))
      (make-directory (file-name-directory file) t)
      (with-temp-file file
        (prin1 cache (current-buffer))))))

(defun code-agent--permission-cache-key (tool-name tool-input)
  "Generate cache key from TOOL-NAME and TOOL-INPUT.
Groups similar operations by directory or command prefix."
  (let ((file-path (plist-get tool-input :file_path))
        (command (plist-get tool-input :command)))
    (cond
     (file-path (format "%s:%s" tool-name (file-name-directory file-path)))
     (command (format "%s:%s" tool-name (car (split-string command))))
     (t tool-name))))
#+END_SRC

     (t tool-name))))

*** Default Implementation: Interactive Prompt

The interactive prompt needs to handle the CLI's response timeout gracefully.
If the user doesn't respond in time, we auto-allow rather than killing the session.

#+BEGIN_SRC elisp
(defcustom code-agent-permission-prompt-timeout 5
  "Timeout in seconds for permission prompts.
If user doesn't respond within this time, the tool is auto-allowed
to prevent CLI timeout and session death.
Set to nil to disable timeout (use standard blocking y-or-n-p)."
  :type '(choice (number :tag "Timeout in seconds")
                 (const :tag "No timeout (blocking)" nil))
  :group 'code-agent)

(defun code-agent--permission-prompt-with-timeout (prompt timeout)
  "Prompt user with PROMPT, auto-allow after TIMEOUT seconds.
Returns t for allow, nil for deny.
If TIMEOUT is nil, uses standard blocking y-or-n-p."
  (if (null timeout)
      ;; No timeout - use standard blocking prompt
      (y-or-n-p prompt)
    ;; Timeout version - auto-allow if no response
    (let ((answer nil)
          (done nil)
          (start-time (float-time)))
      (message "%s (auto-allow in %ds) " prompt timeout)
      ;; Check for input with timeout
      (while (and (not done)
                  (< (- (float-time) start-time) timeout))
        (when (input-pending-p)
          (let ((char (read-char)))
            (cond
             ((memq char '(?y ?Y ?\s))  ; y, Y, or space = yes
              (setq answer t done t))
             ((memq char '(?n ?N ?\d))  ; n, N, or DEL = no
              (setq answer nil done t))
             (t
              ;; Invalid key, remind user
              (message "%s (y or n, auto-allow in %ds) "
                       prompt (- timeout (truncate (- (float-time) start-time))))))))
        (unless done
          (sit-for 0.1 t)))  ; Check every 100ms, allow redisplay
      (if done
          (progn
            (message "%s %s" prompt (if answer "y" "n"))
            answer)
        ;; Timeout - auto-allow
        (message "%s auto-allowed (timeout)" prompt)
        t))))

(defun code-agent-permission-prompt (tool-name tool-input context)
  "Prompt user with timeout, cache decision per session.
Auto-allows after `code-agent-permission-prompt-timeout' seconds
to prevent CLI timeout and session termination."
  (let* ((session-id (plist-get context :session-id))
         (cache (code-agent--permission-cache-load session-id))
         (key (code-agent--permission-cache-key tool-name tool-input))
         (cached (gethash key cache)))
    (if cached
        ;; Return cached decision
        (if (eq (car cached) 'allow)
            '(:behavior "allow")
          '(:behavior "deny" :message "Previously denied"))
      ;; Prompt user with timeout
      (let* ((desc (code-agent--describe-tool-use tool-name tool-input))
             (section (plist-get context :context-label))
             (prompt (if section
                         (format "[%s] Allow %s?" section desc)
                       (format "Allow %s?" desc)))
             (allowed (code-agent--permission-prompt-with-timeout
                       prompt code-agent-permission-prompt-timeout)))
        ;; Cache and save
        (puthash key (cons (if allowed 'allow 'deny) (current-time)) cache)
        (code-agent--permission-cache-save session-id cache)
        (if allowed
            '(:behavior "allow")
          '(:behavior "deny" :message "User denied"))))))
#+END_SRC

*** AskUserQuestion Tool Handling

When Claude needs clarification, it calls the =AskUserQuestion= tool which triggers
the =can_use_tool= permission flow. This function intercepts those requests and
presents questions to the user using =completing-read=.

#+BEGIN_SRC elisp
(defvar code-agent-ask-user-question-mock-answers nil
  "When non-nil, use mock answers instead of prompting user.
This is for testing purposes. Format is an alist mapping question text
to answer string. Example:
  ((\"How should I format?\" . \"Summary\")
   (\"Which features?\" . \"Option1, Option2\"))")

(defun code-agent--format-question-options (options)
  "Format OPTIONS list for completing-read.
Each option is a plist with :label and :description.
Returns alist of (\"LABEL - DESCRIPTION\" . \"LABEL\")."
  (mapcar (lambda (opt)
            (let ((label (plist-get opt :label))
                  (desc (plist-get opt :description)))
              (cons (if desc
                        (format "%s - %s" label desc)
                      label)
                    label)))
          options))

(defun code-agent--ask-single-question (question options header &optional session-name)
  "Ask a single-select QUESTION with OPTIONS.
HEADER is a short label for the question.
SESSION-NAME is optional session/story identifier to show in prompt.
Returns the selected label string, or user's free-text if \"Other\" chosen."
  (let* ((formatted (code-agent--format-question-options options))
         (choices (append (mapcar #'car formatted)
                          '("Other (type your answer)")))
         (prefix (if session-name
                     (format "[%s|%s]" session-name header)
                   (format "[%s]" header)))
         (prompt (format "%s %s " prefix question))
         (selection (completing-read prompt choices nil t)))
    (if (string= selection "Other (type your answer)")
        ;; Free-text input
        (read-string (format "%s Enter your answer: " prefix))
      ;; Return the label for the selected option
      (or (cdr (assoc selection formatted))
          selection))))

(defun code-agent--ask-multi-question (question options header &optional session-name)
  "Ask a multi-select QUESTION with OPTIONS.
HEADER is a short label for the question.
SESSION-NAME is optional session/story identifier to show in prompt.
Returns comma-separated labels, or user's free-text if \"Other\" chosen."
  (let* ((formatted (code-agent--format-question-options options))
         (choices (append (mapcar #'car formatted)
                          '("Other (type your answer)")))
         (prefix (if session-name
                     (format "[%s|%s]" session-name header)
                   (format "[%s]" header)))
         (prompt (format "%s %s (comma-separated): " prefix question))
         ;; Use completing-read-multiple for multi-select
         (selections (completing-read-multiple prompt choices nil t)))
    (if (member "Other (type your answer)" selections)
        ;; Free-text input
        (read-string (format "%s Enter your answer: " prefix))
      ;; Map display strings back to labels and join
      (mapconcat (lambda (sel)
                   (or (cdr (assoc sel formatted)) sel))
                 selections
                 ", "))))

(defun code-agent--get-mock-answer (question-text)
  "Get mock answer for QUESTION-TEXT from `code-agent-ask-user-question-mock-answers'.
Returns the answer string if found, nil otherwise."
  (when code-agent-ask-user-question-mock-answers
    (cdr (assoc question-text code-agent-ask-user-question-mock-answers))))

(defun code-agent-permission-ask-user-question (tool-name tool-input context)
  "Handle AskUserQuestion tool by prompting user interactively.

When TOOL-NAME is \"AskUserQuestion\", parse questions from TOOL-INPUT,
present them to user via completing-read, and return answers.

CONTEXT provides session information including :context-label (the story/session name).

If `code-agent-ask-user-question-mock-answers' is non-nil, use mock
answers for testing instead of prompting the user.

Returns:
  - Plist with :behavior \"allow\" and :updated-input containing answers
  - Or (:behavior \"deny\" :message ...) if user cancels with C-g
  - Or nil if not an AskUserQuestion tool (pass to next handler)"
  (when (string= tool-name "AskUserQuestion")
    (condition-case nil
        (let* ((questions (plist-get tool-input :questions))
               (session-name (plist-get context :context-label))
               (answers (make-hash-table :test 'equal)))
          (unless questions
            (error "AskUserQuestion tool missing 'questions' field"))
          ;; Process each question
          (dolist (q questions)
            (let* ((question-text (plist-get q :question))
                   (header (or (plist-get q :header) "Question"))
                   (options (plist-get q :options))
                   (multi-select (plist-get q :multiSelect))
                   ;; Check for mock answer first
                   (mock-answer (code-agent--get-mock-answer question-text))
                   (answer (or mock-answer
                               (if multi-select
                                   (code-agent--ask-multi-question
                                    question-text options header session-name)
                                 (code-agent--ask-single-question
                                  question-text options header session-name)))))
              (puthash question-text answer answers)))
          ;; Build response with questions passthrough and answers
          ;; Convert hash-table to alist for JSON serialization
          (let ((answers-alist nil))
            (maphash (lambda (k v) (push (cons k v) answers-alist)) answers)
            (list :behavior "allow"
                  :updated-input (list :questions questions
                                       :answers answers-alist))))
      ;; User cancelled with C-g
      (quit
       (list :behavior "deny"
             :message "User cancelled the question prompt")))))

** Pattern Matching

#+BEGIN_SRC elisp
(defun code-agent--get-tool-first-arg (tool-name tool-input)
  "Get the first string argument from TOOL-INPUT for pattern matching.
Different tools have different primary arguments."
  (cond
   ;; File operations use file_path
   ((member tool-name '("Read" "Write" "Edit" "MultiEdit"))
    (plist-get tool-input :file_path))
   ;; Glob uses pattern
   ((string= tool-name "Glob")
    (plist-get tool-input :pattern))
   ;; Grep uses path or pattern
   ((string= tool-name "Grep")
    (or (plist-get tool-input :path)
        (plist-get tool-input :pattern)))
   ;; Bash uses command
   ((string= tool-name "Bash")
    (plist-get tool-input :command))
   ;; WebSearch uses query
   ((string= tool-name "WebSearch")
    (plist-get tool-input :query))
   ;; WebFetch uses url
   ((string= tool-name "WebFetch")
    (plist-get tool-input :url))
   ;; Task uses subagent_type
   ((string= tool-name "Task")
    (plist-get tool-input :subagent_type))
   ;; TodoWrite uses first todo content
   ((string= tool-name "TodoWrite")
    (let ((todos (plist-get tool-input :todos)))
      (when (and todos (listp todos) (car todos))
        (plist-get (car todos) :content))))
   ;; Default: find first string value
   (t
    (cl-loop for val in tool-input
             when (stringp val)
             return val))))

(defun code-agent-permission-match-p (tool-name tool-input pattern)
  "Check if TOOL-NAME with TOOL-INPUT matches PATTERN.
Patterns can be:
  - \"*\" matches all tools
  - \"ToolName\" matches tool by name only
  - \"ToolName(**)\" matches tool with any arguments
  - \"ToolName(arg-pattern)\" matches tool with specific argument pattern
  - \"mcp__server__*\" matches MCP tools with glob"
  (cond
   ;; Wildcard matches everything
   ((string= pattern "*") t)

   ;; Glob pattern for tool names (e.g., mcp__emacs__*)
   ((and (string-match-p "\\*" pattern)
         (not (string-match-p "(" pattern)))
    (let ((regex (concat "^"
                         (replace-regexp-in-string "\\*" ".*" (regexp-quote pattern))
                         "$")))
      ;; Remove double escaping from the replacement
      (setq regex (replace-regexp-in-string "\\\\\\.\\*" ".*" regex))
      (string-match-p regex tool-name)))

   ;; Pattern with arguments: ToolName(args)
   ((string-match "^\\([A-Za-z_][A-Za-z0-9_]*\\)(\\(.*\\))$" pattern)
    (let ((pattern-tool (match-string 1 pattern))
          (pattern-args (match-string 2 pattern)))
      (and (string= tool-name pattern-tool)
           (code-agent--match-arg-pattern
            (code-agent--get-tool-first-arg tool-name tool-input)
            pattern-args))))

   ;; Simple tool name match
   ((string-match "^[A-Za-z_][A-Za-z0-9_]*$" pattern)
    (string= tool-name pattern))

   ;; No match
   (t nil)))

(defun code-agent--match-arg-pattern (arg pattern)
  "Check if ARG matches PATTERN.
Supports:
  - \"**\" matches anything
  - \"prefix:*\" matches args starting with prefix
  - \"*\" at end for prefix matching
  - Literal string matching"
  (when (and arg pattern)
    (cond
     ;; ** matches anything
     ((string= pattern "**") t)

     ;; prefix:* pattern (e.g., git:* matches 'git status')
     ((string-match "^\\([^:]+\\):\\*$" pattern)
      (let ((prefix (match-string 1 pattern)))
        (string-prefix-p prefix arg)))

     ;; Ends with * for prefix matching
     ((string-suffix-p "*" pattern)
      (string-prefix-p (substring pattern 0 -1) arg))

     ;; Contains ** for glob-style matching
     ((string-match-p "\\*\\*" pattern)
      (let ((regex (concat "^"
                           (replace-regexp-in-string
                            "\\*\\*" ".*"
                            (regexp-quote pattern))
                           "$")))
        (setq regex (replace-regexp-in-string "\\\\\\.\\*" ".*" regex))
        (string-match-p regex arg)))

     ;; Literal match
     (t (string= arg pattern)))))
#+END_SRC

** Permission Checking

#+BEGIN_SRC elisp
(defun code-agent-check-permission (tool-name tool-input allow-patterns deny-patterns)
  "Check if TOOL-NAME with TOOL-INPUT should be allowed.
Returns 'allow, 'deny, or 'ask.
Deny patterns are checked first and take precedence."
  (catch 'result
    ;; Always check default deny patterns first
    (dolist (pattern code-agent--default-deny-patterns)
      (when (code-agent-permission-match-p tool-name tool-input pattern)
        (throw 'result 'deny)))
    ;; Check user deny patterns
    (dolist (pattern deny-patterns)
      (when (code-agent-permission-match-p tool-name tool-input pattern)
        (throw 'result 'deny)))
    ;; Check allow patterns
    (dolist (pattern allow-patterns)
      (when (code-agent-permission-match-p tool-name tool-input pattern)
        (throw 'result 'allow)))
    ;; Default to ask
    'ask))

(defun code-agent-get-effective-permissions ()
  "Get effective permissions based on current preset.
Returns plist with :allow and :deny lists."
