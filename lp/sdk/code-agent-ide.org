#+PROPERTY: literate-load yes

* Overview
:PROPERTIES:
:CUSTOM_ID: ide-overview
:END:

This module collects IDE state from Emacs — which file the user is
editing, where the cursor sits, what region is selected, which
buffers are visible — for consumers like the header-line display
(=code-agent-org-header-line.org=).  The =<system-reminder>=-building
half of this module (matching the Python backend's prompt-envelope
format) was deleted 2026-07: zero production callers survived the
org-as-control-plane pivot, since prompts now travel through the
cmux/tmux terminal backends rather than an Emacs-side query API.

Invariant: collected context is built deterministically from the
active buffer + region + visible buffer list at the moment
=code-agent-collect-ide-context= is called.  Exclusions (chat buffer,
=*Messages*=, =*scratch*=, user-configured patterns) MUST be applied
before any buffer content reaches a consumer; secrets-in-buffer
leakage is the failure mode this module was added to prevent.

* Dependencies

Loaded eagerly at file evaluation time — symbols below assume every required feature is already provided.

* IDE Context

** Buffer Filtering

Filter pipeline applied before buffer reaches downstream consumers.

#+BEGIN_SRC elisp
(defcustom code-agent-ide-context-exclude-buffer-names
  '("*Messages*" "*scratch*" "*Completions*")
  "List of buffer name patterns to exclude from IDE context.
Patterns are matched using `string-match-p' (regexp)."
  :type '(repeat string)
  :group 'code-agent)

(defcustom code-agent-ide-context-exclude-modes
  '(special-mode dired-mode)
  "List of major modes to exclude from IDE context."
  :type '(repeat symbol)
  :group 'code-agent)

(defvar code-agent-ide-context-exclude-predicates nil
  "List of predicate functions for excluding buffers from IDE context.
Each function takes a buffer as argument and returns non-nil to exclude it.
This allows dynamic exclusion, e.g., for code-agent-org-mode buffers.")

(defun code-agent--buffer-excluded-p (buffer)
  "Check if BUFFER should be excluded from IDE context.
Returns non-nil if buffer matches any exclusion criteria."
  (or
   ;; Check buffer name patterns
   (cl-some (lambda (pattern)
              (string-match-p pattern (buffer-name buffer)))
            code-agent-ide-context-exclude-buffer-names)
   ;; Check major mode
   (with-current-buffer buffer
     (cl-some (lambda (mode)
                (derived-mode-p mode))
              code-agent-ide-context-exclude-modes))
   ;; Check custom predicates
   (cl-some (lambda (pred)
              (funcall pred buffer))
            code-agent-ide-context-exclude-predicates)))
#+END_SRC

** Buffer Information

Builds the per-buffer slice of the IDE-context payload — buffer name, file, modes, modified flag, point.

#+BEGIN_SRC elisp
(defun code-agent--format-buffer-info (buffer)
  "Format BUFFER info as plist with :name :path :modified."
  (with-current-buffer buffer
    (list :name (buffer-name)
          :path (buffer-file-name)
          :modified (buffer-modified-p))))

(defun code-agent--get-file-buffers (&optional limit)
  "Get list of file-visiting buffers, up to LIMIT (default 10).
Excludes buffers matching `code-agent-ide-context-exclude-buffer-names',
`code-agent-ide-context-exclude-modes', or any predicate in
`code-agent-ide-context-exclude-predicates'."
  (let ((limit (or limit 10))
        (buffers nil))
    (dolist (buf (buffer-list))
      (when (and (buffer-file-name buf)
                 (not (code-agent--buffer-excluded-p buf))
                 (< (length buffers) limit))
        (push (code-agent--format-buffer-info buf) buffers)))
    (nreverse buffers)))
#+END_SRC

** Selection Formatting

Pretty-printing for selection — used by the verbose buffer + status surfaces.

#+BEGIN_SRC elisp
(defun code-agent--format-selection ()
  "Format current selection as plist with :text :start-line :end-line.
Returns nil if no region active. Truncates text at 500 chars."
  (when (use-region-p)
    (let* ((start (region-beginning))
           (end (region-end))
           (text (buffer-substring-no-properties start end))
           (start-line (line-number-at-pos start))
           (end-line (line-number-at-pos end)))
      (list :text (if (> (length text) 500)
                      (concat (substring text 0 500) "...")
                    text)
            :start-line start-line
            :end-line end-line))))
#+END_SRC

** IDE Context Collection

Collects buffers + selection + visible window list into the =<system-reminder>= envelope.  Filters honoured first so excluded buffers never reach the serialiser.

#+BEGIN_SRC elisp
(defun code-agent--get-current-file-buffer ()
  "Get the most recently used file buffer that is not excluded.
Returns the buffer object or nil if none found.
This finds the most recently visited file buffer from `buffer-list',
which works correctly even when called from a special buffer like Claude Chat."
  (cl-loop for buf in (buffer-list)
           when (and (buffer-file-name buf)
                     (not (code-agent--buffer-excluded-p buf)))
           return buf))

(defun code-agent-collect-ide-context (&optional cwd)
  "Collect IDE context from current Emacs state.
Returns plist with:
  :current-file - Most recent non-excluded file buffer info (name, path, modified)
  :open-files   - List of file-visiting buffers (max 10), excludes filtered buffers
  :selection    - Region info if active (text, start-line, end-line)
  :cwd          - Working directory (uses CWD if provided, otherwise default-directory)

Buffers are filtered according to `code-agent-ide-context-exclude-buffer-names',
`code-agent-ide-context-exclude-modes', and `code-agent-ide-context-exclude-predicates'.

Note: current-file is determined by finding the most recently used file buffer
from `buffer-list', not necessarily `current-buffer'. This allows correct context
collection when called from special buffers like Claude Chat."
  (let ((current-buf (code-agent--get-current-file-buffer)))
    (list :current-file (when current-buf
                          (code-agent--format-buffer-info current-buf))
          :open-files (code-agent--get-file-buffers 10)
          :selection (code-agent--format-selection)
          :cwd (or cwd default-directory))))
#+END_SRC
