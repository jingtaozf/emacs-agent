* Claude Org cmux
:PROPERTIES:
:header-args:elisp: :tangle yes
:END:

cmux backend for claude-org.
Sends prompts to Claude Code running in cmux terminal via Unix socket API.
Responses come back via Workspace bridge hooks.
Part of claude-org; loaded after claude-org.org.

* Agent Profiles

Each supported code agent is described by a profile struct.  The profile
drives =build-launch-command=, ready/busy detection, cancel keys, and feature
guards (IDE integration, session resume).  Adding a new agent requires only
pushing a new profile onto =claude-org-cmux--agent-profiles= — no other elisp
changes are needed.

#+BEGIN_SRC elisp
(cl-defstruct claude-org-cmux--agent-profile
  "Describes how to launch and integrate with a code agent in cmux."
  (name nil)                    ; Symbol: 'claude or 'copilot
  (launch-fn nil)               ; (org-file session-id project-root) → shell-cmd string
  (ready-patterns nil)          ; List of regexp strings: TUI is ready for input
  (busy-patterns nil)           ; List of regexp strings: TUI is processing
  (cancel-keys nil)             ; List of cmux key names to send for cancel
  (supports-resume nil)         ; t if --resume SESSION_ID is supported
  (supports-ide nil)            ; t if IDE WebSocket integration available
  (supports-system-prompt nil)  ; t if system prompt flag injection is supported
  (vi-mode nil))                ; t if TUI uses vi-style INSERT mode (send "i" before text)

(defvar claude-org-cmux--agent-profiles nil
  "Registry of known agent profiles.
Populated below after the launch-fn helpers are defined.
To register a third-party agent, push a `claude-org-cmux--agent-profile'
struct onto this list from your own package or init file.")

(defun claude-org-cmux--get-agent-profile (&optional agent-type-override)
  "Return the agent profile to use, or nil to use legacy launch-command path.
Resolution order:
  1. AGENT_TYPE-OVERRIDE argument (for testing)
  2. AGENT_TYPE org property on the session heading
  3. `claude-org-cmux-agent-type' customization variable
  4. nil → use legacy `claude-org-cmux-launch-command' (Claude Code default)"
  (let* ((type-sym (or agent-type-override
                       (let ((prop (condition-case nil
                                       (claude-org-terminal--find-session-property "AGENT_TYPE")
                                     (error nil))))
                         (when (and prop (not (string-empty-p prop)))
                           (intern (downcase prop))))
                       claude-org-cmux-agent-type)))
    (when type-sym
      (cl-find type-sym claude-org-cmux--agent-profiles
               :key #'claude-org-cmux--agent-profile-name))))

(defun claude-org-cmux--resolve-color (color-str)
  "Resolve COLOR-STR to a hex color.
Accepts hex (#RRGGBB) or a named preset (green, blue, etc.).
Returns nil if COLOR-STR is nil or unrecognized."
  (when color-str
    (let ((lower (downcase (string-trim color-str))))
      (cond
       ((string-prefix-p "#" lower) color-str)
       (t (cdr (assoc lower claude-org-cmux--color-presets)))))))

(defun claude-org-cmux--resolve-icon (icon-str)
  "Resolve ICON-STR to an SF Symbol name.
Accepts an SF Symbol name directly (e.g. \"bolt.fill\") or a preset.
Returns nil if ICON-STR is nil."
  (when icon-str
    (let ((lower (downcase (string-trim icon-str))))
      (or (cdr (assoc lower claude-org-cmux--icon-presets))
          lower))))

(defun claude-org-cmux--apply-color (workspace-ref &optional icon)
  "Set sidebar status pill on WORKSPACE-REF using WORKSPACE_COLOR and WORKSPACE_ICON.
Label is derived from the org file name (e.g. \"claude-agent-dev\").
ICON is the default icon, overridden by WORKSPACE_ICON if set.
No-op if WORKSPACE_COLOR is not set."
  (let* ((color-raw (claude-org-terminal--find-session-property "WORKSPACE_COLOR"))
         (color (claude-org-cmux--resolve-color color-raw))
         (icon-raw (claude-org-terminal--find-session-property "WORKSPACE_ICON"))
         (resolved-icon (or (claude-org-cmux--resolve-icon icon-raw) icon "circle.fill"))
         (label (or (when buffer-file-name
                      (file-name-sans-extension
                       (file-name-nondirectory buffer-file-name)))
                    "project")))
    (when color
      ;; Status pill (works on all cmux versions)
      (condition-case nil
          (apply #'claude-org-cmux--call
                 "set-status" "project" label
                 "--icon" resolved-icon
                 "--color" color
                 (when workspace-ref
                   (list "--workspace" workspace-ref)))
        (error nil))
      ;; Workspace tab color via upstream workspace-action CLI
      (when workspace-ref
        (condition-case nil
            (claude-org-cmux--call
             "workspace-action" "set-color"
             "--color" (or color-raw color)
             "--workspace" workspace-ref)
          (error nil))))))

* CLI Gateway

#+BEGIN_SRC elisp
(defun claude-org-cmux--call (subcommand &rest args)
  "Call cmux CLI with SUBCOMMAND and ARGS. Return stdout string.
If ARGS contains :stdin KEY, that value is piped as stdin."
  (claude-agent-with-span "cmux-call"
      (list :span-kind "CLIENT" :subcommand subcommand) nil
  (let* ((stdin-text nil)
         (cli-args nil))
    ;; Separate :stdin from positional args
    (while args
      (if (eq (car args) :stdin)
          (progn
            (setq stdin-text (cadr args))
            (setq args (cddr args)))
        (push (car args) cli-args)
        (setq args (cdr args))))
    (setq cli-args (nreverse cli-args))
    (let* ((effective-socket
            (or claude-org-cmux-socket-path
                ;; Per-story override: CMUX_SOCKET_PATH org property
                (condition-case nil
                    (claude-org-terminal--find-session-property "CMUX_SOCKET_PATH")
                  (error nil))))
           (effective-cli
            (or (condition-case nil
                    (claude-org-terminal--find-session-property "CMUX_CLI_PATH")
                  (error nil))
                claude-org-cmux-cli-path))
           (socket-args (when effective-socket
                          (list "--socket" effective-socket)))
           (flat-args (append socket-args (cons subcommand (flatten-tree cli-args))))
           ;; When using a custom socket (dev app), clear production env vars
           ;; so the CLI doesn't resolve workspaces from the wrong instance
           (process-environment
            (if effective-socket
                (cl-remove-if
                 (lambda (e) (string-match-p "^CMUX_WORKSPACE_ID=" e))
                 process-environment)
              process-environment))
           (stderr-file (make-temp-file "cmux-stderr-"))
           (buf (generate-new-buffer " *cmux-cli*"))
           (stdin-file (when stdin-text
                         (let ((f (make-temp-file "cmux-stdin-")))
                           (with-temp-file f (insert stdin-text))
                           f)))
           (exit-code (apply #'call-process
                             effective-cli stdin-file
                             (list buf stderr-file) nil
                             flat-args)))
      (unwind-protect
          (with-current-buffer buf
            (if (= exit-code 0)
                (string-trim (buffer-string))
              (let ((stderr (string-trim
                             (with-temp-buffer
                               (insert-file-contents stderr-file)
                               (buffer-string)))))
                (error "cmux %s failed (exit %d): %s"
                       subcommand exit-code stderr))))
        (kill-buffer buf)
        (when stdin-file (delete-file stdin-file t))
        (delete-file stderr-file t))))))

* Wait for Ready

#+BEGIN_SRC elisp
(defcustom claude-org-cmux-ready-timeout 30
  "Max seconds to wait for Claude Code to become ready in cmux."
  :type 'integer
  :group 'claude-org)

(defun claude-org-cmux--wait-for-ready (surface-id)
  "Wait for Claude Code TUI to be ready in SURFACE-ID.
Uses cmux wait-for signal from SessionStart hook (50ms polling).
Falls back to capture-pane polling if signal times out."
  (claude-agent-with-span "cmux-wait-for-ready"
      (list :surface-id surface-id :timeout claude-org-cmux-ready-timeout) nil
  (let ((workspace-session-id
         (condition-case nil
             (claude-org-terminal--find-session-property "CLAUDE_SESSION_ID")
           (error nil)))
        (timeout (number-to-string claude-org-cmux-ready-timeout)))
    (if workspace-session-id
        ;; Event-driven: wait for signal from SessionStart hook
        (condition-case nil
            (progn
              (claude-org-cmux--call "wait-for"
                (format "claude-ready-%s" workspace-session-id)
                "--timeout" timeout)
              t)
          (error
           ;; Fallback: poll capture-pane with shorter timeout
           (claude-org-cmux--wait-for-ready-poll surface-id 10)))
      ;; No session ID — poll only
      (claude-org-cmux--wait-for-ready-poll
       surface-id claude-org-cmux-ready-timeout)))))

(defun claude-org-cmux--wait-for-ready-poll (surface-id timeout)
  "Fallback: poll capture-pane for agent ready state every 0.5s up to TIMEOUT seconds.
Uses ready-patterns from the active agent profile, falling back to Claude Code patterns."
  (let* ((profile (claude-org-cmux--get-agent-profile))
         (patterns (if profile
                       (claude-org-cmux--agent-profile-ready-patterns profile)
                     '("type a message" "-- INSERT --" "-- insert --")))
         (pattern-re (mapconcat #'regexp-quote patterns "\\|"))
         (deadline (+ (float-time) timeout))
         (ready nil))
    (while (and (not ready) (< (float-time) deadline))
      (condition-case nil
          (let ((screen (claude-org-cmux--call "capture-pane"
                          "--surface" surface-id)))
            (when (string-match-p pattern-re screen)
              (setq ready t)))
        (error nil))
      (unless ready (sleep-for 0.5)))
    (unless ready
      (error "Timeout: agent not ready in cmux after %ds" timeout))
    t))

* Send Text Helper

#+BEGIN_SRC elisp
(defun claude-org-cmux--ensure-insert-mode (surface-id)
  "Ensure Claude Code TUI is in INSERT mode in SURFACE-ID.
If not already in INSERT mode, sends `i` to enter it and waits briefly."
  (let ((screen (condition-case nil
                    (claude-org-cmux--call "capture-pane" "--surface" surface-id)
                  (error ""))))
    (unless (string-match-p "-- INSERT --\\|-- insert --" screen)
      ;; Send 'i' to enter INSERT mode (like vim)
      (claude-org-cmux--call "send" "--surface" surface-id "i")
      (sleep-for 0.3))))

(defun claude-org-cmux--send-text (surface-id text)
  "Send TEXT to cmux SURFACE-ID.
Strips Emacs text properties to prevent #(...) leaking into terminal.
For agents with vi-mode TUI (Claude Code), ensures INSERT mode first.
For non-vi-mode agents (OpenCode, Copilot), multi-line text is sent line
by line with shift-enter between lines, so newlines appear as literal
characters in the TUI input box rather than submitting each line separately."
  (claude-agent-with-span "cmux-send-text"
      (list :surface-id surface-id
            :text-length (length text)
            :input (truncate-string-to-width (or text "") 2000)) nil
  (let* ((clean-text (substring-no-properties text))
         (profile (claude-org-cmux--get-agent-profile))
         (vi-p (or (null profile)  ; legacy path = Claude Code
                   (claude-org-cmux--agent-profile-vi-mode profile)))
         ;; Non-vi-mode agents need shift-enter for multi-line text because
         ;; `cmux send` interprets \n as Enter, splitting the prompt.
         (use-paste-p (or (>= (length clean-text) 100000)
                          (and (not vi-p)
                               (string-match-p "\n" clean-text)))))
    ;; Only send "i" for vi-mode TUIs (Claude Code); others accept text directly
    (when vi-p
      (claude-org-cmux--ensure-insert-mode surface-id))
    (if use-paste-p
        ;; Multi-line path: send lines with shift-enter between them.
        ;; shift-enter inserts a literal newline in TUI input boxes.
        ;; Replaces broken paste-buffer approach (cmux "Surface is not a terminal").
        (let ((lines (split-string clean-text "\n")))
          (claude-org-cmux--call "send" "--surface" surface-id (car lines))
          (dolist (line (cdr lines))
            (claude-org-cmux--call "send-key" "--surface" surface-id "shift-enter")
            (when (> (length line) 0)
              (claude-org-cmux--call "send" "--surface" surface-id line))))
      ;; Direct send — safe for single-line text or vi-mode agents
      (claude-org-cmux--call "send" "--surface" surface-id clean-text))
    ;; Send Enter to submit the prompt
    (claude-org-cmux--call "send-key" "--surface" surface-id "enter")
    ;; Retry-on-failure: only for vi-mode TUIs where Enter can be lost
    ;; if Claude Code is still in INSERT mode after 3s
    (when vi-p
      (run-at-time 3 nil
        (lexical-let ((sid surface-id))
          (lambda ()
            (condition-case nil
                (let ((screen (claude-org-cmux--call "capture-pane"
                                "--surface" sid)))
                  (when (and (string-match-p "-- INSERT --" screen)
                             (not (string-match-p
                                   "Thinking\\|Tool:\\|Reading\\|Searching\\|Running\\|Generating"
                                   screen)))
                    (claude-org-cmux--call "send-key" "--surface" sid "enter")))
              (error nil)))))))))

* Launch Command

#+BEGIN_SRC elisp
(defun claude-org-cmux--build-launch-command (org-file workspace-session-id project-root)
  "Build the shell command to launch the configured code agent in cmux.
ORG-FILE is the org notebook path, WORKSPACE-SESSION-ID the session,
PROJECT-ROOT the working directory.

Dispatches to the active agent profile's launch-fn.  Falls back to the
legacy `claude-org-cmux-launch-command' path when no agent profile is set."
  (let ((profile (claude-org-cmux--get-agent-profile)))
    (if profile
        (funcall (claude-org-cmux--agent-profile-launch-fn profile)
                 org-file workspace-session-id project-root)
      ;; Legacy fallback: original Claude Code logic via launch-command cond
      (claude-org-cmux--build-claude-legacy-launch-command
       org-file workspace-session-id project-root))))

(defun claude-org-cmux--build-claude-legacy-launch-command
    (org-file workspace-session-id project-root)
  "Build Claude Code launch command using the legacy launch-command variable.
ORG-FILE is the org notebook path, WORKSPACE-SESSION-ID the session,
PROJECT-ROOT the working directory."
  (ignore project-root)
  (let* ((prop-args (claude-org-terminal--find-session-property "CLAUDE_EXTRA_ARGS"))
         (all-extra (append claude-org-cmux-extra-args
                            (when prop-args (split-string prop-args))))
         (extra (mapconcat #'shell-quote-argument all-extra " ")))
    (cond
     ;; claude-workspace: passes org-file and session-id
     ((eq claude-org-cmux-launch-command 'claude-workspace)
      ;; Emacs is the state owner for --resume (reads CLAUDE_CLI_SESSION).
      ;; Do NOT use find-session-property (inheritance=t) here — a stale
      ;; file-level #+PROPERTY: CLAUDE_CLI_SESSION contaminates all stories.
      ;; Read directly from the session heading (no inheritance).
      (let* ((workspace-script claude-org-cmux-workspace-script)
             (cli-session
              (ignore-errors
                (save-excursion
                  (claude-org-terminal--goto-session-heading)
                  (org-entry-get nil "CLAUDE_CLI_SESSION" nil))))
             (parts (list workspace-script
                          (shell-quote-argument (or org-file ""))
                          (shell-quote-argument (or workspace-session-id "")))))
        (when cli-session
          (setq parts (append parts
                              (list "--"
                                    "--resume"
                                    (shell-quote-argument cli-session)))))
        (when (and extra (not (string-empty-p extra)))
          (unless cli-session
            (setq parts (append parts (list "--"))))
          (setq parts (append parts (list extra))))
        (mapconcat #'identity parts " ")))
     ;; bare claude
     ((eq claude-org-cmux-launch-command 'claude)
      (let* ((system-prompt
              (condition-case nil
                  (claude-org-workspace-bridge-system-prompt org-file workspace-session-id)
                (error "")))
             (sp-file (make-temp-file "claude-sp-"))
             (cli-session
              (condition-case nil
                  (claude-org-workspace-bridge-get-cli-session org-file workspace-session-id)
                (error nil)))
             (parts (list "claude")))
        (with-temp-file sp-file (insert (or system-prompt "")))
        (push (format "--system-prompt \"$(cat %s)\"" (shell-quote-argument sp-file))
              parts)
        (when cli-session
          (push (format "--resume %s" (shell-quote-argument cli-session)) parts))
        (when (and extra (not (string-empty-p extra)))
          (push extra parts))
        (mapconcat #'identity (nreverse parts) " ")))
     ;; Custom string command
     ((stringp claude-org-cmux-launch-command)
      (if (and extra (not (string-empty-p extra)))
          (concat claude-org-cmux-launch-command " " extra)
        claude-org-cmux-launch-command))
     (t (error "Invalid claude-org-cmux-launch-command: %S"
               claude-org-cmux-launch-command)))))

(defun claude-org-cmux--build-claude-profile-launch-command
    (org-file workspace-session-id project-root)
  "Build Claude Code launch command via the agent profile path.
ORG-FILE is the org notebook path, WORKSPACE-SESSION-ID the session,
PROJECT-ROOT the working directory.
Used when `claude-org-cmux-agent-type' is `claude'."
  (ignore project-root)
  (let* ((prop-args (claude-org-terminal--find-session-property "CLAUDE_EXTRA_ARGS"))
         (all-extra (append claude-org-cmux-extra-args
                            (when prop-args (split-string prop-args))))
         (extra (mapconcat #'shell-quote-argument all-extra " "))
         ;; Emacs is the state owner for --resume (reads CLAUDE_CLI_SESSION).
         ;; Read directly from the session heading (no inheritance).
         (cli-session
          (ignore-errors
            (save-excursion
              (claude-org-terminal--goto-session-heading)
              (org-entry-get nil "CLAUDE_CLI_SESSION" nil))))
         (workspace-script claude-org-cmux-workspace-script)
         (parts (list workspace-script
                      (shell-quote-argument (or org-file ""))
                      (shell-quote-argument (or workspace-session-id "")))))
    (when cli-session
      (setq parts (append parts
                          (list "--"
                                "--resume"
                                (shell-quote-argument cli-session)))))
    (when (and extra (not (string-empty-p extra)))

    (mapconcat #'identity parts " ")))

  "CLI flags that are Claude Code-specific and must be filtered out for Copilot.
Note: --resume is NOT listed here because Copilot CLI supports it.")

(defvar claude-org-cmux--claude-args-with-value
  '("--permission-mode"
    "--plugin-dir"
    "--mcp-config"
    "--system-prompt"
    "--name")
  "Subset of `claude-org-cmux--claude-only-args' that consume the following argument as a value.")

(defun claude-org-cmux--filter-copilot-args (args)
  "Remove Claude Code-specific flags from ARGS list for Copilot compatibility.
Flags in `claude-org-cmux--claude-only-args' are removed.
Flags in `claude-org-cmux--claude-args-with-value' also consume their following value."
  (let ((result nil)
        (skip-next nil))
    (dolist (arg args)
      (cond
       (skip-next
        ;; This arg is the value of a removed flag — discard it
        (setq skip-next nil))
       ((member arg claude-org-cmux--claude-only-args)
        ;; Remove the flag; if it takes a value, also remove the next arg
        (when (member arg claude-org-cmux--claude-args-with-value)
          (setq skip-next t)))
       (t
        (push arg result))))
    (nreverse result)))

  "Build GitHub Copilot CLI launch command.
ORG-FILE is the org notebook path, WORKSPACE-SESSION-ID the session,
PROJECT-ROOT the working directory.
Reads COPILOT_CLI_SESSION for --resume support (separate from CLAUDE_CLI_SESSION

(defun claude-org-cmux--parse-surface-ref (output)
  "Extract a surface ref (e.g. \"surface:1\") from cmux OUTPUT.
Handles three formats:
  - JSON with .focused.surface_ref key (from identify)
  - \"OK workspace:N\" (from new-workspace)
  - Plain \"surface:N\" string
Returns the surface ref string, or OUTPUT unchanged as fallback."
  (cond
   ;; JSON: extract focused.surface_ref
   ((string-prefix-p "{" (string-trim output))
    (condition-case nil
        (let* ((json (json-read-from-string output))
               (focused (cdr (assq 'focused json)))
               (ref (cdr (assq 'surface_ref focused))))
          (or ref output))
      (error output)))
   ;; "OK workspace:N" → extract workspace ref
   ((string-match "OK \\(workspace:[^ \t\n]+\\)" output)
    (match-string 1 output))
   ;; "* surface:N ..." from list-pane-surfaces
   ((string-match "\\(surface:[^ \t\n]+\\)" output)
    (match-string 1 output))
   ;; Already a clean ref
   (t output)))

(defun claude-org-cmux--get-workspace-surface (workspace-ref)
  "Get the surface ref for WORKSPACE-REF via list-pane-surfaces.
Returns a clean surface ref like \"surface:1\"."
  (let ((output (claude-org-cmux--call
                 "list-pane-surfaces" "--workspace" workspace-ref)))
    (if (string-match "\\(surface:[^ \t\n]+\\)" output)
        (match-string 1 output)
      (error "No surface found in workspace %s" workspace-ref))))

(defun claude-org-cmux--find-workspace-by-name (name)
  "Return the current workspace ref (e.g. \"workspace:7\") for NAME.
Parses `cmux list-workspaces' output, which has the form:

  [* ]  workspace:N  <name>  [selected]

The heading title is the cmux/emacs protocol for workspace identity:
cmux persists workspaces by name across restarts while UUIDs and short
refs are regenerated.  Returns nil if no workspace matches NAME."
  (when (and name (not (string-empty-p name)))
    (condition-case nil
        (let ((lines (split-string
                      (claude-org-cmux--call "list-workspaces")
                      "\n" t)))
          (catch 'found
            (dolist (line lines)
              ;; Strip optional "* " selected marker and leading whitespace.
              (let* ((stripped (replace-regexp-in-string
                                "\\`[ \t]*\\*?[ \t]*" "" line))
                     ;; Drop trailing "  [selected]" marker if present.
                     (cleaned (replace-regexp-in-string
                               "[ \t]+\\[selected\\][ \t]*\\'" "" stripped)))
                (when (string-match "\\`\\(workspace:[^ \t]+\\)[ \t]+\\(.+?\\)[ \t]*\\'"
                                    cleaned)
                  (let ((ref (match-string 1 cleaned))
                        (ws-name (match-string 2 cleaned)))
                    (when (string= ws-name name)
                      (throw 'found ref))))))))
      (error nil))))

(defun claude-org-cmux--session-workspace-title ()
  "Return the stable cmux workspace name for the current session.
The workspace name is the session heading title, truncated to 50 chars
(matching `launch-workspace' when it calls `rename-workspace').  Point
must be somewhere inside the session subtree."
  (save-excursion
    (save-restriction
      (claude-org-terminal--goto-session-heading)
      (truncate-string-to-width
       (or (org-get-heading t t t t) "untitled") 50))))

(defun claude-org-cmux--session-workspace-ref ()
  "Return the current session's cmux workspace ref, resolved by heading name.
Returns a ref like \"workspace:7\" or nil.  This is the single source of
truth for \"which cmux workspace owns this session\" — the heading title
is stable across cmux restart while UUIDs/refs are regenerated.  Never
reads a CMUX_WORKSPACE_ID org property (deprecated)."
  (claude-org-cmux--find-workspace-by-name
   (claude-org-cmux--session-workspace-title)))

(defun claude-org-cmux--restore-workspace (saved-surface workspace-session-id
                                                          tab-title workspace-ref)
  "Restore in-memory state for a cmux workspace that already exists.
SAVED-SURFACE is the surface ref; WORKSPACE-SESSION-ID the org session
identifier; TAB-TITLE the active-story label; WORKSPACE-REF the cmux
workspace ref (resolved by name in `ensure-session').  Rebuilds hash
tables, re-applies color, renames the tab, and starts verbose streaming.
Returns SAVED-SURFACE."
  ;; Restore hash table mappings (lost on Emacs restart or session expiry)
  (when workspace-session-id
    (puthash workspace-session-id (claude-org--current-session-key)
             claude-org-terminal--workspace-to-session-key)
    (when saved-surface
      (puthash workspace-session-id saved-surface
               claude-org-cmux--workspace-to-surface))
    (when workspace-ref
      (let ((ws-uuid (claude-org-cmux--resolve-workspace-uuid workspace-ref)))
        (when ws-uuid
          (puthash workspace-session-id ws-uuid
                   claude-org-cmux--workspace-to-cmux-id)))))
  ;; Store workspace-session-id for verbose tick lookup
  (claude-org--session-put (claude-org--current-session-key)
                           :workspace-session-id workspace-session-id)
  (when workspace-ref
    (claude-org-cmux--apply-color workspace-ref)
    ;; Rename tab to match active story (Claude Code overrides terminal title)
    (condition-case nil
        (claude-org-cmux--call "rename-tab" "--workspace" workspace-ref tab-title)
      (error nil)))
  ;; Start verbose and ensure IDE server (IDE only for agents that support it)
  (claude-org-cmux--start-verbose saved-surface (claude-org--current-session-key))
  (when workspace-session-id
    (let ((profile (claude-org-cmux--get-agent-profile)))
      (when (or (null profile)
                (claude-org-cmux--agent-profile-supports-ide profile))
        (claude-org-cmux--ensure-ide-server
         (or (claude-org--get-project-root) default-directory) workspace-session-id))))
  saved-surface)

(defun claude-org-cmux--resolve-workspace-uuid (workspace-ref)
  "Resolve WORKSPACE-REF to a stable UUID via sidebar-state.
UUIDs are stable within a session and work in raw socket --tab= args,
while refs only work via CLI.  Returns the UUID string or WORKSPACE-REF
if resolution fails."
  (or (when (string-prefix-p "workspace:" workspace-ref)
        (condition-case nil
            (let ((state (claude-org-cmux--call
                          "sidebar-state"
                          "--workspace" workspace-ref)))
              (when (string-match "^tab=\\([0-9A-F-]+\\)" state)
                (match-string 1 state)))
          (error nil)))
      workspace-ref))

(defun claude-org-cmux--persist-workspace-state (workspace-session-id
                                                  workspace-ref surface-id)
  "Save workspace state to in-memory hash tables and session heading.
WORKSPACE-SESSION-ID is the org session identifier, WORKSPACE-REF the
cmux workspace ref (e.g. \"workspace:3\"), SURFACE-ID the terminal
surface ref.  Persists only CMUX_SURFACE_ID as an org property — the
workspace identity is the heading title (stable across cmux restart).
Populates session state and the three mapping hash tables."
  (let ((session-key (claude-org--current-session-key))
        (ws-uuid (claude-org-cmux--resolve-workspace-uuid workspace-ref)))
    (save-excursion
      (save-restriction
        (claude-org-terminal--goto-session-heading)
        (org-set-property "CMUX_SURFACE_ID" surface-id)))
    (claude-org--session-put session-key :cmux-surface surface-id)
    (claude-org--session-put session-key :workspace-session-id workspace-session-id)
    (puthash workspace-session-id surface-id
             claude-org-cmux--workspace-to-surface)
    (puthash workspace-session-id session-key
             claude-org-terminal--workspace-to-session-key)
    (when ws-uuid
      (puthash workspace-session-id ws-uuid
               claude-org-cmux--workspace-to-cmux-id))
    (save-buffer)))

(defun claude-org-cmux--setup-ide-after-ready (surface-id project-root
                                                workspace-session-id)
  "Start IDE server and send /ide command if the agent profile supports it.
SURFACE-ID is the terminal surface, PROJECT-ROOT the working directory,
WORKSPACE-SESSION-ID the org session identifier.  Skips IDE setup for
agent profiles that explicitly disable it (e.g. copilot)."
  (when workspace-session-id
    (let ((profile (claude-org-cmux--get-agent-profile)))
      (when (or (null profile)
                (claude-org-cmux--agent-profile-supports-ide profile))
        (claude-org-cmux--ensure-ide-server project-root workspace-session-id)
        (run-at-time 5 nil
          (lexical-let ((sid surface-id))
            (lambda ()
              (condition-case nil
                  (progn
                    (claude-org-cmux--call "send" "--surface" sid "/ide")
                    (claude-org-cmux--call "send-key" "--surface" sid "Return"))
                (error nil)))))))))

(defun claude-org-cmux--launch-workspace (workspace-session-id tab-title
                                          workspace-title project-root)
  "Launch a new cmux workspace running Claude Code.
WORKSPACE-SESSION-ID identifies the org session, TAB-TITLE and
WORKSPACE-TITLE name the tab and workspace, PROJECT-ROOT sets the
working directory.  Stores org properties, mappings, waits for the TUI
to become ready, and returns the surface ref."
  (unless workspace-session-id
    (user-error "No CLAUDE_SESSION_ID found. Create a workspace section first"))
  (let* ((org-file (buffer-file-name))
         (launch-cmd (claude-org-cmux--build-launch-command
                      org-file workspace-session-id project-root))
         ;; Create new workspace with launch command (returns "OK workspace:N").
         ;; If all tabs are closed, cmux reports "TabManager not available"
         ;; — create a new window first, then retry.
         (result (condition-case err
                     (claude-org-cmux--call
                      "new-workspace"
                      "--cwd" project-root
                      "--command" launch-cmd)
                   (error
                    (if (string-match-p "TabManager" (error-message-string err))
                        (progn
                          (claude-org-cmux--call "new-window")
                          (sleep-for 0.5)
                          (claude-org-cmux--call
                           "new-workspace"
                           "--cwd" project-root
                           "--command" launch-cmd))
                      (signal (car err) (cdr err))))))
         ;; Parse workspace ref, then get its surface ref
         (workspace-ref (claude-org-cmux--parse-surface-ref result))
         (surface-id (if (string-prefix-p "workspace:" workspace-ref)
                         (claude-org-cmux--get-workspace-surface workspace-ref)
                       workspace-ref)))
    ;; Rename workspace to heading name, tab to active story
    (when (string-prefix-p "workspace:" workspace-ref)
      (condition-case nil
          (progn
            (claude-org-cmux--call "rename-workspace"
              "--workspace" workspace-ref workspace-title)
            (claude-org-cmux--call "rename-tab"
              "--workspace" workspace-ref tab-title))
        (error nil)))
    ;; Persist workspace state (org properties + in-memory hash tables)
    (claude-org-cmux--persist-workspace-state
     workspace-session-id workspace-ref surface-id)
    ;; Apply color and start verbose BEFORE waiting for agent TUI.
    ;; These are pure cmux sidebar ops that don't need Claude Code running.
    ;; If we wait first, a slow or stuck agent launch means color/verbose
    ;; never get applied (wait-for-ready can time out).
    (when (string-prefix-p "workspace:" workspace-ref)
      (claude-org-cmux--apply-color workspace-ref))
    (claude-org-cmux--start-verbose surface-id (claude-org--current-session-key))
    (message "Agent starting in cmux (surface: %s) — waiting for TUI..." surface-id)
    ;; Wait for agent TUI to be ready before returning
    (claude-org-cmux--wait-for-ready surface-id)
    ;; IDE server and /ide command DO need the TUI ready.
    (claude-org-cmux--setup-ide-after-ready surface-id project-root
                                            workspace-session-id)
    (message "Agent ready in cmux (surface: %s)" surface-id)
    surface-id))

(defun claude-org-cmux--ensure-session ()
  "Ensure a cmux workspace/surface exists for the current workspace story.
Returns the cmux surface ref (e.g. \"surface:1\").
Launches if needed, reuses by saved ref.
Auto-starts the Emacs MCP server if not running."
  (claude-agent-with-span "cmux-ensure-session"
    (list :session-id (condition-case nil
                          (claude-org-terminal--find-session-property "CLAUDE_SESSION_ID")
                        (error nil))) nil
  ;; Best-effort MCP server auto-start
  (when (and (fboundp 'emacs-mcp-server-running-p)
             (not (emacs-mcp-server-running-p))
             (fboundp 'emacs-mcp-server-start))
    (condition-case err
        (progn
          (emacs-mcp-server-start)
          (run-at-time 1 nil
                       (lambda ()
                         (if (emacs-mcp-server-running-p)
                             (message "Started Emacs MCP server for workspace bridge")
                           (message "WARNING: MCP server didn't start")))))
      (error (message "WARNING: MCP server start failed: %s" err))))
  (let* ((saved-surface (claude-org-terminal--find-session-property "CMUX_SURFACE_ID"))
         (tab-title (claude-org-terminal--tab-title))
         (workspace-title (claude-org-cmux--session-workspace-title))
         (workspace-session-id
          (claude-org-terminal--find-session-property "CLAUDE_SESSION_ID"))
         (project-root (or (claude-org--get-project-root) default-directory))
         ;; Single source of truth: resolve workspace by heading name.
         ;; Names are stable across cmux restart; UUIDs and refs are not.
         (ws-ref (claude-org-cmux--find-workspace-by-name workspace-title)))
    (cond
     ;; Workspace exists in cmux (possibly with a fresh surface after a
     ;; cmux restart) — restore in-memory mappings and refresh the
     ;; surface property if the old one went stale.
     (ws-ref
      (let* ((fresh-surface (condition-case nil
                                (claude-org-cmux--get-workspace-surface ws-ref)
                              (error nil)))
             (surface-to-use (or fresh-surface saved-surface)))
        (when (and fresh-surface (not (equal fresh-surface saved-surface)))
          (save-excursion
            (save-restriction
              (claude-org-terminal--goto-session-heading)
              (org-set-property "CMUX_SURFACE_ID" fresh-surface)))
          (when (buffer-file-name) (save-buffer)))
        (claude-org-cmux--restore-workspace
         surface-to-use workspace-session-id tab-title ws-ref)))
     ;; No workspace by this name — launch a fresh one.
     (t
      (claude-org-cmux--launch-workspace
       workspace-session-id tab-title workspace-title project-root))))))

(defun claude-org-cmux--get-status (workspace-session-id surface-id)
  "Get code agent status for WORKSPACE-SESSION-ID and SURFACE-ID.
WORKSPACE-SESSION-ID keys the hook file; SURFACE-ID keys the cmux API.
Uses ready/busy patterns from the active agent profile."
  (let ((hook-status (claude-org-terminal--read-hook-status workspace-session-id)))
    (if (member hook-status '("ready" "busy"))
        hook-status
      ;; Fallback: read screen content from cmux using profile patterns
      (condition-case nil
          (let* ((profile (claude-org-cmux--get-agent-profile))
                 (ready-re (if profile
                               (mapconcat #'regexp-quote
                                          (claude-org-cmux--agent-profile-ready-patterns profile)
                                          "\\|")
                             "-- INSERT --\\|type a message"))
                 (busy-re (if profile
                              (mapconcat #'regexp-quote
                                         (claude-org-cmux--agent-profile-busy-patterns profile)
                                         "\\|")
                            "Tool:\\|Thinking"))
                 (screen (claude-org-cmux--call "capture-pane"
                           "--surface" surface-id)))
            (cond
             ((string-match-p ready-re screen) "ready")
             ((string-match-p busy-re screen) "busy")
             ((string-empty-p screen) "dead")
             (t "unknown")))
        (error "dead")))))

(defun claude-org-cmux--query-completed (workspace-session-id)
  "Mark the cmux query for WORKSPACE-SESSION-ID as completed.
Unregisters from `claude-agent--active-queries'.
When loop state is active, schedules the next iteration.
Wrapped in condition-case: called from Python via MCP evalElisp,
so an unhandled error would leave the session permanently :busy t."
  (condition-case err
      (let* ((session-key (gethash workspace-session-id claude-org-terminal--workspace-to-session-key))
             (trace-ctx (when session-key
                          (claude-org--session-get session-key :trace-ctx))))
        (claude-agent-with-span "cmux-query-completed"
            (list :session-id workspace-session-id) trace-ctx
        (let ((req-id (claude-org-terminal--read-request-id workspace-session-id)))
          (when req-id
            (let ((state (claude-agent--get-active-query req-id)))
              (when state
                (setf (claude-agent--process-state-closed state) t)
                (claude-agent--unregister-query req-id)))))
        ;; Check loop continuation
        (let ((surface-id (gethash workspace-session-id claude-org-cmux--workspace-to-surface)))
          ;; Workspace mappings persist across queries (workspace outlives stories)
          ;; No pipe-pane streaming to stop (removed — marker was never wired up,
          ;; resulting in every-2s sync `cmux pipe-pane` calls with discarded output).
          ;; No color to re-apply — WORKSPACE_COLOR is static per workspace and
          ;; was already applied at session-creation time.
          ;; Verbose output (separate subsystem) keeps running — user may type
          ;; more queries directly in cmux without going through execute-ai-block.
          (when session-key
            (claude-org--session-put session-key :busy nil)
            (let ((loop-current (or (claude-org--session-get session-key :loop-current) 1))
                  (loop-max (or (claude-org--session-get session-key :loop-max) 1)))
              (if (< loop-current loop-max)
                  ;; More iterations — schedule next
                  (let ((interval (or (claude-org--session-get session-key :loop-interval) 0))
                        (prompt (claude-org--session-get session-key :original-prompt)))
                    (claude-org--session-put session-key :loop-current (1+ loop-current))
                    (if (> interval 0)
                        (run-at-time interval nil
                                     #'claude-org-cmux--loop-send
                                     workspace-session-id surface-id prompt session-key)
                      (claude-org-cmux--loop-send workspace-session-id surface-id prompt session-key)))
                ;; Loop complete — cleanup
                (claude-org--session-put session-key :loop-max nil)
                (claude-org--session-put session-key :loop-current nil)))))))
    (error
     ;; Ensure busy state is cleared even on error
     (let ((session-key (gethash workspace-session-id claude-org-terminal--workspace-to-session-key)))
       (when session-key
         (claude-org--session-put session-key :busy nil)))
     (message "claude-org-cmux: error in query-completed: %S" err))))

(defun claude-org-cmux--loop-send (workspace-session-id surface-id prompt session-key)
  "Send next loop iteration PROMPT to cmux.
WORKSPACE-SESSION-ID and SURFACE-ID identify the session."
  (claude-agent-with-span "cmux-loop-send"
      (list :session-id workspace-session-id) nil

            (org-back-to-heading t)

  (claude-org-cmux--send-text surface-id prompt)
  (let ((current (claude-org--session-get session-key :loop-current))
        (max (claude-org--session-get session-key :loop-max)))
    (message "Sent loop iteration %d/%d to cmux (%d chars)"
             current max (length prompt)))))

* Execution

#+BEGIN_SRC elisp
(defun claude-org-cmux--execute-ai-block ()
  "Execute current AI block via cmux backend.
Ensures session exists, checks readiness, sends prompt.
Registers in `claude-agent--active-queries' for mode-line."
  (let* ((element (org-element-at-point))
         (prompt (when (and (eq (org-element-type element) 'src-block)
                            (string= (org-element-property :language element) "ai"))
                   (substring-no-properties
                    (string-trim (org-element-property :value element))))))
    (unless prompt
      (user-error "Point is not inside a #+begin_src ai block"))
    (let* ((surface-id (claude-org-cmux--ensure-session))
           (workspace-session-id
            (claude-org-terminal--find-session-property "CLAUDE_SESSION_ID"))
           (status (claude-org-cmux--get-status workspace-session-id surface-id)))
      (when (string= status "busy")
        (user-error "Claude Code busy (status: %s). Cancel first?" status))
      ;; Detect story and update ACTIVE_STORY on workspace heading
      (let ((story (claude-org--find-story-heading)))
        (when story
          (claude-org--set-active-story (car story))))
      ;; Store ROOT trace context BEFORE entering child span,
      ;; so async callbacks (cmux-query-completed) join at the root level
      (let ((session-key (claude-org--current-session-key)))
        (when session-key
          (claude-org--session-put session-key :trace-ctx (claude-agent-trace-context))))
      (claude-agent-with-span "cmux-execute"
          (list :session-id workspace-session-id
                :input prompt
                :surface-id surface-id) nil
      ;; Store workspace→session-key mapping
      (when workspace-session-id
        (puthash workspace-session-id (claude-org--current-session-key)
                 claude-org-terminal--workspace-to-session-key))
      ;; Mark backend so generic claude-org-cancel can dispatch to cmux
      (claude-org--session-put (claude-org--current-session-key) :backend "cmux")
      ;; Save org buffer + marker for loop-send
      (claude-org--session-put (claude-org--current-session-key)
                               :loop-org-buffer (current-buffer))
      (claude-org--session-put (claude-org--current-session-key)
                               :loop-block-marker (copy-marker (point)))
      ;; Register in active-queries BEFORE writing from-emacs flag
      ;; (avoids TOCTOU race: hook could fire before request-id is written)
      (save-excursion
        (org-back-to-heading t)
        (when workspace-session-id
          (claude-org-terminal--register-query workspace-session-id surface-id)))
      ;; Write from-emacs flag BEFORE sending
      (when workspace-session-id
        (claude-org-terminal--write-from-emacs-flag workspace-session-id))
      ;; Send prompt via cmux
      (claude-org-cmux--send-text surface-id prompt)
      ;; Start verbose output mirror (async capture-pane polling into its own buffer)
      (claude-org-cmux--start-verbose surface-id (claude-org--current-session-key))
      (message "Sent to cmux (%d chars)" (length prompt))))))
#+END_SRC

* Response Delivery

Response text lands in the org buffer via the =Stop= hook →
=workspace_bridge.handle_response= → =claude-org-workspace-bridge-insert-response=.
No periodic polling from Emacs is needed.

(Historical note: there used to be a =claude-org-cmux--start-streaming= timer
that fired =cmux pipe-pane= every 2 s to mirror terminal output into the org
buffer.  Its insertion marker was never initialised, so every tick did a
200–1500 ms synchronous subprocess call and silently discarded the result —
a persistent Emacs hang for the full duration of every Claude query.  The
subsystem was removed in favour of the =Stop=-hook path, which delivers the
full structured response exactly once.  Live terminal echo is still available
via the async =Verbose Output= mirror below.)

Stream cmux terminal output into a read-only Emacs buffer accessible via
=claude-org-menu → v → Verbose output=. Uses async =capture-pane= polling
with diff-based deduplication.

(defvar-local claude-org-cmux--verbose-prev-text ""
  "Previous capture-pane output for diff-based deduplication.")

(defun claude-org-cmux--create-verbose-buffer (session-key surface-id)
  "Create a verbose output buffer for cmux terminal SESSION-KEY.
Registers it so `claude-org-show-verbose' can find it.  Derives the
workspace identity from SESSION-KEY itself (format `<file>::<session-id>')
so the buffer name stays stable even if `:workspace-session-id' session
state has been wiped by a module reload — never falls back to the ambient
current-buffer's `CLAUDE_SESSION_ID', which would collide with unrelated
buffers."
  (let* ((workspace-id (or (claude-org--session-get session-key :workspace-session-id)
                           (when (and session-key
                                      (string-match "::\\([^:]+\\)\\'" session-key))
                             (match-string 1 session-key))
                           "cmux"))
         (buf-name (format "*cmux: %s*" workspace-id))
         (buf (get-buffer-create buf-name)))
    (with-current-buffer buf
      (unless (eq major-mode 'claude-org-cmux-verbose-mode)
        (claude-org-cmux-verbose-mode)
        (setq-local claude-org-cmux--verbose-prev-text "")
        (setq-local claude-org-cmux--verbose-surface-id surface-id)
        ;; Sticky header with story name and keybindings
        (setq-local header-line-format
          (list "  "
                (propertize workspace-id 'face 'bold)
                "  "
                (propertize "[,]menu [G]follow [q]quit" 'face 'shadow)))))
    ;; Register so claude-org-show-verbose finds it
    (puthash session-key buf claude-agent--session-verbose-buffers)
    buf))

(defun claude-org-cmux--verbose-workspace-uuid (session-key)
  "Return the cmux workspace UUID for SESSION-KEY from the in-memory cache.
Checks `:workspace-session-id' session slot first, then parses the
session-id out of SESSION-KEY (format `<file>::<session-id>'); both feed
into `workspace-to-cmux-id', which is populated by `restore-workspace'
/ `launch-workspace' / `recover-session' when the workspace is resolved
by heading name.  Returns nil on cache miss — callers must skip the
tick; a subsequent `ensure-session' will repopulate the cache."
  (let ((ws-sid (or (claude-org--session-get session-key :workspace-session-id)
                    (when (and session-key
                               (string-match "::\\([^:]+\\)\\'" session-key))
                      (match-string 1 session-key)))))
    (when ws-sid
      (gethash ws-sid claude-org-cmux--workspace-to-cmux-id))))

(defun claude-org-cmux--verbose-strip-ansi (chunk)
  "Best-effort strip of ANSI escapes + control chars from CHUNK.

Stripping is deliberately lossy: we discard colour, cursor
positioning, and clear-screen sequences.  Preserving them would
require a tty emulator inside the verbose buffer — disproportionate
for a read-only debug view.  A handful of safe whitespace chars
(newline, tab, carriage return) survive so the stream remains
line-structured."
  (replace-regexp-in-string
   "\x1b\\[[0-9;?]*[a-zA-Z]\\|\x1b\\][^\a\x1b]*\\(?:\a\\|\x1b\\\\\\)\\|[\x00-\x08\x0b\x0c\x0e-\x1f]"
   "" chunk))

(defun claude-org-cmux--verbose-follow-filter (proc chunk)
  "Append CHUNK (scrubbed) from `cmux capture-pane --follow' to the verbose buffer.

Auto-scrolls only windows that were already at `point-max' — a user
reading an earlier page keeps their scroll position as new content
lands after the visible window."
  (let ((vbuf (process-get proc 'verbose-buffer)))
    (when (buffer-live-p vbuf)
      (let ((scrubbed (claude-org-cmux--verbose-strip-ansi chunk)))
        (when (> (length scrubbed) 0)
          (with-current-buffer vbuf
            (let ((win-states nil))
              (dolist (win (get-buffer-window-list vbuf nil t))
                (push (cons win (= (window-point win) (point-max))) win-states))
              (let ((inhibit-read-only t))
                (save-excursion
                  (goto-char (point-max))
                  (insert scrubbed)))
              (dolist (state win-states)
                (let ((win (car state)) (was-at-end (cdr state)))
                  (when (and (window-live-p win) was-at-end)
                    (set-window-point win (point-max))))))))))))

(defun claude-org-cmux--verbose-follow-sentinel (proc event)
  "Handle `cmux capture-pane --follow' lifecycle events.

On exit we drop the process reference from the session so a later
`--start-verbose' call can restart it (either manually, or when a
workspace is relaunched).  We deliberately DO NOT auto-restart
here — if cmux is genuinely gone there's no workspace to follow,
and silent reconnect loops would mask real failures."
  (let ((sk (process-get proc 'session-key))
        (vbuf (process-get proc 'verbose-buffer)))
    (when (string-match-p "\\(finished\\|exited\\|killed\\|broken\\)" event)
      (when (and sk (eq proc (claude-org--session-get sk :verbose-follow-process)))
        (claude-org--session-put sk :verbose-follow-process nil))
      (when (buffer-live-p vbuf)
        (with-current-buffer vbuf
          (let ((inhibit-read-only t))
            (save-excursion
              (goto-char (point-max))
              (insert (format "\n;; cmux stream ended: %s\n"
                              (string-trim event))))))))))

(defun claude-org-cmux--start-verbose (surface-id session-key)
  "Start the long-running `cmux capture-pane --follow' stream for SESSION-KEY.

Replaces the earlier polling timer (one `capture-pane' fork per
second per session).  A single subprocess per session reads the
terminal as it changes and its filter appends into
`*cmux: <session>*'.  The subprocess is idle-cheap — it blocks on
read between terminal writes, so zero CPU when nothing is
happening.  Restarts automatically when the surface id changes
(workspace relaunched with a new terminal pane)."
  (let ((existing (claude-org--session-get session-key :verbose-follow-process))
        (old-surface (claude-org--session-get session-key :verbose-surface-id)))
    ;; Restart if the surface id moved (new workspace pane replaces old).
    (when (and existing (or (not old-surface)
                            (not (equal old-surface surface-id))))
      (claude-org-cmux--stop-verbose session-key)
      (setq existing nil))
    (when (and existing (not (process-live-p existing)))
      (claude-org-cmux--stop-verbose session-key)
      (setq existing nil))
    (unless existing
      (claude-org--session-put session-key :verbose-surface-id surface-id)
      (let* ((ws-uuid (claude-org-cmux--verbose-workspace-uuid session-key))
             (vbuf (claude-org-cmux--create-verbose-buffer session-key surface-id))
             (sock (or claude-org-cmux-socket-path
                       (condition-case nil
                           (claude-org-terminal--find-session-property "CMUX_SOCKET_PATH")
                         (error nil)))))
        ;; No workspace resolution yet → nothing to follow.  Leave the
        ;; buffer in place; a later call (after `ensure-session'
        ;; populates the workspace map) will spawn the stream.
        (when (and ws-uuid surface-id (buffer-live-p vbuf))
          (let* ((cli (or claude-org-cmux-cli-path "cmux"))
                 (args (append (when sock (list "--socket" sock))
                               (list "capture-pane"
                                     "--workspace" ws-uuid
                                     "--surface" surface-id
                                     "--follow"
                                     "--interval" "0.5")))
                 (proc (apply #'start-process
                              (format "cmux-follow-%s" session-key)
                              nil cli args)))
            (process-put proc 'verbose-buffer vbuf)
            (process-put proc 'session-key session-key)
            (set-process-filter proc #'claude-org-cmux--verbose-follow-filter)
            (set-process-sentinel proc #'claude-org-cmux--verbose-follow-sentinel)
            (claude-org--session-put session-key :verbose-follow-process proc)))))))

(defun claude-org-cmux--stop-verbose (session-key)
  "Kill the verbose stream subprocess for SESSION-KEY, if any.

Also clears the legacy `:verbose-timer' slot — older sessions still
carrying a run-at-time timer (before the follow-stream refactor)
get cleaned up the first time this is called."
  (let ((proc (claude-org--session-get session-key :verbose-follow-process))
        (timer (claude-org--session-get session-key :verbose-timer)))
    (when (and proc (process-live-p proc))
      (delete-process proc))
    (claude-org--session-put session-key :verbose-follow-process nil)
    (when timer
      (cancel-timer timer)
      (claude-org--session-put session-key :verbose-timer nil))))

(defun claude-org-cmux-open-tab ()
  "Open (or focus) the cmux workspace for this workspace session.
Resolves the workspace by heading name.  Launches Claude Code if no
matching workspace exists.  Always re-applies WORKSPACE_COLOR pill
\(status pills are transient in cmux)."
  (interactive)
  (claude-org-cmux--ensure-session)
  ;; Re-resolve by name AFTER ensure-session — it may have launched a
  ;; fresh workspace.  The heading title is the stable identity.
  (let ((ws-ref (claude-org-cmux--session-workspace-ref))
        (surface-id (claude-org-terminal--find-session-property "CMUX_SURFACE_ID")))
    (cond
     (ws-ref
      (condition-case nil
          (progn
            (claude-org-cmux--call "select-workspace" "--workspace" ws-ref)
            ;; Re-apply color (pills are lost on cmux restart)
            (claude-org-cmux--apply-color ws-ref)
            (claude-org-cmux--activate-app)
            (message "cmux workspace focused (%s)" ws-ref))
        (error
         ;; Workspace ref became stale between resolve and call (rare).
         ;; Clear the surface marker and retry via ensure-session.
         (save-excursion
           (save-restriction
             (claude-org-terminal--goto-session-heading)
             (org-delete-property "CMUX_SURFACE_ID")
             (save-buffer)))
         (claude-org-cmux--ensure-session)
         (let ((fresh-ref (claude-org-cmux--session-workspace-ref)))
           (when fresh-ref
             (claude-org-cmux--call "select-workspace" "--workspace" fresh-ref)
             (claude-org-cmux--apply-color fresh-ref))
           (claude-org-cmux--activate-app)
           (message "cmux workspace recreated and focused")))))
     (surface-id
      (claude-org-cmux--activate-app)
      (message "cmux surface: %s (focused app)" surface-id))
     (t (user-error "No cmux session for this story")))))

(defun claude-org-cmux-restart ()
  "Restart Claude Code in the current cmux tab.
Sends /exit to quit the running Claude Code, waits for shell prompt,
then launches a fresh Claude Code in the same tab. Does not destroy
the workspace or affect other tabs.

Calls `claude-org-cmux--ensure-session' first so stale surface/workspace
IDs are refreshed after a cmux restart: the saved UUID may be gone but
cmux restores the workspace by name, and restart must drive the fresh
surface rather than the stale one."
  (interactive)
  (let* ((surface-id (claude-org-cmux--ensure-session))
         (workspace-session-id (claude-org-terminal--find-session-property "CLAUDE_SESSION_ID")))
    (unless surface-id
      (user-error "No cmux session for this story"))
    (let ((launch-cmd (claude-org-cmux--build-launch-command
                       (buffer-file-name)
                       workspace-session-id
                       (or (claude-org--get-project-root) default-directory)))
          (screen (condition-case nil
                      (claude-org-cmux--call "capture-pane" "--surface" surface-id "--lines" "5")
                    (error "")))
          (at-shell nil))
      ;; 1. Check if Claude Code is running or already at shell prompt
      (setq at-shell (or (string-match-p "\\$\\s*$" screen)
                         (string-match-p "❯\\s*$" screen)
                         (string-match-p "%\\s*$" screen)))
      (unless at-shell
        ;; Claude Code is running — send /exit and wait
        (message "Sending /exit to Claude Code...")
        (condition-case nil
            (progn
              (claude-org-cmux--call "send" "--surface" surface-id "/exit")
              (claude-org-cmux--call "send-key" "--surface" surface-id "Return"))
          (error nil))
        (let ((attempts 0))
          (while (< attempts 15)
            (sleep-for 1)
            (let ((s (condition-case nil
                         (claude-org-cmux--call "capture-pane" "--surface" surface-id "--lines" "3")
                       (error ""))))
              (when (or (string-match-p "\\$\\s*$" s)
                        (string-match-p "❯\\s*$" s)
                        (string-match-p "%\\s*$" s))
                (setq attempts 100)))
            (cl-incf attempts))))
      ;; 2. Launch new Claude Code in the same tab
      (message "Launching fresh Claude Code...")
      (claude-org-cmux--call "send" "--surface" surface-id
                             (concat launch-cmd "\n"))
      ;; 3. Rename tab to active story (cmux may reset title on relaunch)
      (let ((ws-ref (claude-org-cmux--session-workspace-ref))
            (tab-title (claude-org-terminal--tab-title)))
        (when ws-ref
          (condition-case nil
              (claude-org-cmux--call "rename-tab" "--workspace" ws-ref tab-title)
            (error nil))
          ;; 4. Re-apply color
          (claude-org-cmux--apply-color ws-ref)))
      (let ((sk (claude-org--current-session-key)))
        (claude-org-cmux--stop-verbose sk)
        (claude-org--session-put sk :verbose-surface-id nil)
        (claude-org-cmux--start-verbose surface-id sk))
      ;; 5. Ensure IDE server and send /ide after ready
      (when workspace-session-id
        (claude-org-cmux--ensure-ide-server
         (or (claude-org--get-project-root) default-directory) workspace-session-id)
        (run-at-time 10 nil
          (lexical-let ((sid surface-id))
            (lambda ()
              (condition-case nil
                  (progn
                    (claude-org-cmux--call "send" "--surface" sid "/ide")
                    (claude-org-cmux--call "send-key" "--surface" sid "Return"))
                (error nil))))))
      (message "Claude Code restarting in surface %s" surface-id))))

(defun claude-org-cmux-cancel ()
  "Cancel the current code agent operation in cmux.
Uses cancel keys from the active agent profile (Escape for Claude Code,
C-c for GitHub Copilot CLI)."
  (interactive)
  (claude-agent-with-span "cmux-cancel" nil nil
  (let* ((surface-id (claude-org-terminal--find-session-property "CMUX_SURFACE_ID"))
         (profile (claude-org-cmux--get-agent-profile))
         (cancel-keys (if profile
                          (claude-org-cmux--agent-profile-cancel-keys profile)
                        '("escape"))))
    (unless surface-id
      (user-error "No cmux session for this story"))
    (dolist (key cancel-keys)
      (claude-org-cmux--call "send-key" "--surface" surface-id key))
    (message "Cancel sent to agent (waiting for hook...)"))))

#+BEGIN_SRC elisp
(defun claude-org-cmux--permission-needed (workspace-session-id tool-name)
  "Handle permission notification for a cmux session.
Registers a mode-line alert, focuses the cmux workspace, and notifies user.
Also ensures the session is marked busy (self-healing: if the Prompt hook
was skipped, the permission event proves a query is active)."
  (claude-agent-with-span "cmux-permission-needed"
      (list :session-id workspace-session-id :tool-name tool-name) nil
  ;; Self-healing: ensure session is marked busy if not already.
  ;; The Prompt hook may have been skipped (e.g., retry/continue in Claude Code).
  (let ((session-key (gethash workspace-session-id claude-org-terminal--workspace-to-session-key)))
    (when (and session-key (not (claude-org--session-get session-key :busy)))
      (claude-org--session-put session-key :busy t)
      ;; Best-effort: register a query so response hook has a request-id.
      ;; Point in source-buffer may not be at the AI block heading, so
      ;; CUSTOM_ID/instruction-num in the query context may be wrong.
      ;; That's acceptable — the alternative is no query registration at all.
      (unless (claude-org-terminal--read-request-id workspace-session-id)
        (condition-case nil
            (let ((buf (claude-org--session-get session-key :source-buffer)))
              (when (and buf (buffer-live-p buf))
                (with-current-buffer buf
                  (claude-org-terminal--register-query workspace-session-id nil))))
          (error nil)))))
  ;; Register mode-line alert
  (claude-agent-add-alert
   (intern workspace-session-id)
   :label (claude-agent--permission-label workspace-session-id)
   :tooltip (format "Permission needed: %s\nClick to focus cmux workspace" tool-name)
   :face '(:foreground "orange" :weight bold)
   :action (lexical-let ((sid workspace-session-id))
             (lambda ()
               (claude-agent-remove-alert (intern sid))
               (claude-org-cmux--focus-workspace-by-session sid))))
  ;; Focus cmux workspace (best-effort)
  (claude-org-cmux--focus-workspace-by-session workspace-session-id)
  ;; Activate cmux app + Emacs (best-effort, macOS only)
  (claude-org-cmux--activate-app)
  ;; macOS notification
  (claude-org--notify "Claude: Permission Needed"
                      (format "%s needs approval for %s"
                              (claude-agent--permission-label workspace-session-id)
                              tool-name))
  "ok"))

* Session Recovery

#+BEGIN_SRC elisp
(defun claude-org-cmux--recover-session (workspace-session-id)
  "Try to recover session mapping by scanning org buffers.
Called when WORKSPACE-SESSION-ID is not found in the hash tables after
Emacs restart.  Finds the heading via `CLAUDE_SESSION_ID' property,
resolves the current cmux workspace ref by heading name (the stable
cmux/emacs identity), and repopulates hash tables.  Returns the
session-key if found, nil otherwise."
  (claude-agent-with-span "cmux-recover-session"
      (list :session-id workspace-session-id) nil
  (cl-block recover
    (dolist (buf (buffer-list))
      (when (and (buffer-file-name buf)
                 (with-current-buffer buf (bound-and-true-p claude-org-mode)))
        (with-current-buffer buf
          (save-excursion
            (save-restriction
              (widen)
              (goto-char (point-min))
              (when (re-search-forward
                     (format ":CLAUDE_SESSION_ID: %s"
                             (regexp-quote workspace-session-id))
                     nil t)
                (org-back-to-heading t)
                (let* ((surface (org-entry-get nil "CMUX_SURFACE_ID" nil))
                       (title (truncate-string-to-width
                               (or (org-get-heading t t t t) "") 50))
                       (ws-ref (and (not (string-empty-p title))
                                    (claude-org-cmux--find-workspace-by-name title)))
                       (ws-uuid (and ws-ref
                                     (claude-org-cmux--resolve-workspace-uuid ws-ref)))
                       (session-key (claude-org--current-session-key)))
                  (when (and surface session-key)
                    (puthash workspace-session-id session-key
                             claude-org-terminal--workspace-to-session-key)
                    (puthash workspace-session-id surface
                             claude-org-cmux--workspace-to-surface)
                    (when ws-uuid
                      (puthash workspace-session-id ws-uuid
                               claude-org-cmux--workspace-to-cmux-id))
                    (cl-return-from recover session-key))))))))))))
