* Claude Org Workspace Bridge
:PROPERTIES:
:header-args:elisp: :tangle yes
:END:

Functions for bridging between Claude Code CLI (running in a terminal) and
org-mode workspace sections. Called non-interactively via MCP ~evalElisp~ from
hook scripts.

** Buffer and Navigation

#+BEGIN_SRC elisp
(defun claude-org-workspace-bridge--ensure-buffer (org-file)
  "Ensure ORG-FILE is loaded in Emacs and return its buffer.
Opens the file via `find-file-noselect' if not already visiting."
  (or (find-buffer-visiting org-file)
      (find-file-noselect org-file)))

(defun claude-org-workspace-bridge-list-sessions (org-file)
  "List all workspace sessions in ORG-FILE.
Returns a newline-separated string of \"SESSION_ID\\tHEADING\\tPARENT\" lines,
one per workspace section found. PARENT is the outline path above the workspace heading."
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (goto-char (point-min))
        (let (results)
          (while (re-search-forward
                  "^[ \t]*:CLAUDE_SESSION_ID:[ \t]+\\(.+?\\)[ \t]*$" nil t)
            (let ((sid (match-string-no-properties 1)))
              (save-excursion
                (org-back-to-heading t)
                (let* ((heading (substring-no-properties (org-get-heading t t t t)))
                       (parent (mapconcat
                                (lambda (s) (substring-no-properties s))
                                (org-get-outline-path)
                                " / ")))
                  (push (format "%s\t%s\t%s" sid heading parent) results)))))
          (mapconcat #'identity (nreverse results) "\n"))))))

(defun claude-org-workspace-bridge--goto-session (session-id)
  "Move point to the workspace section matching SESSION-ID in current buffer.
Searches for :CLAUDE_SESSION_ID: property matching SESSION-ID.
Signals error if not found."
  (goto-char (point-min))
  (let ((found nil))
    (while (and (not found)
                (re-search-forward
                 (format "^[ \t]*:CLAUDE_SESSION_ID:[ \t]+%s[ \t]*$"
                         (regexp-quote session-id))
                 nil t))
      (setq found (point)))
    (unless found
      (error "Workspace session %s not found in buffer" session-id))
    ;; Move to the heading that owns this property
    (org-back-to-heading t)))

(defun claude-org-workspace-bridge--find-workflow (session-id)
  "Find the Workflow subsection under the active story for SESSION-ID.
Uses ACTIVE_STORY to narrow to the correct story, falls back to any Workflow.
Returns the position of the Workflow heading, or signals error."
  (claude-org-workspace-bridge--goto-session session-id)
  (let* ((workspace-end (save-excursion (org-end-of-subtree t t) (point)))
         (active-story (org-entry-get nil "ACTIVE_STORY"))
         (search-start (point))
         (search-end workspace-end)
         (found nil))
    ;; If ACTIVE_STORY is set, narrow search to that story's subtree
    (when active-story
      (save-excursion
        (let ((ws-level (org-current-level)))
          (when (re-search-forward
                 (format "^\\*\\{%d\\} +%s" (1+ ws-level) (regexp-quote active-story))
                 workspace-end t)
            (goto-char (line-beginning-position))
            (setq search-start (point))
            (setq search-end (save-excursion (org-end-of-subtree t t) (point)))))))
    ;; Find Workflow heading (with or without :sdd: tag)
    (save-excursion
      (goto-char search-start)
      (while (and (not found)
                  (re-search-forward "^\\*+ +Workflow\\b" search-end t))
        (setq found (line-beginning-position))))
    (unless found
      (error "Workflow section not found under session %s" session-id))
    (goto-char found)))

(defun claude-org-workspace-bridge-system-prompt (org-file session-id)
  "Build and return the workspace system prompt for SESSION-ID in ORG-FILE.
Called by the launcher script to get the system prompt text.
Navigates to the workspace section and reuses existing prompt-building logic."
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-workspace-bridge--goto-session session-id)
        ;; Narrow to the workspace subtree so collect-system-prompts finds
        ;; only this story's system prompt section
        (org-narrow-to-subtree)
        (claude-org--build-system-prompt)))))

** Prompt Insertion

#+BEGIN_SRC elisp
(defun claude-org-workspace-bridge-insert-prompt (org-file session-id prompt)
  "Insert PROMPT as a new AI block in the workspace Workflow for SESSION-ID in ORG-FILE.
Creates a new Instruction N subsection under the Workflow heading.
Returns the CUSTOM_ID string for correlating with the Stop hook."
  (claude-agent-with-span "workspace-insert-prompt"
      (list :session-id session-id) nil
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (let ((custom-id nil))
      (save-excursion
        (save-restriction
          (widen)
          (claude-org-workspace-bridge--find-workflow session-id)
          (let* ((workflow-level (org-current-level))
                 (instr-level (1+ workflow-level))
                 (stars (make-string instr-level ?*))
                 ;; Use regex to find workflow boundary (safe against embedded headings)
                 (workflow-end (claude-org-workspace-bridge--find-next-sibling-heading
                                workflow-level (point-max)))
                 (instr-count 0))
            ;; Count existing instructions by tag (headings may be renamed by auto-title)
            (save-excursion
              (while (re-search-forward
                      (format "^\\*\\{%d\\} +.+:%s:" instr-level claude-org-heading-tag)
                      workflow-end t)
                (setq instr-count (1+ instr-count))))
            (let* ((instr-num (1+ instr-count))
                   (file-base (file-name-sans-extension (file-name-nondirectory org-file))))
              (setq custom-id (claude-org--generate-instruction-custom-id file-base instr-num session-id))
              ;; Go to end of workflow subtree to insert new instruction
              (goto-char workflow-end)
              ;; Ensure blank line before heading
              (unless (bolp) (insert "\n"))
              (unless (save-excursion (forward-line -1) (looking-at "^[ \t]*$"))
                (insert "\n"))
              (insert (format "%s Instruction %d :%s:\n" stars instr-num claude-org-heading-tag))
              (let ((heading-marker (copy-marker (line-beginning-position 0))))
                (insert ":PROPERTIES:\n")
                (insert (format ":CUSTOM_ID: %s\n" custom-id))
                (insert ":END:\n\n")
                (insert "#+begin_src ai\n")
                (insert prompt "\n")
                (insert "#+end_src\n")
                ;; Register as active query for mode-line and query manager
                (goto-char heading-marker)
                (claude-org-terminal--register-query session-id nil)
                ;; Record in block history (enables "Jump to AI block" for terminal prompts)
                (let ((session-key (claude-org--current-session-key)))
                  (when session-key
                    (run-hook-with-args 'claude-org-execute-hook session-key prompt nil)
                    (claude-org--session-put session-key :busy t)
                    ;; Map workspace-session-id → session-key (shared across backends)
                    (puthash session-id session-key
                             claude-org-terminal--workspace-to-session-key)))
                ;; Set exec-status on the instruction heading
                (claude-org--set-exec-status "executing")
                ;; Auto-generate a meaningful title for the heading
                (when claude-org-auto-generate-title
                  (claude-org--maybe-auto-generate-title prompt heading-marker)))))))
      (save-buffer)
      custom-id))))
#+END_SRC

** Response Insertion

#+BEGIN_SRC elisp
(defun claude-org-workspace-bridge--shift-org-headings (text parent-level)
  "Shift org heading levels in TEXT to be subsections of PARENT-LEVEL.
Lines starting with *+ get PARENT-LEVEL additional stars prepended,
making them children of the parent heading."
  (let ((prefix (make-string parent-level ?*)))
    (replace-regexp-in-string
     "^\\(\\*+ \\)"
     (concat prefix "\\1")
     text)))

(defun claude-org-workspace-bridge--find-next-sibling-heading (level limit)
  "Find the next heading at LEVEL or shallower before LIMIT.
Searches by regex to avoid `org-end-of-subtree' confusion from
embedded org headings in response content.
Returns the position, or LIMIT if none found."
  (save-excursion
    ;; Skip current heading line
    (forward-line 1)
    (let ((pattern (format "^\\*\\{1,%d\\} " level)))
      (if (re-search-forward pattern limit t)
          (line-beginning-position)
        limit))))

(defun claude-org-workspace-bridge--find-heading-by-custom-id (custom-id limit)
  "Find heading with CUSTOM_ID before LIMIT.  Returns position or nil."
  (save-excursion
    (let ((pattern (format ":CUSTOM_ID: %s" (regexp-quote custom-id))))
      (when (re-search-forward pattern limit t)
        (org-back-to-heading t)
        (point)))))

(defun claude-org-workspace-bridge--find-next-instruction-heading (instr-level tag workflow-end)
  "Find the next instruction heading (with TAG) at INSTR-LEVEL before WORKFLOW-END.
Skips response headings (:ai_output:). Returns position, or WORKFLOW-END."
  (save-excursion
    (forward-line 1)
    (let ((pattern (format "^\\*\\{%d\\} +.+:%s:" instr-level tag)))
      (if (re-search-forward pattern workflow-end t)
          (line-beginning-position)
        workflow-end))))

(defun claude-org-workspace-bridge--find-existing-response (instr-pos instr-level next-instr-pos)
  "Find existing Response heading for instruction at INSTR-POS.
Searches between INSTR-POS and NEXT-INSTR-POS for a heading at INSTR-LEVEL
with :ai_output: tag.  Returns (BEG . END) of the response subtree, or nil."
  (save-excursion
    (goto-char instr-pos)
    (forward-line 1)
    (let ((pattern (format "^\\*\\{%d\\} +.+:ai_output:" instr-level)))
      (when (re-search-forward pattern next-instr-pos t)
        (let ((beg (line-beginning-position)))
          ;; Find end of response subtree (next same-level heading or next-instr-pos)
          (let ((end (claude-org-workspace-bridge--find-next-sibling-heading
                      instr-level next-instr-pos)))
            (cons beg end)))))))

(defun claude-org-workspace-bridge-insert-response (org-file session-id response instr-custom-id)
  "Insert RESPONSE after the Instruction with INSTR-CUSTOM-ID in ORG-FILE.
If a Response section already exists for this instruction, appends to it.
Otherwise creates a new Response section.
Org headings in response content are shifted to become subsections."
  (claude-agent-with-span "workspace-insert-response"
      (list :session-id session-id :custom-id instr-custom-id
            :response-length (length response)) nil
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-workspace-bridge--find-workflow session-id)
        (let* ((workflow-level (org-current-level))
               (instr-level (1+ workflow-level))
               (stars (make-string instr-level ?*))
               (workflow-end (claude-org-workspace-bridge--find-next-sibling-heading
                              workflow-level (point-max)))
               (instr-pos (claude-org-workspace-bridge--find-heading-by-custom-id
                           instr-custom-id workflow-end))
               (response-num 0))
          (unless instr-pos
            (error "Instruction %s not found under Workflow for session %s"
                   instr-custom-id session-id))
          ;; Count which instruction this is among siblings (1-indexed)
          (save-excursion
            (while (and (re-search-forward
                         (format "^\\*\\{%d\\} +.+:%s:" instr-level claude-org-heading-tag)
                         workflow-end t)
                        (<= (line-beginning-position) instr-pos))
              (setq response-num (1+ response-num))))
          ;; Find the next instruction heading (skip responses)
          (goto-char instr-pos)
          (let* ((next-instr-pos (claude-org-workspace-bridge--find-next-instruction-heading
                                  instr-level claude-org-heading-tag workflow-end))
                 (existing (claude-org-workspace-bridge--find-existing-response
                            instr-pos instr-level next-instr-pos)))
            (if existing
                ;; Append to existing response section (before its end)
                (let ((response-end (cdr existing)))
                  (goto-char response-end)
                  ;; Back up to insert before the next heading, after existing content
                  (unless (bolp) (insert "\n"))
                  (insert "\n" (claude-org-workspace-bridge--shift-org-headings response instr-level) "\n"))
              ;; Create new response section — insert just before the next instruction
              (let ((timestamp (format-time-string "%Y-%m-%d %H:%M")))
                (goto-char next-instr-pos)
                (unless (bolp) (insert "\n"))
                (unless (save-excursion (forward-line -1) (looking-at "^[ \t]*$"))
                  (insert "\n"))
                (insert (format "%s Response %d (%s) :ai_output:\n"
                                stars response-num timestamp))
                (insert ":PROPERTIES:\n:QUERY_TYPE: terminal\n:END:\n\n")
                (insert (claude-org-workspace-bridge--shift-org-headings response instr-level) "\n")))
            ;; Flip the instruction's status so the header-line shows
            ;; "completed" rather than staying stuck at "executing".
            ;; Previously no code path did this on terminal-typed prompts,
            ;; which is why 4 PCR-dev1 blocks remained "executing" even
            ;; after their responses arrived.
            (save-excursion
              (goto-char instr-pos)
              (org-entry-put nil claude-org-exec-status-property "completed"))))))
    (save-buffer)
    (format "Inserted Response (%d chars)" (length response)))))

(defun claude-org-workspace-bridge-latest-instruction-custom-id
    (org-file session-id)
  "Return the CUSTOM_ID of the most recent :claude_chat: instruction
under SESSION-ID's Workflow in ORG-FILE, or nil if none exists.

Used by the Python bridge when a Stop hook fires without a queued
custom-id — typically for `/loop' autonomous iterations whose
sentinel prompt (`<<autonomous-loop-dynamic>>') was filtered from
queue-append in `_handle_prompt'.  The sentinel response is a
continuation of the user's last real instruction, so we route it
there via `claude-org-workspace-bridge-insert-response' (which
appends to any existing Response section).

Walks the session's workflow subtree and returns the last (deepest
by position) heading that matches `claude-org-heading-tag'."
  (claude-agent-with-span "workspace-latest-instruction-custom-id"
      (list :session-id session-id) nil
    (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
      (save-excursion
        (save-restriction
          (widen)
          (claude-org-workspace-bridge--find-workflow session-id)
          (let* ((workflow-level (org-current-level))
                 (workflow-end (claude-org-workspace-bridge--find-next-sibling-heading
                                workflow-level (point-max)))
                 (instr-level (1+ workflow-level))
                 (latest-cid nil))
            (save-excursion
              (while (re-search-forward
                      (format "^\\*\\{%d\\} +.+:%s:" instr-level
                              claude-org-heading-tag)
                      workflow-end t)
                (save-excursion
                  (org-back-to-heading t)
                  (let ((cid (org-entry-get nil "CUSTOM_ID")))
                    (when cid (setq latest-cid cid))))))
            latest-cid))))))

(defun claude-org-workspace-bridge-mark-cancelled
    (org-file session-id instr-custom-id)
  "Flip INSTR-CUSTOM-ID's AI_EXEC_STATUS to \"cancelled\" in ORG-FILE.
Called by the workspace bridge when a queued prompt was superseded
(user typed a new prompt before Claude Code generated a reply to the
old one).  Leaves the AI block content untouched so the user can
re-execute manually."
  (claude-agent-with-span "workspace-mark-cancelled"
      (list :session-id session-id :custom-id instr-custom-id) nil
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-workspace-bridge--find-workflow session-id)
        (let* ((workflow-level (org-current-level))
               (workflow-end (claude-org-workspace-bridge--find-next-sibling-heading
                              workflow-level (point-max)))
               (instr-pos (claude-org-workspace-bridge--find-heading-by-custom-id
                           instr-custom-id workflow-end)))
          (when instr-pos
            (goto-char instr-pos)
            (org-entry-put nil claude-org-exec-status-property "cancelled")
            (save-buffer)
            (format "Marked %s cancelled" instr-custom-id))))))))

#+BEGIN_SRC elisp
(defun claude-org-workspace-bridge-save-cli-session
    (org-file session-id cli-session-id &optional property-name)
  "Save CLI-SESSION-ID as a property on the workspace section for SESSION-ID in ORG-FILE.
PROPERTY-NAME defaults to `claude-org-cli-session-property' (CLAUDE_CLI_SESSION).
Pass a different property name for other agents (e.g. COPILOT_CLI_SESSION).
Saves silently to persist immediately."
  (let ((prop (or property-name claude-org-cli-session-property)))
    (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
      (save-excursion
        (save-restriction
          (widen)
          (claude-org-workspace-bridge--goto-session session-id)
          (org-set-property prop cli-session-id)
          (let ((inhibit-message t))
            (save-buffer))
          cli-session-id)))))

  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-workspace-bridge--goto-session session-id)
        (org-entry-get nil claude-org-cli-session-property)))))

#+BEGIN_SRC elisp
(defun claude-org-workspace-bridge-update-todos (org-file session-id todos instr-custom-id)
  "Update checkbox list under the Instruction with INSTR-CUSTOM-ID.
TODOS is a list of plists with :content, :status, and :priority keys.
Replaces or creates the checkbox list between the src block and child headings."
  (with-current-buffer (claude-org-workspace-bridge--ensure-buffer org-file)
    (save-excursion
      (save-restriction
        (widen)
        (claude-org-workspace-bridge--find-workflow session-id)
        (let* ((workflow-level (org-current-level))
               (instr-level (1+ workflow-level))
               (workflow-end (claude-org-workspace-bridge--find-next-sibling-heading
                              workflow-level (point-max)))
               (last-instr-pos (claude-org-workspace-bridge--find-heading-by-custom-id
                                instr-custom-id workflow-end)))
          (unless last-instr-pos
            (error "Instruction %s not found under Workflow for session %s"
                   instr-custom-id session-id))
          (goto-char last-instr-pos)
          ;; Find the end of the src block (#+end_src)
          (let* ((instr-end (claude-org-workspace-bridge--find-next-sibling-heading
                             instr-level workflow-end))
                 (src-end nil)
                 (todo-start nil)
                 (todo-end nil))
            ;; Find #+end_src within this instruction
            (save-excursion
              (when (re-search-forward "^#\\+end_src" instr-end t)
                (forward-line 1)
                (setq src-end (point))))
            (when src-end
              ;; Find existing todo region: consecutive lines starting with "- ["
              (goto-char src-end)
              ;; Skip blank lines after src block
              (while (and (< (point) instr-end) (looking-at "^[ \t]*$"))
                (forward-line 1))
              (when (looking-at "^- \\[")
                (setq todo-start (point))
                (while (and (< (point) instr-end) (looking-at "^- \\["))
                  (forward-line 1))
                (setq todo-end (point)))
              ;; Build new checkbox text
              (let ((checkbox-text
                     (mapconcat
                      (lambda (todo)
                        (let ((content (plist-get todo :content))
                              (status (plist-get todo :status)))
                          (format "- [%s] %s"
                                  (if (string= status "completed") "X" " ")
                                  content)))
                      todos "\n")))
                (if todo-start
                    ;; Replace existing todo region
                    (progn
                      (delete-region todo-start todo-end)
                      (goto-char todo-start)
                      (insert checkbox-text "\n"))
                  ;; Insert after src block
                  (goto-char src-end)
                  (unless (looking-at "^[ \t]*$")
                    (insert "\n"))
                  (insert checkbox-text "\n"))
                ;; Update heading cookie
                (goto-char last-instr-pos)
                (let* ((done (cl-count-if
                              (lambda (todo) (string= (plist-get todo :status) "completed"))
                              todos))
                       (total (length todos))
                       (cookie (format " [%d/%d]" done total)))
                  ;; Remove existing cookie if any
                  (when (re-search-forward " \\[[0-9]+/[0-9]+\\]" (line-end-position) t)
                    (replace-match ""))
                  ;; Insert cookie before tags
                  (goto-char last-instr-pos)
                  (when (re-search-forward "\\( +:\\)" (line-end-position) t)
                    (goto-char (match-beginning 1))
                    (insert cookie)))))))
        (save-buffer)
        (format "Updated %d todos" (length todos))))))
